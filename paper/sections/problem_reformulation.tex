\subsection{W-type Regularizer: Definition and Properties}
\label{Section: W-regularizer}
Deep hashing aims to learn network parameters \(\bx \in \RR^{\blue{q}}\) that generate binary codes preserving semantic similarity in the Hamming space. A direct formulation is
\begin{align}
\label{Eq:direct_deep_hashing_problem}
\blue{\min_{\bx \in \RR^{q}}\ \frac{1}{n}\sum_{i=1}^{n}\ell_{s,i}\!\big(\operatorname{sgn}(\mathcal{D}(\bx))\big),}
\end{align}
where \(n\) is the number of training samples, \blue{\(\bD_i:\RR^q\to\RR^d\) is the continuous network output mapping for sample \(i\), \(\mathcal{D}(\bx):=[\bD_1(\bx)^\top;\ldots;\bD_n(\bx)^\top]\), and \(\ell_{s,i}:\RR^{n\times d}\to\RR\) is a similarity-preserving loss} (e.g., pairwise cross-entropy \cite{zhuDeepHashingNetwork2016}). This objective is hard to optimize by backpropagation because \(\operatorname{sgn}(\cdot)\) is nondifferentiable and has zero gradient almost everywhere. 

\blue{To address this issue, penalties with W-shaped graphs, vanishing exactly at
$\pm1$, are added to drive the continuous outputs toward $\{-1,1\}^d$. Such penalties
are standard in deep hashing~\cite{liuDeepSupervisedHashing,yuanCentralSimilarityQuantization2020,he2025mambahash,
yin2026crh}. Their formulas differ, but our analysis depends only on their common
shape, which we formalize next.}

\begin{definition}[W-type regularizer]\label{def:wtype}
Let \(\phi:\mathbb{R}\to[0,+\infty)\). For \(\bz\in\mathbb{R}^{d}\) and \(\bZ\in\mathbb{R}^{n\times d}\), define \(h(\bz):=\sum_{j=1}^{d}\phi(z_j)\) and \(h(\bZ):=\sum_{i=1}^{n}\sum_{j=1}^{d}\phi(z_{ij})\); with a slight abuse of notation, \(h\), \(h^*\), and \(h^{**}\) refer to both the vector and the matrix cases. We call \(h\) a \emph{W-type regularizer} if \(\phi\) satisfies:
\begin{enumerate}[label=\textup{(W\arabic*)}]
  \item\label{W1} \(\phi\) is even and continuous, and \(\phi(t)=0\) if and only if \(t\in\{-1,1\}\);
  \item\label{W2} \(\phi\) is convex on \([1,+\infty)\);
  \item\label{W3} \(\phi\) decreases at a nonvanishing rate on \([0,1)\): for every \(b\in(0,1)\) there exists \(c_b>0\) such that \(\phi(s)-\phi(t)\ge c_b(t-s)\) for all \(0\le s<t\le b\).
\end{enumerate}
\end{definition}
For W-type regularizers, the convex envelope and the conjugate admit the following explicit description. The proof is provided in \Cref{app:wtype-conjugate-proof}. 
\begin{lemma}\label{lem:wtype-conj}
Let \(\phi:\mathbb{R}\to[0,+\infty)\) satisfy \ref{W1}--\ref{W2}. Then:
\begin{enumerate}[label=\textup{(\alph*)}]
\item \(\phi^{**}(t)=\phi\bigl(\max\{|t|,1\}\bigr)\) for all \(t\in\mathbb{R}\); in particular, \(\phi^{**}(t)=\phi(t)\) if and only if \(|t|\ge1\);
\item \(\phi^*\) is proper, lower semicontinuous, convex, and even; \(\phi^*(y)\ge|y|\) for all \(y\); and \(\operatorname{dom}\phi^*\) has nonempty interior.
\end{enumerate}
Consequently, \(h^*(\bLmbd)=\sum_{i,j}\phi^*(\Lambda_{ij})\), \(h^{**}(\bB)=\sum_{i,j}\phi^{**}(b_{ij})\), and \(h(\bB)=h^{**}(\bB)\) if and only if \(|b_{ij}|\ge1\) for all \(i,j\).
\end{lemma}
 


\Cref{tab:wtype} summarizes representative W-type regularizers, their scalar conjugates, and their proximal mappings. Every row has $\partial\phi(0)=\{\pm c\}$ with $c>0$, a downward kink at the origin, so none of these penalties is weakly convex.

\begin{remark}[Role of the conditions]\label{rem:wtype-roles}
Conditions~\ref{W1}--\ref{W2} give the envelope formula in \Cref{lem:wtype-conj}, so the min--max reformulation keeps $h$ where $|b_{ij}|\ge1$ and replaces it by zero on $[-1,1]^{n\times d}$. Condition~\ref{W3} is used only in the necessity part of \Cref{prop:transfer}.
\end{remark}

We say that a W-type regularizer has a \emph{compact dual domain} if $\operatorname{dom}\phi^*=[-\lambda,\lambda]$ for some $\lambda>0$; then $\phi^*$, being convex and finite on a compact interval, is bounded on its domain. In \Cref{tab:wtype}, every row except $p=2$ has a compact dual domain.

For a W-type regularizer, the proximal mapping of \(h^*\) is separable entrywise:
\begin{equation}
\label{Eq: the proximal mapping of conjugate function}
[\operatorname{prox}_{\sigma h^*}(\bY)]_{ij}
=\operatorname{prox}_{\sigma\phi^*}(Y_{ij}),\qquad \sigma>0.
\end{equation}

\subsection{W-type Regularized Deep Hashing Model}
\label{Sec:W-type Regularized Deep Hashing Model}
\blue{With the W-type regularizer \(h\) defined in \Cref{def:wtype},} the model is now trained to produce the \(i\)-th continuous network output \(\bD_i(\bx)\), and binary codes are obtained post-training via \(\operatorname{sgn}(\bD_i(\bx))\blue{\in\{-1,1\}^d}\). \blue{Consequently, we formulate the following core optimization problem for W-type regularized deep hashing methods, which serves as a continuous surrogate of Problem~\eqref{Eq:direct_deep_hashing_problem}:}
\begin{align}
\label{Eq: the primal deep hashing problem}
\blue{\min_{\bx \in \RR^{q}} \quad J_0(\bx):=\frac{1}{n}\sum_{i=1}^n f_i(\bx) + \sum_{i=1}^n h(\bD_i(\bx)),}
\tag{P0}
\end{align}\blue{where
\(f_i(\bx):=\ell_{s,i}\!\left(\mathcal{D}(\bx)\right)+\frac{\mu}{2}\|\bx\|^2\) with \(\mu>0\), and \(\bar{\ell}_s:=n^{-1}\sum_{i=1}^{n}\ell_{s,i}\).}

\blue{We define the quantization error as
\begin{equation}
\label{eq:quantization-error}
Q(\bx):=\operatorname{dist}_1\!\left(\mathcal{D}(\bx),\{-1,1\}^{n\times d}\right)
=\sum_{i=1}^{n}\sum_{j=1}^{d}\bigl||\mathcal{D}_{ij}(\bx)|-1\bigr|.
\end{equation}
For the $\ell_1$ W-type regularizer $\phi(t)=\lambda\bigl||t|-1\bigr|$,
\[
h(\mathcal{D}(\bx))=\lambda Q(\bx).
\]
For every W-type regularizer,
\[
h(\mathcal{D}(\bx))=0\quad\Longleftrightarrow\quad Q(\bx)=0.
\]}


We then employ variable splitting by introducing \blue{continuous auxiliary variables \(\bb_i\in\RR^d\) and \(\bB=[\bb_1^\top;\ldots;\bb_n^\top]\in\RR^{n\times d}\), subject to \(\bb_i=\bD_i(\bx)\) for all \(i=1,\dots,n\),} which yields the following equivalent nonlinearly constrained problem:
\begin{align}
    \blue{\min_{\bx \in \RR^{q},\ \bB \in \RR^{n\times d}}\quad \frac{1}{n}\sum_{i=1}^{n} f_i(\bx)+\sum_{i=1}^{n}h(\bb_i)\quad \text{s.t.}\quad \bb_i=\bD_i(\bx),\ \forall i=1,\dots,n.} \notag
\end{align}
\blue{Let \(\gamma>0\) be a penalty parameter and define}
\begin{align}
    \blue{F(\bx,\bB):= \frac{1}{n}\sum_{i=1}^{n}F_i(\bx,\bb_i),\qquad F_i(\bx,\bb_i):= f_i(\bx)+\frac{\gamma}{2}\|\bD_i(\bx)-\bb_i\|^2.} \notag
\end{align}
\blue{Averaging over \(i\) gives the decomposition \(F=f+P\), where}
\begin{align}
    \blue{f(\bx):= \frac{1}{n}\sum_{i=1}^{n}f_i(\bx),\qquad P(\bx,\bB):= \frac{\gamma}{2n}\sum_{i=1}^{n}\|\bD_i(\bx)-\bb_i\|^2.} \notag
\end{align}
We then \blue{relax these constraints by} the quadratic penalty method \cite[Sec.~17.1]{nocedalWright2006} to obtain the following two-block finite-sum optimization problem:
\begin{align}
    \label{Prop: block-wise regularized problem}
    \blue{\min_{\bx \in \RR^{q},\,\bB \in \RR^{n\times d}}\ J(\bx,\bB):=F(\bx,\bB)+h(\bB).} \tag{P}
\end{align}

This reformulation decomposes the original problem \eqref{Eq: the primal deep hashing problem} into simpler structural blocks, enabling us to exploit the problem structure better. Building upon Problem~\eqref{Prop: block-wise regularized problem}, we develop a stochastic primal-dual algorithm {that addresses the lack of tractable proximal mappings in the nonsmooth composite form and the absence of convergence guarantees in existing deep hashing methods.}


\subsection{\blue{Basic Assumptions}}
\begin{assumption}
\label{Ass: basic regularity}
\leavevmode
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2em,font=\color{blue}]
    \item \blue{The average loss is bounded below: $\underline{\ell}:=\inf_{\bx}\bar{\ell}_{s}\!\left(\mathcal{D}(\bx)\right)>-\infty$.}
    \item \blue{The function $f:=n^{-1}\sum_{i=1}^{n}f_i$ is $L_f$-smooth on $\RR^q$, and $\nabla\mathcal{D}$ is $L_{\nabla\mathcal{D}}$-Lipschitz on $\RR^q$, with $L_f,L_{\nabla\mathcal{D}}>0$. There exist constants $C_{\mathcal{D}},G_{\mathcal{D}}>0$ such that
    \[
        \|\mathcal{D}(\bx)\|_F\le C_{\mathcal{D}},\qquad
        \|\nabla\mathcal{D}(\bx)\|\le G_{\mathcal{D}},\qquad
        \forall\,\bx\in\RR^q.
    \]}
\end{enumerate}
\end{assumption}
