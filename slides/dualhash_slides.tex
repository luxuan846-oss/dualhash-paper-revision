\begin{frame}[t]{Original Mathematical Formulation in Deep Hashing}
% \small
\vspace{0.1cm}

\[
\min_{\bx\in\RR^{d_{\bx}}}\
\frac{1}{n}\sum_{i=1}^{n}
\ell_{s,i}\bigl(\red{\operatorname{sgn}}(\bC_i(\bx))\bigr).
\]

\vspace{-0.06cm}
\begin{itemize}
  % \setlength{\itemsep}{0.08cm}
  \item $\bx\in\RR^{d_{\bx}}$ denotes the trainable neural network parameters.
  \item $\ell_{s,i}$ denotes the similarity-preserving loss for the $i$-th sample (e.g., pairwise cross-entropy).
  \item $\bC_i(\bx)\in\RR^d$ denotes the output for the $i$-th sample.
  \item $\operatorname{sgn}(\bC_i(\bx))\in\{-1,+1\}^d$ denotes the binary hash code.
\end{itemize}

\vspace{0.06cm}
\begin{block}{Quantization Difficulties}
\small
\begin{itemize}
% \setlength{\itemsep}{0.08cm}
\item[$\triangleright$] \red{Zero gradient of $\operatorname{sgn}$:}
\[
\frac{\operatorname{sgn}(2.5)-\operatorname{sgn}(0.1)}{2.5-0.1}=0.
\]
\item[$\triangleright$] \red{Loss of magnitude information of the output:}
\[
|0.9-1|\ll|2.5-1|.
\]
\end{itemize}
\end{block}

\vspace{-0.04cm}
% \begin{center}\includegraphics[width=0.72\textwidth,height=0.25\textheight,keepaspectratio]{papers_img/quantization_error_cha.png}
% \end{center}

\end{frame}


\begin{frame}[t]{From Discrete Model to Continuous Optimization}
\small
\vspace{0.04cm}

\begin{block}{Continuous Solution}
\begin{itemize}
  \setlength{\itemsep}{0.04cm}
  \item[$\triangleright$] \textbf{Continuous relaxation:} 
  $\operatorname{sgn}(\bC_i(\bx)) \Longrightarrow \bD_i(\bx)=\tanh(\bC_i(\bx))$ during training.
  \item[$\triangleright$] \textbf{W-type regularization:} use
  \(h(\bz)=\lambda\bigl\||\bz|-\One\bigr\|_1\), with \(\lambda>0\),
  to drive the continuous outputs toward $\{-1,+1\}^d$.
\end{itemize}
\end{block}

\begin{frame}{From Discrete Model to Continuous Optimization}
% \small
\vspace{-0.6cm}

% \vspace{-0.12cm}
\begin{align*}
\min_{\bx\in\RR^{d_{\bx}}}\quad
\frac{1}{n}\sum_{i=1}^{n}
\ell_{s,i}\!\left(\red{\operatorname{sgn}\!\left(\bC_i(\bx)\right)}\right)
\end{align*}

\vspace{-0.18cm}
\begin{center}
  {\blue{\Large $\Downarrow$}}\\[-0.12cm]
\end{center}

\vspace{-0.12cm}
\begin{center}
{\textbf{Continuous W-regularized model}}
\end{center}
\vspace{-0.14cm}
\begin{align}
\min_{\bx\in\RR^{d_{\bx}}}\quad \{f(\bx)=
\frac{1}{n}\sum_{i=1}^n f_i(\bx)\}
+\sum_{i=1}^n h\!\left(\blue{\bD_i(\bx)}\right).
\tag{\blue{P0}}
\end{align}

\vspace{-0.18cm}
\begin{itemize}
  \setlength{\itemsep}{0.01cm}
  \item Finite-sum optimization: $n$ can be large.
  \item The loss function $f_i(\bx)$ and
        $\bD_i(\bx)$ are continuously differentiable.
  \item $h$ is a proper l.s.c. nonconvex regularizer.
\end{itemize}

\vfill
\begin{center}
  \blue{\textbf{Our goal: efficiently optimize the W-type regularized problem (P0).}}
\end{center}
\vspace{0.02cm}
\end{frame}

\begin{frame}[t]{Related Work: Nonsmooth Composite Optimization}
\small
\vspace{-0.4cm}
\[\min\limits_{\bx} f(\bx)+h(\bD(\bx)), \] where
\[
\bD \text{ nonconvex}, \quad h \text{ nonconvex and nonsmooth}.
\]

\par\vspace{0.1cm}

{\red{\textbf{Challenges}}}
\vspace{-0.08cm}
\begin{itemize}
  \setlength{\itemsep}{0.012cm}
  \item[\blue{$\blacktriangleright$}] {{\textbf{Stochastic proximal methods}}}
  {\footnotesize \blue{(Liu et al. 16; Cao et al. 17; Jiang et al. 18; Bai et al. 19)}}\\[-0.07cm]
  \begin{itemize}
    \setlength{\itemsep}{0pt}
    \item[\blue{$\bullet$}] {\footnotesize {\(\prox_{h\circ \bD}\) generally unavailable.}}
  \end{itemize}

  \item[\blue{$\blacktriangleright$}] {{\textbf{Nonlinear composite methods}}}
  {\footnotesize \blue{(Davis--Drusvyatskiy 19; Yao et al. 24; Zhang et al. 25)}}\\[-0.07cm]
  \begin{itemize}
    \setlength{\itemsep}{0pt}
    \item[\blue{$\bullet$}] {\footnotesize Weak convexity is typically required; nonconvex \(h\) is difficult.}
  \end{itemize}

  \item[\blue{$\blacktriangleright$}] {{\textbf{ADMM / primal-dual splitting}}}
  {\footnotesize \blue{(Jiang et al. 19; Bolte et al. 18; Zhu et al. 24; Chen et al. 25)}}\\[-0.07cm]
  \begin{itemize}
    \setlength{\itemsep}{0pt}
    \item[\blue{$\bullet$}] {\footnotesize {Nonlinear \(\bD_i(\bx)\), nonconvex \(h\); multiplier boundedness may be hard.}}
  \end{itemize}
\end{itemize}

\vspace{0.2cm}
{\blue{\textbf{Possible solutions}}}
\vspace{-0.08cm}
\begin{itemize}
  \setlength{\itemsep}{0.005cm}
  \item[\blue{$\blacktriangleright$}] {{\textbf{Primal-Dual Hybrid Gradient (PDHG)}}}
  {\footnotesize {(Chambolle--Pock 11)}}\\[-0.07cm]
  {
  \[
  {\min_{\bx}\; f(\bx)+h(\bA\bx)
  \quad\overset{h\ \mathrm{convex}}{\Longleftrightarrow}\quad
  \min_{\bx}\max_{\by}\; f(\bx)+\langle\bA\bx,\by\rangle-h^*(\by).}
  \]
  }
    \begin{itemize}
    \setlength{\itemsep}{0pt}
    \item[\blue{$\bullet$}] {\footnotesize Variants for convex problems: Stochastic PDHG \blue{(Chambolle et al. 18)}; Nonlinear PDHG \blue{(Clason et al. 19; Gao--Zhang 23; Shumaylov et al. 24)}};
  \end{itemize}

  \item[\blue{$\blacktriangleright$}] {{\textbf{Proximal updates in PDHG}}}
  {\footnotesize
  \[
  {\bx^{k+1}=\prox_{\sigma f}\!\bigl(\bx^k-\sigma\bA^\top\by^k\bigr),}
  \]
  \[
  {\by^{k+1}=\prox_{\tau h^*}\!\bigl(\by^k+\tau\bA(\bx^{k+1}+\theta(\bx^{k+1}-\bx^k))\bigr),\quad 0<\theta\leq 1.}
  \]
  }

\end{itemize}
\end{frame}

\vspace{0.24cm}
\[
h^*(\bLmbd)=\sum_{i=1}^n\sum_{j=1}^d h_{ij}^*(\Lambda_{ij}),
\quad
h_{ij}^*(\Lambda_{ij})
=
\begin{cases}
|\Lambda_{ij}|, & |\Lambda_{ij}|\le \lambda,\\
+\infty, & |\Lambda_{ij}|>\lambda.
\end{cases}
\]

\vspace{-0.1cm}

\[
[\prox_{s h^*}(\bY)]_{ij}
=
\operatorname{sgn}(Y_{ij})\,
\min\big\{(|Y_{ij}|-s)_+,\lambda\big\}.
% \qquad s=\tau^{-1}\ \text{in DualHash}.
\]
\end{itemize}

\end{frame}


\begin{frame}{Primal-Dual Decoupling Reformulation}
% \small
\vspace{0.1cm}
\begin{itemize}
\item[\blue{$\blacktriangleright$}] \textbf{Primal Problem (variable splitting).}
Introduce $\bb_i=\bD_i(\bx)$ and $\bB=[\bb_1,\ldots,\bb_n]^\top\in\RR^{n\times d}$, and apply \blue{the quadratic penalty} to (P0):
\begin{align}
\label{eq:primal}
\min_{\bx\in\RR^{d_{\bx}},\,\bB\in\RR^{n\times d}}\;
J(\bx,\bB)
=
F(\bx,\bB) + h(\bB), \tag{\blue{P}}
\end{align}
% each $\bb_i$ denotes the $i$-th row of $\bB$, and
% $\gamma>0$ decouples $h(\bB)$ from $\bD(\bx)$.
where $F(\bx,\bB)=\frac{1}{n}\sum_{i=1}^n F_i(\bx,\bb_i)$
with $F_i(\bx,\bb_i):=f_i(\bx)+\blue{\frac{\gamma}{2}\|\bD_i(\bx)-\bb_i\|^2,}$ 
and $h(\bB):=\sum_{i=1}^n h(\bb_i)$.
\vspace{0.3cm}

\item[\blue{$\blacktriangleright$}] \textbf{Dual Problem (Fenchel conjugate).}
Recalling the definition of the conjugate function
$h^*(\bLmbd)=\sup_{\bY}\{\langle\bLmbd,\bY\rangle-h(\bY)\}$, we obtain:

\begin{align}
&\max_{\bLmbd}\min_{\bx,\,\bB,\,\blue{\bY}}\;F(\bx,\bB)+h(\blue{\bY})+\langle\bLmbd,\bB-\blue{\bY}\rangle \notag\\
% &= \max_{\bLmbd}\min_{\bx,\,\bB}\;F(\bx,\bB)+\langle\bLmbd,\bB\rangle + \min_{\bY}\{h(\bY)-\langle\bLmbd,\bY\rangle\} \notag\\
=& \max_{\bLmbd}\min_{\bx,\,\bB}\;F(\bx,\bB)+\langle\bLmbd,\bB\rangle - h^*(\bLmbd), \tag{\blue{D}}\label{eq:dual}
\end{align}
\end{itemize}
\end{frame}

\section{DualHash: A Stochastic Primal-Dual Algorithm}

\begin{frame}{DualHash: Three-Block Alternating Update Framework}
%\vspace{0.1cm}
\small

\begin{gather}
% &= \max_{\bLmbd}\min_{\bx,\,\bB}\;F(\bx,\bB)+\langle\bLmbd,\bB\rangle + \min_{\bY}\{h(\bY)-\langle\bLmbd,\bY\rangle\} \notag\\
\max_{\bLmbd}\min_{\bx,\,\bB}\;F(\bx,\bB)+\langle\bLmbd,\bB\rangle-h^*(\bLmbd), \tag{\blue{D}}
% \\[-0.25em]
% \text{where } . \notag
\end{gather}
where $F(\bx,\bB)=\tfrac{1}{n}\sum_{i=1}^n f_i(\bx)+\tfrac{\gamma}{2n}\|\bD(\bx)-\bB\|^2$.


\nii\textbf{Idea:} Employ an alternating update scheme for $(\bx,\bB,\bLmbd)$.

\vspace{0.15cm}
\begin{itemize}
\item[\blue{$\triangleright$}] \textbf{$\bx$-update (stochastic gradient descent on $\calL$):}
\[
\bx^{k+1} = \bx^k - \eta_k\,\widetilde\nabla_{\bx} F(\bx^k,\bB^k),
\]
where $\widetilde\nabla_{\bx} F$ uses \blue{inertial acceleration} or \blue{variance reduction}.

\vspace{0.05cm}
\item[\blue{$\triangleright$}] \textbf{$\bB$-update (gradient descent on $\calL$):}
\[
\bB^{k+1}
= \bB^k - \tau\bigl(\nabla_{\bB} F(\bx^{k+1},\bB^k) + \bLmbd^k\bigr).
\]

\vspace{0.05cm}
\item[\blue{$\triangleright$}] \textbf{$\bLmbd$-update (proximal ascent with extrapolation):}
\[
\bLmbd^{k+1} = \prox_{\blue{\tau^{-1}} h^*}\!\bigl(\bLmbd^k + \blue{\tau^{-1}}(2\bB^{k+1}-\bB^k)\bigr).
\]
\end{itemize}

\vspace{0.1cm}
% \begin{block}{Key ingredients}
% The $\bx$-update admits two variants: \textbf{DualHash-StoM} (inertial) and \textbf{DualHash-StoRM} (recursive momentum).
% \end{block}
\end{frame}


\begin{frame}{Two Instances of the $\bx$-Update}
\vspace{0.1cm}

% \nii Both instances follow the generic $\bx$-update with a stochastic gradient \blue{estimator}:
\[
\bx^{k+1} = \bx^k - \eta_k\, \blue{\widetilde\nabla_{\bx} F(\bx^k,\bB^k)}.
\]

\begin{itemize}
\item[\blue{$\triangleright$}] \textbf{Instance 1 (DualHash-StoM): inertial momentum,}
estimator at the extrapolated point $\bz^k=\bx^k+\beta_k(\bx^k-\bx^{k-1})$, step from $\by^k=\bx^k+\alpha_k(\bx^k-\bx^{k-1})$:
\[
\blue{\bG^k} = \nabla_{\bx} F(\bz^k,\bB^k;\calJ_k),
\] where $\nabla_{\bx}F(\cdot;\calJ_k):=\frac{1}{|\calJ_k|}\sum_{i\in\calJ_k}\nabla_{\bx}F_i(\cdot,\bb_i)$ is the mini-batch gradient.

\item[\blue{$\triangleright$}] \textbf{Instance 2 (DualHash-StoRM): recursive momentum}\footnote{Cutkosky A., Orabona F. Momentum-based variance reduction in non-convex SGD. NeurIPS, 2019: 15236--15245.}
{\small
\[
\blue{\bD^k} =
\begin{cases}
\nabla_{\bx} F(\bx^1,\bB^1;\calJ_1), & k=1,\\
\nabla_{\bx} F(\bx^k,\bB^k;\calJ_k) + (1-\rho_{k-1})\bigl(\bD^{k-1}-\nabla_{\bx} F(\bx^{k-1},\bB^{k-1};\calJ_k)\bigr), & k>1.
\end{cases}
\]
}
\end{itemize}

\vspace{0.1cm}
\begin{block}{Remark}
Variance reduction \blue{without} full-gradient or large-batch computation \blue{at every iteration}.
\end{block}
\end{frame}

\section{Complexity Results}

\begin{frame}[t]{Main Assumptions}
\small
\vspace{0.1cm}
\begin{equation}
  \max_{\bLmbd}\min_{\bx,\bB}\;
\calL(\bx,\bB,\bLmbd)
:=
F(\bx,\bB)
+\langle\bB,\bLmbd\rangle
-h^*(\bLmbd), \tag{\blue{D}}
\end{equation}

\vspace{0.2cm}
\nii\textbf{AS.1}
\begin{itemize}
\item[(i)] The objective of Problem (P) is lower bounded by $J^*$.
\item[(ii)] $F(\bx,\bB)$ is $L_F=\calO(1)$-smooth w.r.t.\ $(\bx,\bB)$.
\item[(iii)] For $i\sim\mathrm{Unif}\{1,\dots,n\}$: $\EE_i[\nabla_{\bx}F_i(\bx,\bb_i)]=\nabla_{\bx}F(\bx,\bB)$ and $\EE_i[\|\nabla_{\bx}F_i-\nabla_{\bx}F\|^2]\le\sigma^2$.
\end{itemize}

\vspace{0.2cm}
\textbf{AS.2 (Mean-squared smoothness).}\footnote{Xu Yangyang, Xu Yibo, Momentum-based variance-reduced proximal stochastic gradient method for composite nonconvex stochastic optimization. Journal of Optimization Theory and Applications, 2023, 196(1): 266--297.}
\[
\EE_i\!\left[\|\nabla_{\bx}F_i(\bx,\bb_i)-\nabla_{\bx}F_i(\bx',\bb_i')\|^2\right]\le L_F^2\!\left[\|\bx-\bx'\|^2+\|\bB-\bB'\|^2\right],
\]
Here, $\bb_i$ and $\bb_i'$ denote the $i$-th rows of $\bB$ and $\bB'$, respectively.

\vspace{0.08cm}
\begin{block}{Lemma 1}
\footnotesize
Under AS.1(i), the sequence $\{(\bx^k,\bB^k,\bLmbd^k)\}_{k\in\mathbb{N}}$ generated by
DualHash is bounded.
\end{block}
\end{frame}

\begin{frame}[t]{Convergence Criterion: Stochastic $\varepsilon$-Critical Point}
% \small
\vspace{0.1cm}

We analyze the complexity in terms of the number of
\blue{stochastic first-order oracle (SFO)} calls required to find a
\blue{stochastic $\varepsilon$-critical point}
of the Lagrangian function $\calL$.

\vspace{0.1cm}
\begin{itemize}
  \item \textbf{Critical point.} A primal-dual triple $(\bar{\bx},\bar{\bB},\bar{\bLmbd})$
  with $\bm{0}\in\partial\calL(\bar{\bx},\bar{\bB},\bar{\bLmbd})$.
  \item \textbf{Specifically, we seek a random output}
  $(\bx^R,\bB^R,\bLmbd^R)$ \textbf{satisfying}
% \vspace{-0.10cm}

\[
\EE\!\left[
\operatorname{dist}^2
\bigl(\bm{0},\partial\calL(\bx^R,\bB^R,\bLmbd^R)\bigr)
\right]
\le \varepsilon^2.
\]
% i.e.,
% \vspace{-0.24cm}
% \[
% \begin{aligned}
% \left[
% \EE\!\left[
% \|\nabla_{\bx}F(\bx^R,\bB^R)\|^2
% +
% \|\nabla_{\bB}F(\bx^R,\bB^R)+\bLmbd^R\|^2
% \right]
% \right]^{1/2}
% &\le \varepsilon,\\
% \left[
% \EE\!\left[
% \operatorname{dist}^2
% \!\left(\bB^R,\partial h^*(\bLmbd^R)\right)
% \right]
% \right]^{1/2}
% &\le \varepsilon .
% \end{aligned}
% \]

\end{itemize}


\vspace{0.04cm}
\begin{block}{Lemma 2}
Under AS.1(i), if
$(\bx^*,\bB^*,\bLmbd^*)$ is an optimal solution of Problem $(P)$
and its dual $(D)$, then
$\bm{0}\in \partial\calL(\bx^*,\bB^*,\bLmbd^*).$
\end{block}
\end{frame}

\begin{frame}{Oracle Complexity of DualHash-StoM}
\vspace{0.1cm}
\small

\nii\textbf{Lyapunov function:}

\[
\Psi_m(\calQ^k)=\calL(\bx^k,\bB^k,\bLmbd^k)-C_1\Delta_{\bB}^{k+1}+C_2\Delta_{\bB}^k-C_3\Delta_{\bx}^{k+1}+C_4\Delta_{\bx}^k,
\]
where $\calQ^k:=\bigl\{\bx^k,\bB^k,\bLmbd^k,\bB^{k+1},\bB^{k-1},\bx^{k+1},\bx^{k-1}\bigr\},$ $\Delta_{\bx}^{k}:=\tfrac{1}{2}\|\bx^{k}-\bx^{k-1}\|^2$
and $\Delta_{\bB}^{k}:=\tfrac{1}{2}\|\bB^{k}-\bB^{k-1}\|^2$.

 \bigskip

\nii\textbf{Parameter setting:}
\begin{equation}
  \eta_k=\frac{\eta}{L_F},\quad \alpha_k=\alpha,\quad \beta_k=\beta,\quad \tau\le\frac{\sqrt{\tilde\delta}}{L_F},\quad b_k=b=c_b \red{T,} \quad \forall k \geq 1. \tag{S.1}
\end{equation}
\vspace{0.2cm}
\begin{block}{Theorem 1 (DualHash-StoM)}
Under AS.1 and condition (S.1), there exists $R\sim\mathrm{Unif}\{2,\dots,T+1\}$ such that $(\bx^R,\bB^R,\bLmbd^R)$ is a stochastic $\varepsilon$-critical point, provided that
\[
T=\left\lceil\frac{L_\Delta\Delta_1+C_\sigma\sigma^2/c_b}{\varepsilon^2}\right\rceil.
\] Consequently, the total number of SFO calls is \blue{$\calO(\varepsilon^{-4})$}.
\end{block}
\end{frame}


\begin{frame}{Oracle Complexity of DualHash-StoRM}
\vspace{0.1cm}

\nii\textbf{Lyapunov function:}

\Psi_r(\calQ^k)=\calL(\bx^k,\bB^k,\bLmbd^k)-C_1\Delta_{\bB}^{k+1}+C_2\Delta_{\bB}^k-C_3\Delta_{\bx}^{k+1}.
\]

\nii\textbf{Parameter setting:}
\begin{equation}
  \eta_k=\frac{\eta}{\tilde L_F T^{1/3}},\quad \rho_k=\frac{8\rho\eta^2}{T^{2/3}},\quad \tau\le\frac{\sqrt{\tilde\delta}}{L_F},\quad b_1=c_b T^{1/3}, \quad b_k = \calO(1), \quad \forall k \geq 2.  \tag{S.2}
\end{equation}

\vspace{0.2cm}
\begin{block}{Theorem 2 (DualHash-StoRM)}
Under AS.1(i), (iii), AS.2 and condition (S.2), there exists $R\sim\mathrm{Unif}\{2,\dots,T+1\}$ such that $(\bx^R,\bB^R,\bLmbd^R)$ is a stochastic $\varepsilon$-critical point, provided that

\] Consequently, the total number of SFO calls is \blue{$\calO(\varepsilon^{-3})$}.
\end{block}
\end{frame}

\section{Numerical Experiments}

\begin{frame}{Experimental Setup}
\small
\vspace{0.1cm}

We evaluate on \textbf{CIFAR-10}, \textbf{NUS-WIDE}, and \textbf{ImageNet-100} datasets, covering single-label, multi-label, and large-scale image retrieval settings.

\begin{itemize}
\item[\blue{$\blacktriangleright$}] \textbf{The W-regularized deep hashing problem.}
{\footnotesize
\begin{align*}
\min_{\bx,\bB}\quad
-\frac{1}{|\mathcal{S}|}\sum_{s^o_{ij}\in\mathcal{S}}
\left\{ 
\log\!\left(1+e^{-\tilde\alpha s^o_{ij}s^h_{ij}}\right)+\frac{\mu}{2}\|\bx\|_2^2 \right\} +\frac{\gamma}{2n}\sum_{i=1}^n\|\bb_i-\bu_i\|_2^2
+\sum_{i=1}^n\lambda\|\,|\bb_i|-\bm{1}\,\|_1,
\end{align*}} where \(s^h_{ij}=\tfrac12\langle\bu_i,\bu_j\rangle\) denotes the
pairwise similarity, \(\bu_i=\tanh(\bC(\bx;\ba_i))\) is the continuous network output,
and \(\bb_i\) is an auxiliary code approximating the binary code; \(\mu>0\) ensures coercivity.

\item[\blue{$\blacktriangleright$}] \textbf{Baselines.} We compare with three groups of methods:
(i) \textbf{SDH}~\blue{[SSLS15]} as the discrete hashing baseline;
(ii) \textbf{W-type regularization methods}: DHN~\blue{[ZLWC16]}, DSH~\blue{[LWSC16]}, DTSH~\blue{[WSK16]}, and MDSHC~\blue{[WPL+23]};
(iii) \textbf{quantization-error-oriented methods}: HashNet~\blue{[CLWY17]}, DSDH~\blue{[LSHT17]}, and OrthoHash~\blue{[HNZ+21]}.
\item[\blue{$\blacktriangleright$}] \textbf{Backbones.} AlexNet and ResNet-50, with ELU (smooth) activation\footnote{The gradient of ELU is 1-Lipschitz continuous.}.
\item[\blue{$\blacktriangleright$}] \textbf{Main Metrics.} mAP and quantization error.
\end{itemize}
\end{frame}

\begin{frame}{Summary}
\vspace{0.1cm}

\begin{itemize}
\item[\blue{$\bullet$}] \textbf{DualHash:} a stochastic primal-dual algorithm for deep hashing via Fenchel duality.
\begin{itemize}
\item Handles \blue{nonsmooth W-type regularization} with nonlinear composition.
\item Closed-form dual proximal update bypasses the intractable $\prox_{h\circ\bD}$.
\item Oracle complexity results:
\[
\text{DualHash-StoM:}\ \calO(\varepsilon^{-4}),\qquad
\text{DualHash-StoRM:}\ \calO(\varepsilon^{-3}).
\]
\end{itemize}
\item  {Future work}
\begin{itemize}
 \item Extend the decoupling-dual-structure principle to quantization-aware training of low-bit neural networks.
\item Characterize the gap between stationary points of (P) and the original composite problem as penalty parameter $\gamma\to\infty$.
\end{itemize}

\end{itemize}
