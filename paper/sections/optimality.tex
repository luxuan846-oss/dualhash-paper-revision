\subsection{\textcolor{blue}{Min--Max Reformulation and KKT Conditions}}
\label{Sec: first-order optimality conditions}
\blue{To derive a partial Fenchel dual formulation, we} consider the equivalent constrained formulation of Problem~\eqref{Prop: block-wise regularized problem}:
\begin{align}
\min_{\bx \in \RR^{\blue{q}}, \bB \in \RR^{n\times d}, \bY \in \RR^{n\times d}} &\quad F(\bx, \bB) + h(\bY) \notag \\
\text{s.t. } &\quad \bY = \bB, \notag
\end{align}
whose dual problem is
\begin{equation}\label{Prop: the dual problem}
\max_{\bLmbd \in \RR^{n\times d}} \min_{\bx \in \RR^{\blue{q}}, \bB \in \RR^{n\times d}}\quad \calL(\bx,\bB,\bLmbd):=F(\bx,\bB)+\langle\bB,\bLmbd\rangle-h^*(\bLmbd), \tag{D}
\end{equation}
\begingroup\color{blue}
which is obtained by minimizing over $\bY$ using the definition of the conjugate function $h^*$.

It is worth noting that the nonconvex--concave min--max problem associated with $\calL$ is closely related to Problem~\eqref{Prop: the dual problem} and admits the following equivalent formulation:
\begin{equation}\label{Eq: minmax problem}
\min_{\bx,\bB}\ F(\bx,\bB)+h^{**}(\bB) = \min_{\bx,\bB}\max_{\bLmbd}\calL(\bx,\bB,\bLmbd).
\end{equation}

In general, a saddle point of $\calL$ may fail to exist, and even checking whether a point is a local minimizer of the nonconvex problem \eqref{Prop: block-wise regularized problem} is NP-hard for nonconvex quadratic programs~\cite{Murty1987NP}. We therefore seek points satisfying the first-order necessary optimality condition for the min--max problem~\eqref{Eq: minmax problem}. Following~\cite{chambolleFirstOrderPrimalDualAlgorithm2011,
clasonAccelerationGlobalConvergence2019,
botBohm2023alternating}, we call
$(\tilde{\bx},\tilde{\bB},\tilde{\bLmbd})$ a
\emph{KKT point} of $\calL$ if
$\bm{0}\in\mathcal{K}(\tilde{\bx},\tilde{\bB},\tilde{\bLmbd})$,
where
\begin{equation}\label{Eq: KKT mapping}
\mathcal{K}(\bx,\bB,\bLmbd):=
\begin{pmatrix}
\nabla_{\bx}F(\bx,\bB)\\
\nabla_{\bB}F(\bx,\bB)+\bLmbd\\
\bB-\partial h^*(\bLmbd)
\end{pmatrix}
=
\begin{pmatrix}
\nabla_{\bx}\calL(\bx,\bB,\bLmbd)\\
\nabla_{\bB}\calL(\bx,\bB,\bLmbd)\\
-\partial_{\bLmbd}(-\calL)(\bx,\bB,\bLmbd)
\end{pmatrix}.
\end{equation}
Equivalently,
\begin{equation}\label{Eq: KKT conditions}
\nabla_{\bx}F(\tilde{\bx},\tilde{\bB})=\bm{0},
\qquad
\nabla_{\bB}F(\tilde{\bx},\tilde{\bB})
+\tilde{\bLmbd}=\bm{0},
\qquad
\tilde{\bB}\in\partial h^*(\tilde{\bLmbd}).
\end{equation}
% Notice that the last relation in \eqref{Eq: KKT conditions}
% is the first-order condition for maximizing $\calL$ with
% respect to $\bLmbd$, namely
% $\bm{0}\in\partial_{\bLmbd}(-\calL)$.
% In contrast, for standard Lagrangian-based methods such as
% ADMM and ALM~\cite{wangGlobalConvergenceADMM2018,Bolte2018},
% the multiplier enters affinely, so
% $\bm{0}\in\partial\calL$ yields the usual KKT system,
% with the multiplier component enforcing primal feasibility.
% In our formulation, however, the nonsmooth concave term
% $-h^*(\bLmbd)$ prevents identifying
% $\bm{0}\in\partial\calL$ with
% $\bm{0}\in\mathcal{K}$. We therefore use
% $\bm{0}\in\mathcal{K}$. 

Notice that KKT points of $\calL$ correspond one-to-one to stationary points of the min--max problem~\eqref{Eq: minmax problem}, i.e.,
\begin{equation}\label{Eq: saddle-KKT-convexified}
\nabla_{\bx}F(\tilde{\bx},\tilde{\bB})=\bm{0},
\qquad
\nabla_{\bB}F(\tilde{\bx},\tilde{\bB})+\tilde{\bLmbd}=\bm{0},
\qquad
\tilde{\bLmbd}\in\partial h^{**}(\tilde{\bB}),
\end{equation}which motivates the characterization of when a KKT point of $\calL$ is also stationary for Problem~\eqref{Prop: block-wise regularized problem}. In the following, we denote by $\operatorname{crit}\calL:=\{(\bx, \bB, \bLmbd) \in \RR^{q} \times \RR^{n\times d} \times \RR^{n\times d}: \bm{0} \in \mathcal K(\bx, \bB, \bLmbd)\}$ the set of KKT points of $\calL$.
\endgroup

\subsection{\blue{Exact Stationarity Relations for Problem~\eqref{Prop: block-wise regularized problem}}}
\label{Sec: original optimality relations}
\begingroup\color{blue}
In this subsection, we relate KKT points of $\calL$ to Problem~\eqref{Prop: block-wise regularized problem}, first through the duality gap and then through stationarity.

Recall that strong duality holds for Problems~\eqref{Prop: block-wise regularized problem} and~\eqref{Prop: the dual problem} if their duality gap is zero and both optimal values are attained. Here, zero duality gap and strong duality are equivalent, since both problems admit optimal solutions by the coercivity of $J$ and $\phi^*(y)\ge|y|$ (Lemma~\ref{lem:wtype-conj}).
We first prove that any primal--dual optimal solution $(\bx^*,\bB^*,\bLmbd^*)$ with zero duality gap is a KKT point of $\calL$. The proof is provided in \Cref{app:zero-gap-proof}.
\begin{proposition}\label{prop:zerogap}
Suppose that $(\bx^*,\bB^*)$ and $\bLmbd^*$ are optimal solutions of Problems~\eqref{Prop: block-wise regularized problem} and~\eqref{Prop: the dual problem}, respectively, with zero duality gap. Then $(\bx^*,\bB^*,\bLmbd^*)\in\operatorname{crit}\calL$.
\end{proposition}
While \Cref{prop:zerogap} requires a zero duality gap, this gap can be positive even when $F$ is convex, due to replacing $h$ with $h^{**}$. To see this, let $p^*$, $p^{**}$, and $d^*$ denote the optimal values of Problem~\eqref{Prop: block-wise regularized problem}, of Problem~\eqref{Eq: minmax problem}, and of Problem~\eqref{Prop: the dual problem}, respectively. The duality gap splits as
\[
p^*-d^*=(p^*-p^{**})+(p^{**}-d^*),
\]
where the first term is the loss caused by replacing $h$ with $h^{**}$, and the second term is the min--max gap of $\calL$. \Cref{ex:gap} illustrates this with a convex $F$, for which $p^{**}=d^*$.
\begin{example}\label{ex:gap}
Consider the problem
\[
p^*=\min_{b\in\RR}\Bigl\{\tfrac12 b^2+\bigl||b|-1\bigr|\Bigr\}.
\]
Its optimal value is $p^*=\tfrac12$, attained at $b^*=\pm1$.
The dual value is
\[
d^*=\sup_{|\Lambda|\le1}\Bigl\{-\tfrac12\Lambda^2-|\Lambda|\Bigr\}=0,
\]
attained at $\Lambda^*=0$. This follows from $\min_b\{\tfrac12 b^2+\Lambda b\}=-\tfrac12\Lambda^2$ and $h^*(\Lambda)=|\Lambda|$ on $[-1,1]$ ($+\infty$ elsewhere).
Hence $p^*-d^*=\tfrac12>0$, and no zero-gap primal--dual pair exists.
\end{example}

The following proposition characterizes the stationarity relation. The proof is provided in \Cref{app:transfer-proof}.
\begin{proposition}\label{prop:transfer}
Let \(h\) be a W-type regularizer, and let $(\tilde{\bx},\tilde{\bB},\tilde{\bLmbd})$ be a KKT point of $\calL$. The following statements are equivalent:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $(\tilde{\bx},\tilde{\bB})$ is a stationary point of Problem~\eqref{Prop: block-wise regularized problem}, i.e.,
$\nabla_{\bx}F(\tilde{\bx},\tilde{\bB})=\bm{0}$ and
$\bm{0}\in\nabla_{\bB}F(\tilde{\bx},\tilde{\bB})+\partial h(\tilde{\bB})$;
\item $h(\tilde{\bB})+h^*(\tilde{\bLmbd})=\langle\tilde{\bLmbd},\tilde{\bB}\rangle$;
\item $|\tilde{b}_{ij}|\ge1$ for all $i,j$.
\end{enumerate}
\end{proposition}
\endgroup

\subsection{From KKT Residuals to Problem~\eqref{Eq: the primal deep hashing problem}}
\label{Sec: original objective guarantees}
\begingroup\color{blue}
Proposition~\ref{prop:transfer} concerns exact KKT points, whereas the algorithms return an approximate KKT point $(\bx,\bB,\bLmbd)$ of $\calL$, whose accuracy is typically measured by the KKT residual
\begin{align}
    \label{Eq: kkt residual}
r_{\mathcal{K}}:=\operatorname{dist}\bigl(\bm{0},\mathcal{K}(\bx,\bB,\bLmbd)\bigr).
\end{align} \blue{Transferring such a point to $J_0$, however, incurs two errors. The optimization error is controlled by $r_{\mathcal{K}}$ and can be driven to zero by the algorithm. The structural error stems from replacing $h$ with $h^{**}$ since the min--max problem~\eqref{Eq: minmax problem} does not penalize non-binary outputs in $[-1,1]^{n\times d}$. In this subsection, we quantify both errors and show that the structural error, which is intrinsic to the model, is controlled by the quantization error~\eqref{eq:quantization-error}, which gives this empirical quantity a theoretical interpretation.}
% Consequently, the stationarity measure of $J_0$ at a point $\widehat{\bx}$ near $\bx$ is bounded by an optimization error, controlled by $r_{\mathcal{K}}$, and a structural error, controlled by $h(\mathcal{D}(\bx))$; for the $\ell_1$ W-regularizer, the latter equals $\lambda Q_1(\bx)$ with $Q_1$ defined in~\eqref{eq:quantization-error}.
% \[
% \operatorname{dist}\bigl(\bm{0},\partial J_0(\widehat{\bx})\bigr)
% \le
% \underbrace{\calO\bigl(r_{\mathcal{K}}\bigr)}_{\text{optimization error}}
% +
% \underbrace{\calO\bigl(\sqrt{h(\mathcal{D}(\bx))}\bigr)}_{\text{structural error}}.
% \]

Since $J_0$ is nonsmooth and nonconvex, our goal for Problem~\eqref{Eq: the primal deep hashing problem} is to find a point satisfying the following nearby-point criterion, related to near-approximate stationarity~\cite[Definition~2.5]{tian2022stationarity}.
\begin{definition}[Near-stationarity]\label{def:near-stationarity}
Given $\delta,\varepsilon\ge0$, a point $\bx$ is called \emph{$(\delta,\varepsilon)$-near-stationary} for Problem~\eqref{Eq: the primal deep hashing problem} if there exists $\widehat{\bx}$ such that
\begin{equation}\label{eq:near-stationarity}
\|\widehat{\bx}-\bx\|\le\delta,
\qquad
\operatorname{dist}\bigl(\bm{0},\partial J_0(\widehat{\bx})\bigr)\le\varepsilon,
\end{equation}
where $\partial$ denotes the limiting subdifferential defined in~\eqref{Eq: limiting subdifferential}.
\end{definition}
To address the nonsmoothness of $J_0$, we introduce a smooth lower bound based on Fenchel duality.
The following lemma bounds $\|\nabla s_{\bLmbd}(\bx)\|$ by the KKT residual and gives a two-sided bound on the gap $J_0(\bx)-s_{\bLmbd}(\bx)$.
\[
s_{\bLmbd}(\by):=f(\by)+\langle\bLmbd,\mathcal{D}(\by)\rangle-h^*(\bLmbd)\le J_0(\by).
\]
\begin{lemma}\label{lem:smooth-lower-bound}
Under Assumption~\ref{Ass: basic regularity}(ii), let $h$ be a W-type regularizer defined in \Cref{def:wtype}, and let $\kappa:=(1+m^2)/(4m)$. Then, for any $(\bx,\bB,\bLmbd)$ satisfying $r_{\mathcal{K}}<\infty$, the following hold:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $s_{\bLmbd}\le J_0$ on $\RR^q$, and $s_{\bLmbd}$ is $L_s(\|\bLmbd\|)$-smooth with $L_s(\|\bLmbd\|):=L_f+\|\bLmbd\|L_{\nabla\mathcal{D}}$;
\item $\|\nabla s_{\bLmbd}(\bx)\|\le\sqrt{1+G_{\mathcal{D}}^2}\;r_{\mathcal{K}}$;
\item the gap at $\bx$ satisfies
\begin{equation}\label{eq:lower-bound-gap}
\begin{split}
J_0(\bx)-s_{\bLmbd}(\bx) \in\bigl[\,h(\mathcal{D}(\bx))-h^{**}(\mathcal{D}(\bx)), h(\mathcal{D}(\bx))+\kappa\,r_{\mathcal{K}}^2\,\bigr].
\end{split}
\end{equation}
For the $\ell_1$ W-regularizer, if $\mathcal{D}(\bx)\in[-1,1]^{n\times d}$, then
\[
J_0(\bx)-s_{\bLmbd}(\bx)\in
\bigl[\,\lambda Q_1(\bx),\ \lambda Q_1(\bx)+\kappa\,r_{\mathcal{K}}^2\,\bigr].
\]
\end{enumerate}
\end{lemma}
The following theorem bounds the near-stationarity of $J_0$ explicitly. The proof is given in \Cref{app:original-certificate-proof}.
\begin{theorem}\label{prop:original-certificate}
Under the conditions of \Cref{lem:smooth-lower-bound},
let $L>0$ satisfy
$L\ge L_s(\|\bLmbd\|)$. Then there exists $\widehat{\bx}\in\RR^q$ such that
\begin{align*}
\|\widehat{\bx}-\bx\|
&\le\sqrt{\frac{2}{L}}
\Bigl(\sqrt{\kappa}\,r_{\mathcal{K}}
+\sqrt{h(\mathcal{D}(\bx))}\Bigr),\\
\operatorname{dist}\bigl(\bm{0},\partial J_0(\widehat{\bx})\bigr)
&\le
\underbrace{
\Bigl(\sqrt{1+G_{\mathcal{D}}^2}
+2\sqrt{2\kappa L}\Bigr)r_{\mathcal{K}}
}_{\text{optimization error}}
+\underbrace{
2\sqrt{2L\,h(\mathcal{D}(\bx))}
}_{\text{structural error}}.
\end{align*}
Moreover, the structural error satisfies
\[
2\sqrt{2L\,h(\mathcal{D}(\bx))}\ 
\begin{cases}
\le 2\sqrt{2Lc\,Q_1(\bx)}, & \text{in general},\\
= 2\sqrt{2L\lambda\,Q_1(\bx)}, & \text{for the $\ell_1$ W-regularizer},
\end{cases}
\]
where $c$ and $\lambda$ are given in \Cref{tab:wtype}. If $h$ has a compact dual domain, then $\|\bLmbd\|\le\lambda\sqrt{nd}$, and one can take $L=L_0:=L_f+\lambda\sqrt{nd}\,L_{\nabla\mathcal{D}}$, independent of $\bLmbd$.
\end{theorem}

{The theorem shows that the quantization error is, to a large extent, \blue{the gap between the min--max problem and the original objective}.} Specifically, the optimization error can be made arbitrarily small by driving $r_{\mathcal{K}}$ to zero, whereas the structural error, determined by the continuous outputs $\mathcal{D}(\bx)$, need not vanish as $r_{\mathcal{K}}\to0$. The structural error is measured by $h(\mathcal{D}(\bx))$, the W-type penalty that many deep hashing methods add to the training loss, and is controlled by the quantization error, with equality for the $\ell_1$ W-regularizer. Hence, driving both $r_{\mathcal{K}}$ and $Q_1(\bx)$ to zero yields near-stationary points of the original problem.

{It is worth noting that the penalty parameter $\gamma$ affects only the optimization error, and can therefore be fixed, without the limit $\gamma\to\infty$ required by penalty methods. A larger regularization parameter $\lambda$ may accelerate binarization during training, but it also increases the structural error through its dependence on $\lambda$ and $L_0$; designing W-type regularizers or output layers that drive $\mathcal{D}(\bx)$ toward $\pm1$ without relying on a large $\lambda$ is left for future work.}
