\section{Complexity Analysis}
\label{Section: Convergence analysis}
We measure their complexity by the number of stochastic first-order oracle (SFO) calls required to find an $\varepsilon$-KKT point of $\calL$. Given $\varepsilon>0$, a random point $(\bx,\bB,\bLmbd)$ is an \emph{$\varepsilon$-KKT point} if
\begin{equation}
\label{equ:esp}
\mathbb{E}\big[\operatorname{dist}^{2}\big(\bm{0},\mathcal{K}(\bx,\bB,\bLmbd)\big)\big]
\leq\varepsilon^2.
\end{equation}
\subsection{\blue{Main Assumption}}
\blue{Under Assumption~\ref{Ass: basic regularity}(ii), $F$ is $L_F$-smooth on $\RR^q\times\{\bB:\|\bB\|\le R_B\}$, and by the initialization and the $\bB$-update this set contains all primal iterates; $L_F$ and $R_B$ are given in Appendix~\ref{app:smoothness-discussion}.}

\blue{Let $T$ denote the maximum number of iterations.} \blue{Under the mini-batch sampling scheme
introduced in Section~\ref{Sec: Algorithm implementations}, let $(\Omega,\mathcal F,\mathbb P)$
be the underlying probability space. For $k\geq1$, define
$\mathcal F_k:=\sigma(\mathcal J_1,\ldots,\mathcal J_{k-1})$ with $\mathcal F_1$ trivial, as
the filtration generated by the past mini-batches. We write
$\mathbb E_k[\cdot]:=\mathbb E[\cdot\mid\mathcal F_k]$ for the conditional expectation
given the past and $\mathbb E[\cdot]$ for the total expectation. All equalities and
inequalities involving $\mathbb E_k[\cdot]$ are understood to hold almost surely.}

In the remainder of this paper, we impose the following assumptions.
\begin{assumption}[\blue{Stochastic first-order oracle}]
\label{Ass: main assumptions}
\leavevmode
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2em,font=\color{blue}]
    \item \blue{The set $\operatorname{crit}\calL$ of KKT points of $\calL$ is nonempty.}
    \item \blue{There exists a constant $\sigma^2\geq0$ such that for all $\bx\in\RR^{q}$ and $\bB\in\RR^{n\times d}$ with $\|\bB\|\le R_B$,}
    \begin{align}
        \label{Eq: finite-sum sampling identities}
        &\frac{1}{n}\sum_{i=1}^{n}\nabla_{\bx} F_i(\bx, \bb_i)
        = \nabla_{\bx} F(\bx, \bB),\\
        &\frac{1}{n}\sum_{i=1}^{n}
        \left\|\nabla_{\bx} F_i(\bx, \bb_i)
        - \nabla_{\bx} F(\bx, \bB)\right\|^2
        \leq \sigma^2.
    \end{align}
\end{enumerate}
\end{assumption}

\begin{remark}
    \blue{The proximal optimality condition in \eqref{eq:proximal-initialization} gives $\bLmbd^1\in\partial h^{**}(\bB^1)$. Together with the definition of $\bLmbd^1$ and subgradient inversion, this yields
\[
\nabla_{\bB}F(\bx^1,\bB^1)+\bLmbd^1=\bm{0},\qquad
\bB^1\in\partial h^*(\bLmbd^1),
\]
so the common one-step estimates apply from the first iteration.}

\end{remark}

\blue{The dual update can increase $\calL$, so $\calL$ need not decrease along the iterates and standard descent arguments do not apply directly} \cite{xu2023momentum,zhang2022stochastic}.
To this end, we construct suitable Lyapunov functions (see \eqref{Eq: the Lyapunov function in SGDM} and \eqref{Eq: the Lyapunov function in STORM}) to obtain the approximate descent property.  We first analyze the dual dynamics by establishing bounds on $\|\bLmbd^{k+1}-\bLmbd^k\|^2$ in terms of the primal residuals, enabling the Lyapunov construction. \blue{The proof is provided in \Cref{app:dual-difference-proof}}.
\begin{lemma}[Dual increment bound]
\label{Lemma: dual boundedness}
Under Assumptions~\ref{Ass: basic regularity} and~\ref{Ass: main assumptions}, for any $k \geq 1$, it holds that
    \begin{align}
        &\|\bm{\Lambda}^{k+1}- \bm{\Lambda}^k \|^2 \notag \\
        \leq&\,3 \tau^{-2} \|\bB^{k+2} - \bB^{k+1}\|^2 + 3(\tau^{-1} + L_F)^2  \|\bB^{k+1} - \bB^{k}\|^2 + 3L_F^2  \|\bx^{k+2} - \bx^{k+1}\|^2.
         \label{Eq: dual boundedness}
\end{align}
\end{lemma}

Next, we analyze how the Lagrangian function \eqref{Eq: simplified_lagrangian} changes
during the updates of primal and dual variables in one iteration. The total change can be decomposed as:
\begin{align}
    \label{Eq: One-iteration progress decomposition} & \calL(\bx^{k+1}, \bB^{k+1}, \bLmbd^{k+1}) - \calL(\bx^{k}, \bB^{k}, \bLmbd^{k}) \notag                                                                                                                                                                                       \\
    =                                                & \,\underbrace{\calL(\bx^{k+1}, \bB^{k+1}, \bLmbd^{k+1}) - \calL(\bx^{k+1}, \bB^k, \bLmbd^k)}_{\text{one-iteration progress on $\bB$ and $\bLmbd$}}+ \underbrace{\calL(\bx^{k+1}, \bB^{k}, \bLmbd^{k}) - \calL(\bx^{k}, \bB^k, \bLmbd^k)}_{\text{one-iteration progress on $\bx$}}.
\end{align}
Since the updates of $\bB$ and $\bLmbd$ are common to Algorithms~\ref{alg:our_algorithm_sgdm} and~\ref{alg:our_algorithm_storm}, the following lemma bounds their one-step change.  For notational simplicity, we define
\[
    \Delta_{\bx}^{k}= \frac{\|\bx^{k}- \bx^{k-1}\|^{2}}{2},\quad \Delta_{\bm{B}}^{k}
    = \frac{\|\bm{B}^{k}- \bm{B}^{k-1}\|^{2}}{2}.
\]  
\begin{lemma}[One-iteration progress on $\bB$ and $\bLmbd$]
    \label{Lemma: One-iteration progress of B and Lambda}
    Consider the sequence $\{
    \bx^{k}, \bB^{k}, \bm{\Lambda}^{k}\}_{k \in \NN}$ generated by \Cref{alg:our_algorithm_sgdm}
    or \Cref{alg:our_algorithm_storm}. Under \Cref{Ass: basic regularity}(ii), for any $k \geq 1$, it holds that
    \begin{align}
\label{item: B and Lambda}
&\calL(\bx^{k+1}, \bB^{k+1}, \bLmbd^{k+1}) - \calL(\bx^{k+1}, \bB^{k}, \bLmbd^{k}) \notag \\
\le& K_1 \Delta_{\bB}^{k+2} - K_2 \Delta_{\bB}^{k+1} + K_3 \Delta_{\bB}^{k} 
+ K_4 \Delta_{\bx}^{k+2} + K_3 \Delta_{\bx}^{k+1}, 
\end{align}{where $K_1=\frac{1}{2\tau}$, $K_2=\frac{3}{2\tau}-2L_F-\frac{\tau L_F^2}{2}$, $K_3=6\tau L_F^2$, and $K_4=\frac{\tau L_F^2}{2}$.}
\end{lemma}
\blue{The proof is provided in \Cref{app:primal-dual-progress-proof}.}

We now analyze the stochastic $\bx$-updates, which \blue{introduce} algorithm-specific challenges:
as \eqref{Eq: dual boundedness} and \blue{\eqref{item: B and Lambda}} show, randomness in the $\bx$-updates propagates to the $\bB$- and $\bLmbd$-updates, necessitating careful parameter tuning and the construction of Lyapunov functions.
\subsection{Oracle Complexity of DualHash-StoM}
\label{Section: complexity of StoM}
Let us denote the gradient error in the $k$-th iteration as
\[
\be^{k} = \blue{\bg_m^{k}} - \nabla_{\bx}F(\bz^{k}, \bB^{k}),
\]where $\blue{\bg_m^k}$ is defined in \eqref{Eq: the stochastic approximate in SGDM}. Under Assumption~\ref{Ass: main assumptions}(ii), one has \blue{
\begin{align}
    \label{eq: the gradient error in SGDM}
    \mathbb{E}\!\left[\|\be^k\|^2\right]
    \leq \frac{\sigma^2}{b_k}.
\end{align}}The one-iteration progress on $\bx$ is estimated as follows, with the proof provided in \Cref{app:stom-primal-progress-proof}.
\begin{lemma}[One-iteration progress on $\bx$]
    \label{Lemma: One-iteration progress of x in SGDM}
    Consider the sequence $\{\bx^k, \bB^k, \bm{\Lambda}^k\}_{k \in \NN}$ generated by \Cref{alg:our_algorithm_sgdm}. Under \Cref{Ass: basic regularity}(ii), \blue{with $\eta_k>0$ and $\alpha_k,\beta_k\in[0,1)$,} for any $k \geq 1$, it holds that
    \begin{align}
        \label{Eq: One-iteration progress of x in SGDM}
        \calL (\bx^{k+1}, \bB^{k}, \bLmbd^{k}) - \calL (\bx^{k}, \bB^{k}, \bLmbd^{k})
        \leq -K_5\Delta_{\bx}^{k+1}+K_6\Delta_{\bx}^{k}+ \frac{1}{L_{F}}\|\be^{k}\|^{2},
    \end{align}
    where \(K_5 = \frac{1 - \alpha_{k}}{\eta_{k}}- 2L_F\) and \(K_6 = \frac{\alpha_{k}}{\eta_{k}}+ 2L_F\beta_k^2\).
\end{lemma}

\blue{To handle the oscillations of the Lagrangian function in \eqref{Eq: simplified_lagrangian}, we construct a Lyapunov function for \Cref{alg:our_algorithm_sgdm}. We choose constant algorithm parameters as follows:
\begin{equation}
    \label{Eq: the parameters conditions of dualhash-sgdm in suppl}
    \eta_{k}= \frac{\eta}{L_{F}},\quad \alpha_{k}= \alpha,\quad \beta_{k}= \beta,\quad 0<\tau<\frac{\sqrt{30}-2}{13L_F},\quad \eta=\frac{(1 - 2\alpha - \nu)/2}{\frac{\bar{L}_{F}}{L_{F}} + \frac{(1 + \nu)\beta^{2}}{(1 - \alpha)\alpha}},
\end{equation}
where $\bar{L}_F:=2L_F+K_3+K_4$,
$0<\nu<1$, $0<\alpha<(1-\nu)/2$, and $0<\beta<1$.
These parameters are independent of $T$.}
\blue{With these constant parameters, we define the Lyapunov function and its associated sequence as follows:}
\begin{align}
    \Psi_m(\mathcal{Q}^{k}) &= \mathcal{L}(\bx^{k}, \bm{B}^{k}, \bLmbd^{k}) 
        - C_{1}\Delta_{\bB}^{k+1} + C_{2}\Delta_{\bB}^{k} 
        - C_{3}\Delta_{\bm{x}}^{k+1} + C_{4}\Delta_{\bm{x}}^{k}, 
        \label{Eq: the Lyapunov function in SGDM} \\
    \mathcal{Q}^{k} &:= \{\bx^{k}, \bB^{k}, \bm{\Lambda}^{k}, 
        \bB^{k+1}, \bB^{k-1}, \bx^{k+1}, \bx^{k-1}\}, \notag
        % \label{Eq: the Lyapunov sequence in SGDM}
\end{align}\blue{where the positive coefficients $C_1,C_2,C_3,C_4$ are independent of $k$ under \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl} and are chosen as follows:}
\begin{align}
    \label{Eq: the coefficients of the Lyapunov function in SGDM}
    \begin{cases}
        C_{1} &= K_{1},\\
        C_{2} &= \frac{1}{2}(K_{2} - K_{1} + K_{3}),\\
        C_{3} &= K_{4},\\
        C_{4} &= \frac{\bar{L}_F + \frac{2L_F \beta^2}{\alpha}}{1 - 2\alpha - \nu}.
    \end{cases}
\end{align}

The construction yields the following in-expectation approximate sufficient
descent property of \blue{$\{\Psi_m(\mathcal{Q}^{k})\}$}.
\begin{lemma}
    \label{Lemma: approximate sufficient descent property in SGDM} Under Assumptions~\ref{Ass: basic regularity} and~\ref{Ass: main assumptions} and the parameter conditions \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl}, for any $k \geq 1$, it holds that
    \begin{align}
        \label{Eq: the descent property of Lyapunov func in SGDM}\EE [\Psi_m(\mathcal{Q}^{k+1})] - \EE[\Psi_m(\mathcal{Q}^{k})] \leq -C_{\bB}\EE[\Delta_{\bB}^{k+1}+ \Delta_{\bB}^{k}] -C_{\bx}\EE [\Delta_{\bm{x}}^{k+1}+ \Delta_{\bm{x}}^{k}] + \frac{\sigma^{2}}{b_{k}L_{F}},
    \end{align}
    where the descent coefficients are given by
    \begin{align}
        \label{Eq: the descent coefficients of potential function in SGDM}\begin{cases}C_{\bB}&= C_{2}- K_{3}= \frac{K_2 - K_3 -K_1}{2}>0, \\ C_{\bx}&= \nu C_{4}>0.\end{cases}
    \end{align}
\end{lemma}
The proof is provided in
\Cref{app:stom-descent-proof}, and the positivity of the coefficients in
\eqref{Eq: the coefficients of the Lyapunov function in SGDM} and the descent
constants in \eqref{Eq: the descent coefficients of potential function in SGDM}
\blue{are} established in \Cref{lem:lyapunov-coef-positivity}.

\blue{Before deriving the convergence rate, we establish a lower bound for the Lyapunov function. Unlike $F$, the potential $\Psi_m$ contains the coupling term $\langle\bB^k,\bLmbd^k\rangle$ and the negative increments $-C_1\Delta_{\bB}^{k+1}$ and $-C_3\Delta_{\bx}^{k+1}$, so it is not trivially bounded below. However, under the parameter conditions \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl}, these terms are dominated by $F-\underline{\ell}$. Consequently, under Assumptions~\ref{Ass: basic regularity} and~\ref{Ass: main assumptions}, there exists $\Psi_m^*>-\infty$, independent of $T$, such that
\[
\EE[\Psi_m(\calQ^k)]\ge\Psi_m^*,\qquad k\ge1.
\]
The proof and the explicit value of $\Psi_m^*$ are given in \Cref{Lemma: lower bound of Psi_m}. For a deterministic initial point, define $\Delta_1:=\calL(\bx^1,\bB^1,\bLmbd^1)-\Psi_m^*$ as an upper bound on the initial potential gap.}

We now present the convergence rate of \Cref{alg:our_algorithm_sgdm} under the parameter conditions \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl}, which guides the selection of the batch size $b_k$. 
\begin{lemma}[Convergence rate of \Cref{alg:our_algorithm_sgdm}]
\label{lemma: convergence rate sgdm} Under the conditions of \Cref{Lemma:
    approximate sufficient descent property in SGDM}, for any $T \geq 1$, there exists
    \blue{$R$ uniformly selected from $\{1,\ldots,T\}$} such that $(\bx^{R}, \bB
    ^{R}, \bLmbd^{R})$ satisfies \eqref{equ:esp} with
    \begin{align}
        \label{Eq: the convergence rate of sgdm}\mathbb{E}[\mathrm{dist}^{2}(\bm{0}, \mathcal K(\bx^{R}, \bB^{R}, \bLmbd^{R}))] \leq \frac{L_{\Delta}\Delta_{1}}{T}+ \blue{\frac{C_{\sigma}\sigma^{2}}{T}\sum_{k=1}^{T}\frac{1}{b_{k}}},
    \end{align}\blue{where $L_{\Delta}:=\frac{\max\{S_1,S_2,S_3,S_4\}}{\min\{C_{\bB},C_{\bx}\}}$ and $C_{\sigma}:=\frac{L_{\Delta}}{L_F}+3$ are constants independent of $T$, with $S_1,\ldots,S_4$ defined in \eqref{Eq: the coefficients of stationarity error bound in SGDM}.}
\end{lemma}

\blue{To achieve the desired oracle complexity, we choose the batch size according to \eqref{Eq: the convergence rate of sgdm} as
\begin{align}
\label{Eq: the parameters conditions of dualhash-sgdm}
    b_k=b=c_bT,
\end{align}
where $c_b>0$ is a constant independent of $T$.} The complexity result is established in \Cref{Theorem: the oracle complexity in SGDM}, with the proof in  \Cref{app:stom-oracle-proof}.
\begin{theorem}
% [Oracle complexity of \Cref{alg:our_algorithm_sgdm}]
    \label{Theorem: the oracle complexity in SGDM} 
Under Assumptions~\ref{Ass: basic regularity} and~\ref{Ass: main assumptions}, let \(\{\bx^k, \bB^k, \bLmbd^k\}_{k \in \NN}\) be the iterate sequence from \Cref{alg:our_algorithm_sgdm} with the parameters satisfying \blue{\eqref{Eq: the parameters conditions of dualhash-sgdm in suppl} and} \eqref{Eq: the parameters conditions of dualhash-sgdm}. \blue{Assume further that $h$ has a compact dual domain.} Then there exists \blue{$R$ uniformly selected from $\{1,\dots,T\}$} such that \( (\bx^R, \bB^R, \bLmbd^R )\) satisfies \eqref{equ:esp} \blue{with}
\begin{align}
    \label{Eq: the iterate number upper bound of algorithm SGDM}
        T = \left\lceil \frac{L_{\Delta} \Delta_1+ C_{\sigma}\frac{\sigma^2}{c_b}}{\varepsilon^2} \right\rceil.
\end{align}
Consequently, the total number of SFO calls is of order \(\calO(\varepsilon^{-4})\).
\end{theorem}
\subsection{Oracle Complexity of DualHash-StoRM} 
Before proceeding, we introduce an additional assumption, also known as the mean-squared smoothness
assumption, widely used in variance reduction methods \cite{arjevani2023lower,driggs2021stochastic}.
\blue{
\begin{assumption}[Mean-square smoothness]
\label{Ass: the mean squared smoothness of F}
    There exists $L_{F} > 0$ such that for all $\bx, \bx' \in \RR^{\blue{q}}$ and $\bB, \bB' \in \RR^{n\times d}$, it holds that
\begin{align}
    \label{eq:mss}
    \frac{1}{n}\sum_{i=1}^{n}
    \left\| \nabla_{\bx} F_{i}(\bx,\bb_i) - \nabla_{\bx} F_{i}(\bx',\bb_i')\right\|^{2}
    \leq L^{2}_F \left[ \|\bx - \bx'\|^{2} + \|\bB - \bB'\|^{2} \right]. 
\end{align}
\end{assumption}}\blue{By Jensen's inequality, \eqref{eq:mss} implies that $\nabla_{\bx}F$ is $L_F$-Lipschitz in $(\bx,\bB)$. The converse need not hold.}

Let us denote the gradient error \[\be^{k} = \blue{\bg_s^{k}} - \nabla_{\bx}F(\bx^{k}, \bB^{k})
,\]where $\blue{\bg_s^k}$ is defined in \eqref{Eq: the stochastic approximate in STORM}. Similarly, the following lemma estimates the one-iteration progress
of the Lagrangian function \eqref{Eq: simplified_lagrangian} via the $\bx$-update \eqref{Eq: the update of x in STORM}. \blue{Its proof follows from \cite[Lemma~2.1]{xu2023momentum} and is given in \Cref{app:storm-primal-progress-proof}.}
\begin{lemma}[One-iteration progress on $\bx$ \cite{xu2023momentum}]
    \label{Lemma: One-iteration progress of x in STORM} Consider the sequence $\{\bx^k, \bB^k, \bm{\Lambda}^k\}_{k \in \NN}$ generated by \Cref{alg:our_algorithm_storm}. Under \Cref{Ass: basic regularity} and \Cref{Ass: main assumptions}(i), for any $k \geq 1$, it holds that
    \begin{align}
        \label{Eq: the recursive relation of L on x}\calL (\bx^{k+1}, \bB^{k}, \bLmbd^{k}) - \calL (\bx^{k}, \bB^{k}, \bLmbd^{k}) \leq \frac{\eta_{k}}{2}\| \be^{k}\|^{2}- (\eta_{k}^{-1}- L_{F})\Delta_{\bx}^{k+1}.
    \end{align}
\end{lemma}

We \blue{then} introduce the following Lyapunov function of \Cref{alg:our_algorithm_storm}:
% and sequence specific to \Cref{alg:our_algorithm_storm}:
\begin{align}
    \Psi_r(\mathcal{Q}^{k}) &= \calL(\bx^{k}, \bm{B}^{k}, \bLmbd^{k}) 
    - C_{1}\Delta_{\bB}^{k+1}+ C_{2}\Delta_{\bB}^{k}- C_{3}\Delta_{\bm{x}}^{k+1},
    \label{Eq: the Lyapunov function in STORM} 
    % \\
    % \{ \calQ^{k}\}_{k \in \NN} &= \{\bx^{k}, \bB^{k}, \bm{\Lambda}^{k}, 
    % \bB^{k+1}, \bB^{k-1}, \bx^{k+1}\}_{k \in \NN},
    % \label{Eq: the Lyapunov sequence in STORM}
\end{align}where $C_1, C_2, C_3$, independent of $k$, are chosen as in 
\eqref{Eq: the coefficients of the Lyapunov function in SGDM}. In the next lemma, we establish an \blue{in-expectation} approximate sufficient descent property 
of $\Psi_r$, with the proof provided in \Cref{app:storm-descent-proof}.
\begin{lemma}
\label{Lemma: approximate sufficient descent property storm}
    Under \Cref{Ass: basic regularity} and \Cref{Ass: main assumptions}(i), with \blue{$0<\tau<\frac{\sqrt{30}-2}{13L_F}$ and $\eta_{k}^{-1}- \tilde{L}_{F}>0$}, for any $k \geq 1$, it holds that
    \blue{\begin{align}
        \EE[\Psi_r(\mathcal{Q}^{k+1})] - \EE[\Psi_r(\mathcal{Q}^{k})]
        &\leq -C_{\bB}\EE[\Delta_{\bB}^{k+1}+\Delta_{\bB}^{k}]-C_{\bx}^{k}\EE[\Delta_{\bx}^{k+1}]+\frac{\eta_k}{2}\EE[\|\be^{k}\|^{2}],
        \label{Eq: the descent property of Lyapunov func in storm}
    \end{align}}
    where the descent coefficients are given by:
    \begin{align}
        % \label{Eq: the descent coefficients of potential function}
        \begin{cases}C_{\bB}&= C_{2}- K_{3}= \frac{K_2 - K_3 -K_1}{2}>0, \\ C_{\bx}^{k}&=\eta_{k}^{-1}- \tilde{L}_{F} > 0, \quad \blue{\tilde{L}_F:=L_F+K_3+K_4}.\end{cases} \notag 
    \end{align}
\end{lemma}
\blue{We now analyze the convergence rate of \Cref{alg:our_algorithm_storm} with the following parameter conditions:}
\blue{\begin{align}
    &\eta_k \equiv\eta_T:= \frac{\eta}{\tilde{L}_{F} T^{1/3}},\quad
    \rho_k \equiv\rho_T:= \frac{8\rho\eta^2}{T^{2/3}},\quad
    b_k = \begin{cases} b_1, & k = 1, \\ b, & k \geq 2, \end{cases}
    \label{Eq: the parameters conditions in dualhash-storm}\\
    &0<\tau<\frac{\sqrt{30}-2}{13L_F},\quad
    \rho\ge\frac{2L_F^{2}(1+G)}{b\tilde{L}_F^{2}},\quad
    0<\eta\le\min\Bigl\{\frac12,\frac{1}{4\sqrt{\rho}}\Bigr\},
    \label{Eq: the constants for storm}
\end{align}
where $\eta$, $\rho$, and $b$ are positive constants independent of $T$, and $G:=1753G_{\mathcal{D}}^{2}$ with $G_{\mathcal{D}}$ given in Assumption~\ref{Ass: basic regularity}(ii).}
\blue{Unlike $F$, the potential $\Psi_r$ contains the coupling term $\langle\bB^k,\bLmbd^k\rangle$ and the negative increments $-C_1\Delta_{\bB}^{k+1}$ and $-C_3\Delta_{\bx}^{k+1}$, and its descent involves the estimator error with the weight $\eta_T/\rho_T$; hence it is not trivially bounded below. However, these parameter conditions give $\eta_T\tilde{L}_F\le\frac12$ and $\rho_T\le\frac12$, so these terms are dominated by $F-\underline{\ell}$ and $\frac{\eta_T}{\rho_T}\|\be^k\|^2$. Consequently, under Assumptions~\ref{Ass: basic regularity} and~\ref{Ass: main assumptions}, there exists $\Psi_r^{*}>-\infty$, independent of $T$, such that
\[
\EE[\Psi_r(\calQ^{k})]+\frac{\eta_T}{\rho_T}\EE\|\be^{k}\|^{2}\ge\Psi_r^{*},\qquad k\ge1.
\]
The proof and the explicit value of $\Psi_r^*$ are given in \Cref{Lemma: lower bound of W}. For a deterministic initial point, define $\tilde{\Delta}_{1}:=\calL(\bx^{1},\bB^{1},\bLmbd^{1})-\Psi_r^{*}$.}
\begin{lemma}[Convergence rate of \Cref{alg:our_algorithm_storm}]
    \label{Lemma: convergence rate of storm} Under \blue{Assumptions~\ref{Ass: basic regularity}, \ref{Ass: main assumptions}, and~\ref{Ass: the mean squared smoothness of F}} and the parameter conditions \eqref{Eq: the parameters conditions in dualhash-storm}--\eqref{Eq: the constants for storm}, for any $T \geq 1$, there exists $R$ uniformly selected from \blue{$\{1,\ldots,T\}$} such that $(\bx^{R}, \bB^{R}, \bLmbd^{R})$ satisfies \eqref{equ:esp} with
    \blue{\begin{align}
    \label{Eq: the convergence rate of storm}\EE[\mathrm{dist}^{2}(\bm{0}, \mathcal K(\bx^{R}, \bB^{R}, \bm{\Lambda}^{R}))] \leq \frac{\tilde{L}_{\Delta}\tilde{\Delta}_{1}}{T^{2/3}}+\frac{\tilde{C}_{\sigma}\sigma^{2}}{4\rho\eta^{2}b_1T^{1/3}}+\frac{32\rho\eta^{2}\tilde{C}_{\sigma}\sigma^{2}}{b\,T^{2/3}},
    \end{align}}
    where
    \blue{\[
    \tilde{L}_{\Delta}:=\max\Bigl\{\frac{S_1}{C_{\bB}},\frac{S_2}{C_{\bB}},\frac{48\tilde{L}_F}{\eta}+\frac{16\eta L_F^{2}(1+3\tau^{2})}{\tilde{L}_F}\Bigr\},\qquad \tilde{C}_{\sigma}:=\frac{\eta\tilde{L}_{\Delta}}{2\tilde{L}_F}
    \]}
    are constants independent of $T$, with $S_1,S_2$ defined in \eqref{Eq: the coefficients of stationarity error bound in SGDM}.
\end{lemma}
\blue{To achieve the desired oracle complexity, we choose $b_1=c_bT^{1/3}$, where $c_b>0$ is a constant independent of $T$.}
\begin{theorem}
\label{Theorem: the oracle complexity in STORM}
Under \blue{Assumptions~\ref{Ass: basic regularity}, \ref{Ass: main assumptions}, and~\ref{Ass: the mean squared smoothness of F}}, let \(\{\bx^k, \bB^k, \bLmbd^k\}_{k \in \NN}\) be the iterate sequence from \Cref{alg:our_algorithm_storm} with the parameters satisfying \eqref{Eq: the parameters conditions in dualhash-storm}--\eqref{Eq: the constants for storm} \blue{and $b_1=c_bT^{1/3}$}. \blue{Assume further that $h$ has a compact dual domain.} Then there exists \blue{$R$ uniformly selected from $\{1,\dots,T\}$} such that \( (\bx^R, \bB^R, \bLmbd^R )\) satisfies \eqref{equ:esp} whenever
\begin{align}
    \label{Eq: the iterate number upper bound of algorithm VR}
        \blue{T = \left\lceil \frac{\Bigl(\tilde{L}_{\Delta}\tilde{\Delta}_{1}+\tilde{C}_{\sigma}\bigl(\frac{\sigma^{2}}{4\rho\eta^{2}c_b}+\frac{32\rho\eta^{2}\sigma^{2}}{b}\bigr)\Bigr)^{3/2}}{\varepsilon^3} \right\rceil,}
\end{align}
with \blue{$\tilde{L}_{\Delta}$ and $\tilde{C}_{\sigma}$ from \Cref{Lemma: convergence rate of storm}}. \blue{Consequently,} the total number of SFO calls is of order \(\calO(\varepsilon^{-3})\).
\end{theorem} 
\begin{remark}
\begin{enumerate}[label=(\roman*), leftmargin=2em]
\item The results in Theorems \ref{Theorem: the oracle complexity in SGDM} and \ref{Theorem: the oracle complexity in STORM} are novel for stochastic, non-convex, multi-block primal-dual optimization, a challenging area where existing work has largely focused on one-block \cite{xu2023momentum,shi2025momentum} or deterministic settings \cite{ADMMNolinear2024}.  
    \item  \Cref{alg:our_algorithm_sgdm} \blue{under the parameter conditions \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl} and} \eqref{Eq: the parameters conditions of dualhash-sgdm} attains both oracle and sample \(\mathcal{O}(\varepsilon^{-4})\) complexities, which matches that of stochastic first-order methods (SFOMs) for one-block stochastic optimization \cite{Ghadimi2013MinibatchSA}. Meanwhile, \Cref{alg:our_algorithm_storm} with the refined constant parameters from \eqref{Eq: the parameters conditions in dualhash-storm} attains both oracle and sample $\mathcal{O}(\varepsilon^{-3})$ complexities. This matches the best-known lower bound \cite{arjevani2023lower} and the complexity of variance-reduced SFOMs for single-block problems \cite{shi2025momentum,xu2023momentum}. Notably, despite tackling a multi-block problem, our primal-dual framework preserves this efficiency without degrading the dependence on $\varepsilon$.
\end{enumerate}
\end{remark}

\subsection{Guarantees for the Original Objective}
\label{Sec: guarantees original objective}
\Cref{Theorem: the oracle complexity in SGDM,Theorem: the oracle complexity in STORM} control the KKT residual of the conjugate model. \Cref{prop:original-certificate} transfers this control to the original objective.

\begin{corollary}[Near-stationarity of the original objective]\label{cor:original-stationarity}
Under the assumptions of \Cref{Theorem: the oracle complexity in SGDM} (resp.\ \Cref{Theorem: the oracle complexity in STORM}), let $T$ be chosen as in that theorem, so that the expected squared KKT residual satisfies \eqref{equ:esp}. Then, almost surely, $\bx^R$ is $(\delta_R,\varepsilon_R)$-near-stationary for $J_0$, with tolerances from \Cref{prop:original-certificate}(b) satisfying
\begin{align*}
\EE\,\varepsilon_R &\le\Bigl(\sqrt{1+G_{\mathcal{D}}^2}+2\sqrt{2\kappa L_0}\Bigr)\varepsilon+2\sqrt{2L_0\,\EE\,h\bigl(\mathcal{D}(\bx^R)\bigr)},\\
\EE\,\delta_R &\le\sqrt{\frac{2}{L_0}}\Bigl(\sqrt{\kappa}\,\varepsilon+\sqrt{\EE\,h\bigl(\mathcal{D}(\bx^R)\bigr)}\Bigr),
\end{align*}
after $\calO(\varepsilon^{-4})$ (resp.\ $\calO(\varepsilon^{-3})$) SFO calls. Here $\kappa=(1+m^2)/(4m)$, $m=\gamma/n$, and $L_0=L_f+\lambda\sqrt{nd}\,L_{\nabla\mathcal{D}}$. For the row $p=1$ of \Cref{tab:wtype}, $h(\mathcal{D}(\bx^R))=\lambda Q_1(\bx^R)$.
\end{corollary}
\begin{proof}
Apply \Cref{prop:original-certificate}(b) at $(\bx^R,\bB^R,\bLmbd^R)$, split the square roots by $\sqrt{a+b}\le\sqrt{a}+\sqrt{b}$, and take expectations using Jensen's inequality.
\end{proof}

\begin{remark}[Guidance on the code length]\label{rem:code-length}
For the row $p=1$ of \Cref{tab:wtype}, a sufficient condition for the structural term in \Cref{cor:original-stationarity} to be at most $\varepsilon/2$ is $\lambda\,\EE\,Q_1(\bx^R)\le\varepsilon^2/(32L_0)$. Under this condition, the same SFO complexity orders yield $\EE\,\varepsilon_R=\calO(\varepsilon)$ and $\EE\,\delta_R=\calO(\varepsilon)$. This sufficient quantization-error threshold is quadratic in the target accuracy. In terms of the average per-bit quantization error $Q_1/(nd)$, which is the quantity reported in \Cref{Section: Numerical experiments}, this condition reads
\[
\frac{\EE\,Q_1(\bx^R)}{nd}\le\frac{\varepsilon^2}{32\,\lambda\,nd\,\bigl(L_f+\lambda\sqrt{nd}\,L_{\nabla\mathcal{D}}\bigr)}.
\]
For fixed $n$, $\lambda$, and smoothness constants, this sufficient per-bit threshold decreases at least as $d^{-1}$ in the code length $d$, and as $d^{-3/2}$ when $\lambda\sqrt{nd}\,L_{\nabla\mathcal{D}}$ dominates $L_f$. Thus, certifying the same accuracy for the original objective requires a smaller average per-bit error as the code length increases. This is consistent with the association between low quantization error and high mAP observed at 64 bits in \Cref{Section: Numerical experiments}.
\end{remark}
