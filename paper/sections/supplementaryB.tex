\begin{proof}
    According to \cite[Theorem 2.158]{temporary-citekey-11039},
    $(\bx^{*}, \bB^{*}, \bLmbd^{*})$ are optimal solutions to the
    primal problem \eqref{Prop: block-wise regularized problem} and the dual problem
    \eqref{Prop: the dual problem} \blue{with zero duality gap} if and only if:
    \begin{align}
        \label{Eq: the sufficient and necessary condition}
        \begin{cases}
        (\bx^{*}, \bB^{*}) \in \argmin\limits_{\bx \in \RR^{q}, \bB \in \RR^{n\times d}}\calL(\blue{\bx, \bB}, \bLmbd^{*}), \\
        0 = h(\bB^{*}) + h^{*}(\bLmbd^{*}) - \langle \bLmbd^{*}, \bB^{*}\rangle.
        \end{cases}
    \end{align}
    The first condition in \eqref{Eq: the sufficient and necessary condition} implies
    that $\bm{0}$ belongs to the subdifferential of $\calL$ with respect to
    $(\bx, \bB)$ at $(\bx^{*}, \bB^{*}, \bLmbd^{*})$. Since $\calL$
    is continuously differentiable in $(\bx, \bB)$, we have
    \begin{align}
        \label{Eq: the sufficient and necessary condition for primal}
        \begin{cases}
        \bm{0} = \blue{\nabla_{\bx}F(\bx^{*}, \bB^{*})},\\
        \bm{0} = \blue{\nabla_{\bB}F(\bx^{*}, \bB^{*})} + \bLmbd^{*}.
        \end{cases}
    \end{align}
    For the second condition in \eqref{Eq: the sufficient and necessary condition},
    \blue{since $h^{**}\le h$, the Fenchel--Young inequality for $h^{**}$ yields
    $\langle\bLmbd^*,\bB^*\rangle\le h^{**}(\bB^*)+h^*(\bLmbd^*)\le h(\bB^*)+h^*(\bLmbd^*)=\langle\bLmbd^*,\bB^*\rangle$.
    Thus $h^{**}(\bB^*)+h^*(\bLmbd^*)=\langle\bLmbd^*,\bB^*\rangle$, which, by the subgradient inversion rule~\cite[Theorem~23.5]{rockafellar1970convex}, gives}
    \begin{align}
        \label{Eq: the sufficient and necessary condition for dual}\bB^{*}\in \partial h^{*}(\bLmbd^{*}).
    \end{align}
    Combining \eqref{Eq: the sufficient and necessary condition for primal} and \eqref{Eq: the sufficient and necessary condition for dual}, we obtain \blue{$\bm{0}\in\mathcal{K}(\bx^{*}, \bB^{*}, \bLmbd^{*})$ with $\mathcal{K}$ in \eqref{Eq: KKT mapping}}, completing the proof.
\end{proof}

\subsection{\blue{Proof of \Cref{prop:transfer}}}
\label{Subsec: proof transfer}
\begingroup\color{blue}
\begin{proof}
    Since $h^{*}$ is proper, lower semicontinuous, and convex, its conjugate is $h^{**}(\bB)=\lambda\sum_{i,j}(|b_{ij}|-1)_{+}$, and the subgradient inversion rule~\cite[Theorem~23.5]{rockafellar1970convex} gives
    \begin{align}
        \label{Eq: inversion rule of the conjugate}
        \tilde{\bB}\in\partial h^{*}(\tilde{\bLmbd})
        \iff \tilde{\bLmbd}\in\partial h^{**}(\tilde{\bB})
        \iff h^{**}(\tilde{\bB})+h^{*}(\tilde{\bLmbd})=\langle\tilde{\bLmbd},\tilde{\bB}\rangle.
    \end{align}
    Since $(\tilde{\bx},\tilde{\bB},\tilde{\bLmbd})$ satisfies \eqref{Eq: KKT conditions}, all three statements in \eqref{Eq: inversion rule of the conjugate} hold, and $\tilde{\bLmbd}=-\nabla_{\bB}F(\tilde{\bx},\tilde{\bB})$ together with $\nabla_{\bx}F(\tilde{\bx},\tilde{\bB})=\bm{0}$ shows that (a) holds if and only if $\tilde{\bLmbd}\in\partial h(\tilde{\bB})$.

    (b)$\iff$(c). By the last equality in \eqref{Eq: inversion rule of the conjugate}, (b) holds if and only if $h(\tilde{\bB})=h^{**}(\tilde{\bB})$. Since $h$ and $h^{**}$ are separable and $\lambda\big||b|-1\big|=\lambda(|b|-1)_{+}$ if and only if $|b|\ge1$, this is equivalent to $|\tilde{b}_{ij}|\ge1$ for all $i,j$.

    (b)$\Rightarrow$(a). By the definition of $h^{*}$, (b) means that $\tilde{\bB}$ is a global minimizer of $h(\cdot)-\langle\tilde{\bLmbd},\cdot\rangle$. Fermat's rule~\cite[Theorem~10.1]{Rockafellar1998VariationalA} and \cite[Exercise~8.8(c)]{Rockafellar1998VariationalA} then give $\bm{0}\in\partial h(\tilde{\bB})-\tilde{\bLmbd}$, i.e., $\tilde{\bLmbd}\in\partial h(\tilde{\bB})$, which is (a).

    (a)$\Rightarrow$(c). By \eqref{Eq: inversion rule of the conjugate} and (a), $\tilde{\bLmbd}\in\partial h(\tilde{\bB})\cap\partial h^{**}(\tilde{\bB})$. Since $h$ and $h^{**}$ are separable~\cite[Proposition~10.5]{Rockafellar1998VariationalA}, it suffices to consider a scalar entry $b=\tilde{b}_{ij}$. If $|b|<1$, then $\partial h^{**}(b)=\{0\}$, whereas $\partial h(b)=\{-\lambda\operatorname{sgn}(b)\}$ for $b\neq0$ and $\partial h(0)=\{-\lambda,\lambda\}$; hence $0\notin\partial h(b)$ and the intersection is empty. Therefore $|\tilde{b}_{ij}|\ge1$ for all $i,j$.
\end{proof}

    _{\mathcal{D}B}.

    L_{s}= \frac{\gamma}{n}\sqrt{2\max\{L^{2}_{\nabla \mathcal{D}}C^{2}_{\mathcal{D}B}+

\subsubsection{Lower bounds for the Lyapunov functions}
\label{app-toc-3}
\label{sec:lowerbounds}
Throughout this section, let $m:=\gamma/n$, $M:=\lambda\sqrt{nd}$, and
$\mathcal{D}(\bx):=(\bD_1(\bx),\ldots,\bD_n(\bx))$. We use the convention
$\bx^0:=\bx^1$ and write $\bm{r}_k:=\bB^k-\mathcal{D}(\bx^k)$,
$\bm{p}_k:=\bx^k-\bx^{k-1}$, and $\bm{u}_k:=\bx^{k+1}-\bx^k$.
The output condition in Appendix~\ref{Appendix: discussion on assumptions} gives
$\|\mathcal{D}(\bx)\|_F\le C_{\mathcal{D}}$, and $\mathcal{D}$ is
$L_{\mathcal{D}}$-Lipschitz. The constants $C_1,\ldots,C_4$ are those in the
Lyapunov functions $\Psi_m$ and $\Psi_r$.

\begin{proposition}
\label{prop:lyapunov-lower-bound}
Let \Cref{Ass: main assumptions} hold and let
$\bLmbd^1\in[-\lambda,\lambda]^{n\times d}$, as required by the initialization
of both algorithms. Define
\[
c_0:=\frac{4M^2}{m}+MC_{\mathcal{D}}+\lambda nd,
\qquad
A_B:=\frac{3}{2}C_1\tau^2m^2,
\qquad
H_m:=\frac{3}{2}C_1\tau^2m^2L_{\mathcal{D}}^2+\frac{1}{2}C_3.
\]
\begin{enumerate}[label=\textup{(\roman*)},nosep]
\item \textup{(DualHash-StoM)} Let $\eta_k\equiv\eta/L_F$, $\alpha_k\equiv\alpha$,
and $\beta_k\equiv\beta$, and put
\[
\Theta:=\frac{12H_m\eta^2}{L_F},
\qquad Q:=3\alpha^2+6\eta^2\beta^2.
\]
If
\begin{align}
\frac{m}{2}(1-\Theta)-\frac{m}{16}-A_B&\ge\frac{m}{8},
\tag{P1}\label{eq:P1}\\
C_4&\ge2H_mQ,
\tag{P2}\label{eq:P2}
\end{align}
then, for every $k\ge1$,
\[
\mathbb{E}[\Psi_m(\calQ^k)]
\ge \underline{\ell}+\frac{\mu}{8}\mathbb{E}\|\bx^k\|^2
+\frac{m}{8}\mathbb{E}\|\bm{r}_k\|_F^2-c_1
\ge \Psi_m^*:=\underline{\ell}-c_1,
\]
where
\[
c_1:=c_0+\frac{3}{2}C_1\tau^2M^2+\frac{3H_m\eta^2\sigma^2}{L_F^2}
\]
is independent of $T$.

\item \textup{(DualHash-StoRM)} Let $\eta_k\equiv\eta_T$, let
$\be^k:=\bg_s^k-\nabla_{\bx}F(\bx^k,\bB^k)$, and put
\[
\Gamma:=2H_m\eta_T^2,
\qquad
W_k:=\mathbb{E}[\Psi_r(\calQ^k)]+\frac{\eta_T}{\rho_T}\mathbb{E}\|\be^k\|^2.
\]
If
\begin{align}
\Gamma&\le\frac{\eta_T}{\rho_T},
\tag{P1$'$}\label{eq:P1s}\\
\frac{m}{2}(1-2L_F\Gamma)-\frac{m}{16}-A_B&\ge\frac{m}{8},
\tag{P2$'$}\label{eq:P2s}
\end{align}
then, for every $k\ge1$,
\[
W_k\ge\underline{\ell}+\frac{\mu}{8}\mathbb{E}\|\bx^k\|^2
+\frac{m}{8}\mathbb{E}\|\bm{r}_k\|_F^2-c_1'
\ge\Psi_r^*:=\underline{\ell}-c_1',
\qquad
c_1':=c_0+\frac{3}{2}C_1\tau^2M^2.
\]
\end{enumerate}
\end{proposition}

\begin{proof}
Since $f_i=\ell_{s,i}+\frac{\mu}{2}\|\cdot\|^2$ and
$P(\bx,\bB)=\frac{m}{2}\|\bB-\mathcal{D}(\bx)\|_F^2$, Assumption~\ref{Ass: main assumptions}(i) gives
\begin{equation}
\label{eq:F-lower}
F(\bx,\bB)-\underline{\ell}
\ge\frac{\mu}{2}\|\bx\|^2+\frac{m}{2}\|\bB-\mathcal{D}(\bx)\|_F^2\ge0.
\end{equation}
Because $F$ is $L_F$-smooth and bounded below by $\underline{\ell}$,
\[
\|\nabla F(\bx,\bB)\|^2\le2L_F\bigl(F(\bx,\bB)-\underline{\ell}\bigr).
\]
Moreover, $\operatorname{dom}h^*=[-\lambda,\lambda]^{n\times d}$ and
$0\le h^*(\bLmbd)\le\lambda nd$ on this box. Since $\bLmbd^1$ lies in this box and the closed-form proximal mapping \eqref{Eq: the proximal mapping of conjugate function} maps into it, $\|\bLmbd^k\|_F\le M$ for all $k\ge1$.

Since $\|\bB^k\|_F\le\|\bm r_k\|_F+C_{\mathcal{D}}$, Young's inequality gives
\[
\langle\bB^k,\bLmbd^k\rangle
\ge-\frac{m}{16}\|\bm r_k\|_F^2-\frac{4M^2}{m}-MC_{\mathcal{D}}.
\]
Consequently,
\begin{equation}
\label{eq:L-lower}
\calL(\bx^k,\bB^k,\bLmbd^k)
\ge F(\bx^k,\bB^k)-\frac{m}{16}\|\bm r_k\|_F^2-c_0.
\end{equation}

The $\bB$-update and
$\nabla_{\bB}F(\bx,\bB)=m(\bB-\mathcal{D}(\bx))$ imply
\[
\Delta_{\bB}^{k+1}
\le\frac{3\tau^2}{2}\left(m^2\|\bm r_k\|_F^2
 +m^2L_{\mathcal{D}}^2\|\bm u_k\|^2+M^2\right),
\]
and hence
\begin{equation}
\label{eq:B-inc}
-C_1\Delta_{\bB}^{k+1}-C_3\Delta_{\bx}^{k+1}
\ge-A_B\|\bm r_k\|_F^2-H_m\|\bm u_k\|^2
-\frac{3}{2}C_1\tau^2M^2.
\end{equation}

For DualHash-StoM, the $\bx$-update and the gradient-error estimate give
\[
\mathbb{E}\|\bm u_k\|^2
\le12\eta_k^2L_F\,\mathbb{E}[F(\bx^k,\bB^k)-\underline{\ell}]
 +Q\,\mathbb{E}\|\bm p_k\|^2+3\eta_k^2\sigma^2.
\]
Combining this estimate with \eqref{eq:F-lower}--\eqref{eq:B-inc}, and using
\eqref{eq:P1}--\eqref{eq:P2}, yields
\[
\mathbb{E}[\Psi_m(\calQ^k)]
\ge\underline{\ell}+\frac{\mu}{8}\mathbb{E}\|\bx^k\|^2
 +\frac{m}{8}\mathbb{E}\|\bm r_k\|_F^2-c_1.
\]

For DualHash-StoRM, write
$\bm u_k=-\eta_T(\nabla_{\bx}F(\bx^k,\bB^k)+\be^k)$. Then
\[
\mathbb{E}\|\bm u_k\|^2
\le2\eta_T^2\left(\mathbb{E}\|\nabla_{\bx}F(\bx^k,\bB^k)\|^2+\mathbb{E}\|\be^k\|^2\right).
\]
Using \eqref{eq:L-lower}--\eqref{eq:B-inc}, the smoothness bound, and
\eqref{eq:P1s}--\eqref{eq:P2s}, gives
\[
W_k\ge\underline{\ell}+\frac{\mu}{8}\mathbb{E}\|\bx^k\|^2
 +\frac{m}{8}\mathbb{E}\|\bm r_k\|_F^2-c_1'.
\]
This proves both lower bounds.
\end{proof}

\begin{proof}

    \begin{align*}
    \bLmbd^{k+1}-\bLmbd^k
    ={}&\tau^{-1}(\bB^{k+1}-\bB^{k+2})
    +\tau^{-1}(\bB^{k+1}-\bB^k)\\
    &+\nabla_{\bB}F(\bx^{k+1},\bB^k)
    -\nabla_{\bB}F(\bx^{k+2},\bB^{k+1}).
    \end{align*}
    By the $L_F$-smoothness of $F$ in \Cref{Ass: main assumptions},
    \begin{align*}
    \|\bLmbd^{k+1}-\bLmbd^k\|
    \leq{}&\tau^{-1}\|\bB^{k+2}-\bB^{k+1}\|
    +(\tau^{-1}+L_F)\|\bB^{k+1}-\bB^k\|\\
    &+L_F\|\bx^{k+2}-\bx^{k+1}\|.
    \end{align*}
    Squaring this inequality and using $(a+b+c)^2\leq3(a^2+b^2+c^2)$ proves
    \begin{align}
    \|\bLmbd^{k+1}-\bLmbd^k\|^2
    \leq{}&3\tau^{-2}\|\bB^{k+2}-\bB^{k+1}\|^2
    +3(\tau^{-1}+L_F)^2\|\bB^{k+1}-\bB^k\|^2\notag\\
    &+3L_F^2\|\bx^{k+2}-\bx^{k+1}\|^2,
    \label{Eq: dual boundedness in proof}
    \end{align}
    which is the claimed result.
\end{proof}

    \begin{align}
        \label{Eq: the inner product in the recursive relation of L on B and Lambda} & \langle \bLmbd^{k+1}- \bLmbd^{k}, \tau (\nabla_{\bB}F(\bx^{k}, \bB^{k-1}) - \nabla_{\bB}F(\bx^{k+1}, \bB^{k})) \rangle \notag \\

\begin{lemma}
    \label{Lemma: auxiliary lemma for f} Let $f(\bx): \mathbb{R}^{q}\rightarrow \mathbb{R}$
    be a continuously differentiable function with an $L$-Lipschitz continuous gradient.
    Then, for any $\bx,\, \by,\,\blue{\bg} \in \mathbb{R}^{q}$ and any $\bx^{+}\in \mathbb{R}
    ^{q}$ defined by
    \[
        \bx^{+}= \by - \eta \blue{\bg}, \hspace{5pt}\eta > 0,
    \]it holds that
    \begin{align}
        \label{Eq: auxiliary lemma for f}f(\bx^{+}) \leq f(\bx) + \langle \nabla f(\bx) - \blue{\bg}, \bx^{+}- \bx \rangle + \frac{L}{2}\|\bx^{+}- \bx\|^{2}+ \frac{1}{2\eta}\|\bx - \by\|^{2}- \frac{1}{2\eta}\|\bx^{+}- \by\|^{2}.
    \end{align}
\end{lemma}
\begin{proof}
    By the definition of $\bx^{+}$, we obtain that
    \[
        \bx^{+}= \argmin\limits_{\bu \in \mathbb{R}^{q}}\{ \langle \blue{\bg}, \bu
        - \by \rangle + \frac{1}{2\eta}\| \bu - \by\|^{2}\},
    \]
    and hence, in particular, by taking $\bu = \bx$, we obtain
    \begin{align*}
        0 \leq \langle \blue{\bg}, \bx - \bx^{+}\rangle + \frac{1}{2\eta}\| \bx - \by\|^{2}- \frac{1}{2\eta}\| \bx^{+}- \by\|^{2}.
    \end{align*}
    Invoking the descent lemma for a smooth function yields the result. 
\end{proof}

\begin{lemma}[Recursive bound on gradient error]

        \EE[\| \be^{k+1}\|^{2}] \leq \left(1 - \rho_{k}\right)^{2}\EE[\| \be^{k}\|^{2}] + \frac{2 \rho_{k}^{2}\sigma^{2}}{b_{k+1}}+ \frac{4(1 - \rho_{k})^{2}L_{F}^{2}}{b_{k+1}}\EE\left[\Delta_{\bx}^{k+1}+\blue{\Delta_{\bB}^{k+1}}\right]. \notag
    \end{align}}

\begin{proof}
    \blue{For $k\geq1$, recall that $\mathcal F_{k+1}=\sigma(\calJ_1,\ldots,\calJ_k)$. Then $\bx^{k},\bB^{k},\bx^{k+1},\bB^{k+1}$ and $\be^{k}$ are $\mathcal F_{k+1}$-measurable, while $\calJ_{k+1}$ is sampled uniformly without replacement and independently of $\mathcal F_{k+1}$.     \blue{\begin{align}
        \EE_{k+1}[\|\be^{k+1}\|^{2}]
        = & \,\EE_{k+1}[\|\bg_s^{k+1}-\nabla^{k+1}\|^{2}]\notag\\
        = & \,\EE_{k+1}[\|\tilde\nabla^{k+1}-\nabla^{k+1}+(1-\rho_k)(\bg_s^{k}-\tilde\nabla^{k})\|^{2}]\notag\\
        = & \,\EE_{k+1}[\|(1-\rho_k)\be^{k}+\tilde\nabla^{k+1}-\nabla^{k+1}+(1-\rho_k)(\nabla^{k}-\tilde\nabla^{k})\|^{2}]\notag\\
        = & \,(1-\rho_k)^{2}\|\be^{k}\|^{2}+\EE_{k+1}[\|\tilde\nabla^{k+1}-\nabla^{k+1}+(1-\rho_k)(\nabla^{k}-\tilde\nabla^{k})\|^{2}]\quad\text{a.s.},
        \label{Eq: the error recursion first step in StoRM}
    \end{align}}
    \blue{where \eqref{Eq: the error recursion first step in StoRM} follows from \Cref{Ass: main assumptions}(iii), which ensures that $\EE_{k+1}[\tilde\nabla^{j}]=\frac1n\sum_{i=1}^{n}\nabla_{\bx}F_i(\bx^{j},\bb_i^{j})=\nabla^{j}$ a.s. for $j\in\{k,k+1\}$, and hence
    \begin{align*}
        \EE_{k+1}[\langle\tilde\nabla^{k+1}-\nabla^{k+1},\be^{k}\rangle] = & \,0,\\

\begin{lemma}[Tracking of the $\bB$-increments]
    \label{Lemma: tracking of B increments}
    \blue{Let $\mathcal{D}$ be $L_{\mathcal{D}}$-Lipschitz, $0<r\le\frac27$, and let \Cref{alg:our_algorithm_storm} be initialized as in \Cref{Sec: Algorithm implementations}. Then $H_k\ge0$, $H_1=0$, and for every $k\ge1$,
    \[
        \Delta_{\bB}^{k+1}\ \le\ \frac{12}{r}\bigl(H_k-H_{k+1}\bigr)+G\,\Delta_{\bx}^{k+1},\qquad G:=1753L_{\mathcal{D}}^{2}.
    \]}
\end{lemma}
\begin{proof}
    \blue{\emph{Step 1: update identities.} The $\bB$-update gives $\bB^{k+1}=(1-r)\bB^{k}-\bm{Z}^{k}+r\bm{U}^{k+1}$. By the proximal form of the dual update, $\bm{Z}^{k+1}=P(\bm{Z}^{k}+2\bB^{k+1}-\bB^{k})$ with $P(\bm v):=\tau\operatorname{prox}_{\tau^{-1}h^{*}}(\bm v/\tau)$, which acts entrywise as a nondecreasing $1$-Lipschitz scalar function because $h^*$ is convex and separable. Both identities also hold at the index $0$: the initialization gives $\bm{Z}^{1}=r(\bm{U}^{1}-\bB^{1})$, which is the first identity with $(\bB^0,\bm Z^0)=(\bB^1,\bm Z^1)$; and $\bLmbd^{1}\in\partial h^{**}(\bB^{1})$ is equivalent to $\bB^{1}\in\partial h^{*}(\bLmbd^{1})$, i.e., $\bm{Z}^{1}=P(\bm{Z}^{1}+\bB^{1})$, which is the second identity. In particular $\bm\beta_1=\bm\zeta_1=\bm w_1=\bm 0$, so $H_1=0$.

    \emph{Step 2: increment dynamics.} Let $\bm u_k:=\bm U^{k+1}-\bm U^{k}$. Subtracting consecutive identities gives $\bm\beta_{k+1}=(1-r)\bm\beta_k-\bm\zeta_k+r\bm u_k=-r\bm\beta_k-\bm w_k+r\bm u_k$. For each entry, let $d\in[0,1]$ be the slope of $P$ between its two consecutive arguments; then $\zeta_{k+1}=d(\zeta_k+2\beta_{k+1}-\beta_k)$ entrywise, and hence $w_{k+1}=\zeta_{k+1}-\beta_{k+1}=r(1-2d)\beta_k+(1-d)w_k+r(2d-1)u_k$. With $y_k:=(\beta_k,w_k)^{\top}$ (one entry),
    \[
        y_{k+1}=A_dy_k+f_du_k,\qquad A_d=\begin{pmatrix}-r&-1\\ r(1-2d)&1-d\end{pmatrix},\quad f_d=\begin{pmatrix}r\\ r(2d-1)\end{pmatrix}.
    \]
    The slope $d$ may depend on the entry and on $k$.

    \emph{Step 3: a common quadratic form.} Let $\Pi_r:=\begin{pmatrix}r&r\\ r&1\end{pmatrix}\succ0$, so that $H_k=\sum_{\text{entries}}y_k^{\top}\Pi_ry_k\ge0$. Direct computation gives
    \[
        \Pi_r-A_0^{\top}\Pi_rA_0=\begin{pmatrix}r-r^2+r^3&r^2\\ r^2&r\end{pmatrix},\qquad
        \Pi_r-A_1^{\top}\Pi_rA_1=\begin{pmatrix}r-r^2-3r^3&r-2r^2\\ r-2r^2&1-r\end{pmatrix}.
    \]
    For $0<r\le\frac27$, the first matrix minus $\frac r2I$ has positive leading entry and determinant $r^2(\frac14-\frac r2-\frac{r^2}2)\ge\frac{13}{196}r^2>0$, so it is $\succeq\frac r2I$. The second matrix minus $\frac r4I$ has leading entry $r(\frac34-r-3r^2)\ge\frac{43}{196}r>0$ and determinant $r\,p(r)$ with $p(r)=\frac34-\frac{47}{16}r+\frac94r^2-\frac14r^3$, which is decreasing on $(0,\frac27]$ with $p(\frac27)=\frac{243}{2744}>0$, so it is $\succeq\frac r4I$. Since $A_d=(1-d)A_0+dA_1$ and $\|\cdot\|_{\Pi_r}^2$ is convex, $\|A_dy\|_{\Pi_r}^2\le(1-d)\|A_0y\|_{\Pi_r}^2+d\|A_1y\|_{\Pi_r}^2\le\|y\|_{\Pi_r}^2-\frac r4\|y\|^2$ for all $d\in[0,1]$. Moreover $\|\Pi_r\|\le1+r\le\frac97$, $\|A_d\|\le\|A_d\|_F\le\sqrt{2+2r^2}\le\frac32$, and $\|f_d\|\le\sqrt2\,r$, so $\|A_d^{\top}\Pi_rf_d\|\le3r$ and $f_d^{\top}\Pi_rf_d\le\frac{18}{7}r^2\le r$. Therefore, using $6r\|y\||u|\le\frac r8\|y\|^2+72ru^2$,
    \[
        y_{k+1}^{\top}\Pi_ry_{k+1}\le y_k^{\top}\Pi_ry_k-\tfrac r4\|y_k\|^2+6r\|y_k\||u_k|+ru_k^2\le y_k^{\top}\Pi_ry_k-\tfrac r8\|y_k\|^2+73ru_k^2 .
    \]
    Summing over the entries,
    \begin{equation}\label{Eq: the common quadratic decrease}
        H_{k+1}-H_k\ \le\ -\tfrac r8\bigl(\|\bm\beta_k\|_F^2+\|\bm w_k\|_F^2\bigr)+73r\|\bm u_k\|_F^2 .
    \end{equation}

    \emph{Step 4.} From Step 2, $\|\bm\beta_{k+1}\|_F^2\le3(\|\bm\beta_k\|_F^2+\|\bm w_k\|_F^2)+3r^2\|\bm u_k\|_F^2$. Inserting $\|\bm\beta_k\|_F^2+\|\bm w_k\|_F^2\le\frac8r(H_k-H_{k+1})+584\|\bm u_k\|_F^2$ from \eqref{Eq: the common quadratic decrease} and using $\|\bm u_k\|_F^2\le L_{\mathcal{D}}^2\|\bx^{k+1}-\bx^{k}\|^2=2L_{\mathcal{D}}^2\Delta_{\bx}^{k+1}$ and $3r^2<1$ yields $\Delta_{\bB}^{k+1}=\frac12\|\bm\beta_{k+1}\|_F^2\le\frac{12}{r}(H_k-H_{k+1})+1753L_{\mathcal{D}}^2\Delta_{\bx}^{k+1}$.}
\end{proof}

\begin{lemma}[Descent of the augmented potential]
    \label{Lemma: Weighted error accumulation} \blue{Under \Cref{Ass: main assumptions}(iii), \Cref{Ass: the mean squared smoothness of F}, the conditions of \Cref{Lemma: tracking of B increments}, and the parameter conditions \eqref{Eq: the parameters assumptions in storm appendix}, let $q_k:=\EE\|\be^{k}\|^2$, $d_0:=48L_F^2/(br)$, and
    \[
        \widehat{W}_k:=\EE[\Psi_r(\calQ^{k})]+\frac{\eta_T}{\rho_T}\bigl(q_k+d_0\,\EE H_k\bigr).
    \]
    Then for every $k\ge1$,
    \begin{align}
        \widehat{W}_{k+1}\ \le\ \widehat{W}_k-C_{\bB}\,\EE\bigl[\Delta_{\bB}^{k+1}+\Delta_{\bB}^{k}\bigr]-\frac{1}{4\eta_T}\EE[\Delta_{\bx}^{k+1}]-\frac{\eta_T}{2}\,q_k+\frac{2\eta_T\rho_T\sigma^{2}}{b}.
        \label{Eq: descent of the augmented potential}
    \end{align}}
\end{lemma}
\begin{proof}
    \blue{By \Cref{Lemma: approximate sufficient descent property storm}, $C_{\bx}^{k}=\eta_T^{-1}-\tilde L_F\ge\frac{1}{2\eta_T}$, and taking expectations,
    \[
        \EE\Psi_r(\calQ^{k+1})-\EE\Psi_r(\calQ^{k})\le-C_{\bB}\EE[\Delta_{\bB}^{k+1}+\Delta_{\bB}^{k}]-\tfrac{1}{2\eta_T}\EE\Delta_{\bx}^{k+1}+\tfrac{\eta_T}{2}q_k .
    \]
    For $k\ge1$ we have $b_{k+1}=b$, and \Cref{Lemma: Recursive bound on gradient error} with $(1-\rho_T)^2\le1-\rho_T$ gives $q_{k+1}\le(1-\rho_T)q_k+\frac{2\rho_T^2\sigma^2}{b}+\frac{4L_F^2}{b}\EE[\Delta_{\bx}^{k+1}+\Delta_{\bB}^{k+1}]$. By \Cref{Lemma: tracking of B increments}, $\EE\Delta_{\bB}^{k+1}\le\frac{12}{r}\EE[H_k-H_{k+1}]+G\,\EE\Delta_{\bx}^{k+1}$, hence
    \[
        \bigl(q_{k+1}+d_0\EE H_{k+1}\bigr)-\bigl(q_k+d_0\EE H_k\bigr)\le-\rho_Tq_k+\frac{2\rho_T^2\sigma^2}{b}+\frac{4L_F^2(1+G)}{b}\EE\Delta_{\bx}^{k+1}.
    \]
    Multiply by $\eta_T/\rho_T$ and add it to the first display. By \eqref{Eq: the parameters assumptions in storm appendix}, $\frac{\eta_T}{\rho_T}\cdot\frac{4L_F^2(1+G)}{b}\le\frac{1}{4\eta_T}$, so the coefficient of $\EE\Delta_{\bx}^{k+1}$ is at most $-\frac{1}{4\eta_T}$, the coefficient of $q_k$ is $\frac{\eta_T}{2}-\eta_T=-\frac{\eta_T}{2}$, and the noise term is $\frac{\eta_T}{\rho_T}\cdot\frac{2\rho_T^2\sigma^2}{b}=\frac{2\eta_T\rho_T\sigma^2}{b}$.}
\end{proof}

\subsection{Proof of the Averaged Stationarity Error Bound for DualHash-StoRM}
\label{app-toc-16}
\label{subsubsec:proof-avg-stationarity-storm}
We now take $\eta_{k}\equiv \eta_{T}$ and $\rho_{k}\equiv \rho_{T}$. The first result estimates the averaged expected stationarity error.
\begin{lemma}
    \label{lemma: the averaged stationarity error bound} 
    \blue{Under the conditions of \Cref{Lemma: Weighted error accumulation}, let $S_3:=4(3\eta_T^{-2}+L_F^2+3(\tau L_F)^2)$, $N_T:=\max\{S_1/C_{\bB},\,S_2/C_{\bB},\,4\eta_TS_3,\,4/\eta_T\}$, and let $\Psi_r^{*}>-\infty$ be the lower bound, independent of $T$, of $\EE[\Psi_r(\calQ^{k})]+\frac{\eta_T}{\rho_T}\EE\|\be^{k}\|^2$ for all $k\ge1$ given in \Cref{prop:lyapunov-lower-bound}(ii). Then for any $T \geq 1$,
    \begin{align}
        \frac{1}{T}\sumT \EE [\mathrm{dist}^{2}(\bm{0}, \mathcal K(\bx^{k}, \bB^{k}, \bm{\Lambda}^{k}))]
        \leq \frac{N_T}{T}\Bigl(\tilde{\Delta}_{1}+\frac{\eta_T\sigma^2}{\rho_Tb_1}\Bigr)+\frac{2N_T\eta_T\rho_T\sigma^{2}}{b}, \label{Eq: the averaged stationarity error bound in lemma}
    \end{align}
    where $\tilde{\Delta}_{1}:=\calL(\bx^{1},\bB^{1},\bLmbd^{1})-\Psi_r^{*}$.}
\end{lemma}
\begin{proof}
    \blue{Taking expectations in \eqref{Eq: the stationarity error bound in Lemma in storm} and using $S_3^{k}=S_3$,
    \[
        \EE\,\mathrm{dist}^{2}(\bm{0}, \mathcal K(\bx^{k}, \bB^{k}, \bm{\Lambda}^{k}))\le S_1\EE\Delta_{\bB}^{k+1}+S_2\EE\Delta_{\bB}^{k}+S_3\EE\Delta_{\bx}^{k+1}+2q_k .
    \]
    By the definition of $N_T$, $S_1\le N_TC_{\bB}$, $S_2\le N_TC_{\bB}$, $S_3\le N_T/(4\eta_T)$, and $2\le N_T\eta_T/2$, so the right-hand side is at most
    \[
        N_T\Bigl(C_{\bB}\EE[\Delta_{\bB}^{k+1}+\Delta_{\bB}^{k}]+\tfrac{1}{4\eta_T}\EE\Delta_{\bx}^{k+1}+\tfrac{\eta_T}{2}q_k\Bigr)\le N_T\Bigl(\widehat{W}_k-\widehat{W}_{k+1}+\tfrac{2\eta_T\rho_T\sigma^2}{b}\Bigr),
    \]
    where the last step is \eqref{Eq: descent of the augmented potential}. Summing over $k=1,\ldots,T$ and dividing by $T$, the average is at most $\frac{N_T}{T}(\widehat{W}_1-\widehat{W}_{T+1})+\frac{2N_T\eta_T\rho_T\sigma^2}{b}$. Since $H_k\ge0$, $\widehat{W}_{T+1}\ge\EE\Psi_r(\calQ^{T+1})+\frac{\eta_T}{\rho_T}q_{T+1}\ge\Psi_r^{*}$. For $\widehat{W}_1$: $H_1=0$ and $q_1\le\sigma^2/b_1$ by \Cref{Ass: main assumptions}(iii); moreover $\Delta_{\bB}^{1}=0$ because $\bB^0=\bB^1$, so $\Psi_r(\calQ^{1})=\calL(\bx^1,\bB^1,\bLmbd^1)-C_1\Delta_{\bB}^{2}-C_3\Delta_{\bx}^{2}\le\calL(\bx^1,\bB^1,\bLmbd^1)$. Hence $\widehat{W}_1\le\calL(\bx^1,\bB^1,\bLmbd^1)+\frac{\eta_T\sigma^2}{\rho_Tb_1}$, which gives \eqref{Eq: the averaged stationarity error bound in lemma}.}
\end{proof}

\label{app-toc-17}
\label{Subsubsec: convergence rate STORM}
\begin{proof}
    \blue{Insert $\eta_T=\eta/(\tilde L_FT^{1/3})$, $\rho_T=8\rho\eta^2/T^{2/3}$, $b_1\ge c_bT^{1/3}$ and $b_k=b$ into \eqref{Eq: the averaged stationarity error bound in lemma}.
