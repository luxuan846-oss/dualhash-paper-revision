\section{\blue{Problem Formulation and Preliminaries}}

\subsection{{Preliminaries}}
\blue{An} extended real-valued function $r:\RR^{q}\to\blue{[-\infty,+\infty]}$ is called
\textit{proper} if \blue{$r(\bx)>-\infty$ for all $\bx\in\RR^{q}$ and} its domain
$\operatorname{dom}(r):=\{\bx\in\RR^{q}\mid r(\bx)<+\infty\}$ is nonempty; \blue{it is called
\textit{lower semicontinuous} if its epigraph
$\operatorname{epi}(r):=\{(\bx,t)\in\RR^{q+1}\mid r(\bx)\le t\}$ is closed.}
\blue{We write
$\by\xrightarrow{r}\bx$ to denote $\by\to\bx$ and $r(\by)\to r(\bx)$. The \textit{(limiting) subdifferential} \cite[Definition~8.3]{Rockafellar1998VariationalA} of a function $r:\RR^{q}\to[-\infty,+\infty]$ at a point
$\bx$ where $r(\bx)$ is finite is given by
\begin{equation}\label{Eq: limiting subdifferential}
\begin{aligned}
\partial r(\bx):={}&\Bigl\{\bv\in\RR^{q}:\ \exists\,\bx^{k}\xrightarrow{r}\bx,\ \bv^{k}\to\bv\ \text{with}\\
&\quad \liminf_{\substack{\by\to\bx^{k}\\ \by\neq\bx^{k}}}
\frac{r(\by)-r(\bx^{k})-\langle\bv^{k},\by-\bx^{k}\rangle}{\|\by-\bx^{k}\|}
\ge0\quad\text{for all } k\Bigr\}.
\end{aligned}
\end{equation}When $r$ is continuously differentiable or convex, the above subdifferential coincides with the
classical gradient or convex subdifferential of $r$
\cite[Exercise~8.8(b) and Proposition~8.12]{Rockafellar1998VariationalA}.
Moreover, for any continuously differentiable $g$, it holds that
$\partial(r+g)(\bx)=\partial r(\bx)+\nabla g(\bx)$
\cite[Exercise~8.8(c)]{Rockafellar1998VariationalA}; and if
$r(\bx_{1},\bx_{2})=r_{1}(\bx_{1})+r_{2}(\bx_{2})$ with $r_{1},r_{2}$ proper and lower
semicontinuous, then $\partial r(\bx_{1},\bx_{2})=\partial r_{1}(\bx_{1})\times\partial r_{2}(\bx_{2})$
\cite[Proposition~10.5]{Rockafellar1998VariationalA}.}
For a proper function $r:\RR^{l}\to(-\infty,+\infty]$, its conjugate is $r^*(\by):=\sup_{\bx}\{\langle\by,\bx\rangle-r(\bx)\}$ and its biconjugate is $r^{**}:=(r^*)^*$. \blue{If $r$ has an affine minorant, then $r^{**}$ is the largest lower semicontinuous convex function below $r$~\cite[Theorem~11.1]{Rockafellar1998VariationalA}.}
For a proper, \blue{lower semicontinuous, and bounded-below} function $r$, the \textit{proximal mapping} of $r$, denoted by $\operatorname{prox}_{\tau r}(\by)$, is defined as
\begin{align}
    % \label{Def: the definition of proximal operator}
    \operatorname{prox}_{\tau r}(\by) := \argmin\limits_{\bx \in \operatorname{dom}(r)}\left\{ r(\bx) + \frac{1}{2\tau}\|\bx - \by\|^{2}\right\}, \notag
\end{align}
where $\tau$ is a positive scalar parameter. The proximal mapping admits the following
properties.
\begin{proposition}\label{prop:rockafellar}
\textnormal{\cite{Rockafellar1998VariationalA}}
    Let $r : \mathbb{R}^{q}\rightarrow \mathbb{R}\cup \{+\infty\}$ be proper, \blue{lower semicontinuous, and bounded below}. Then, the following holds.
    \begin{enumerate}[label=(\roman*)]
        \item The set $\operatorname{prox}_{\tau r}(\bx)$ is nonempty and compact
            for any $\bx \in \mathbb{R}^{q}$ and $\tau > 0$.

        \item If $r$ is convex, then $\operatorname{prox}_{\tau r}(\bx)$
            contains exactly one value for any $\bx \in \mathbb{R}^{q}$ and
            $\tau > 0$.
    \end{enumerate}
\end{proposition}

\blue{A differentiable} function $g: \RR^q \rightarrow \RR$ is said to be $L_g$-smooth if $\| \nabla g(\bx) - \nabla g(\bx') \| \leq L_g \| \bx - \bx'\|$ for all $\bx,\, \bx' \in \RR^q$.
A mapping $\mathcal{M} : \blue{\Omega} \to \RR^l$ is said to be $C$-Lipschitz continuous if $\|\mathcal{M}(\bx) - \mathcal{M}(\bx')\| \leq C\|\bx - \bx'\|$ for all $\bx, \bx' \in \blue{\Omega}$. For an $L_g$-smooth function, \blue{the descent lemma \blue{\cite{Bertsekas1997}} states that
\begin{align}
\label{Eq: the descent lemma}
    g(\bx) \leq g(\bx') + \langle \nabla g(\bx'), \bx - \bx' \rangle + \frac{L_g}{2} \| \bx - \bx'\|^2.
\end{align}}

\subsection{W-type Regularizer: Definition and Properties}
\label{Section: W-regularizer}
Deep hashing aims to learn network parameters \(\bx \in \RR^{\blue{q}}\) that generate binary codes preserving semantic similarity in the Hamming space. A direct formulation is
\begin{align}
\label{Eq:direct_deep_hashing_problem}
\blue{\min_{\bx \in \RR^{q}}\ \frac{1}{n}\sum_{i=1}^{n}\ell_{s,i}\!\big(\operatorname{sgn}(\mathcal{D}(\bx))\big),}
\end{align}
where \(n\) is the number of training samples, \blue{\(\bD_i:\RR^q\to\RR^d\) is the continuous network output mapping for sample \(i\), \(\mathcal{D}(\bx):=[\bD_1(\bx)^\top;\ldots;\bD_n(\bx)^\top]\), and \(\ell_{s,i}:\RR^{n\times d}\to\RR\) is a similarity-preserving loss} (e.g., pairwise cross-entropy \cite{zhuDeepHashingNetwork2016}). This objective is hard to optimize by backpropagation because \(\operatorname{sgn}(\cdot)\) is nondifferentiable and has zero gradient almost everywhere.

\blue{To address this issue, penalties with W-shaped graphs, vanishing exactly at
$\pm1$, are added to drive the continuous outputs toward $\{-1,1\}^d$. Such penalties
are standard in deep hashing~\cite{liuDeepSupervisedHashing,yuanCentralSimilarityQuantization2020,he2025mambahash,
yin2026crh}. Their formulas differ, but our analysis depends only on their common
shape, which we formalize next.}
\begin{definition}[W-type regularizer]\label{def:wtype}
Let \(\phi:\mathbb{R}\to[0,+\infty)\). For \(\bz\in\mathbb{R}^{d}\) and \(\bZ\in\mathbb{R}^{n\times d}\), define \(h(\bz):=\sum_{j=1}^{d}\phi(z_j)\) and \(h(\bZ):=\sum_{i=1}^{n}\sum_{j=1}^{d}\phi(z_{ij})\); with a slight abuse of notation, \(h\), \(h^*\), and \(h^{**}\) refer to both the vector and the matrix cases. We call \(h\) a \emph{W-type regularizer} if \(\phi\) satisfies:
\begin{enumerate}[label=\textup{(W\arabic*)}]
  \item\label{W1} \(\phi\) is even and continuous, and \(\phi(t)=0\) if and only if \(t\in\{-1,1\}\);
  \item\label{W2} \(\phi\) is convex on \([1,+\infty)\);
  \item\label{W3} \(\phi\) decreases at a nonvanishing rate on \([0,1)\): for every \(b\in(0,1)\) there exists \(c_b>0\) such that \(\phi(s)-\phi(t)\ge c_b(t-s)\) for all \(0\le s<t\le b\).
\end{enumerate}
\end{definition}

For W-type regularizers, the convex envelope and the conjugate admit the following explicit description. The proof is provided in \Cref{app:wtype-conjugate-proof}.
\begin{lemma}\label{lem:wtype-conj}
Let \(\phi:\mathbb{R}\to[0,+\infty)\) satisfy \ref{W1}--\ref{W2}. Then the following holds
\begin{enumerate}[label=\textup{(\alph*)}]
\item \(\phi^{**}(t)=\phi\bigl(\max\{|t|,1\}\bigr)\) for all \(t\in\mathbb{R}\); in particular, \(\phi^{**}(t)=\phi(t)\) if and only if \(|t|\ge1\);
\item \(\phi^*\) is proper, lower semicontinuous, convex, and even; \(\phi^*(y)\ge|y|\) for all \(y\); and \(\operatorname{dom}\phi^*\) has nonempty interior.
\end{enumerate}
Consequently, \(h^*(\bLmbd)=\sum_{i,j}\phi^*(\Lambda_{ij})\), \(h^{**}(\bB)=\sum_{i,j}\phi^{**}(b_{ij})\), and \(h(\bB)=h^{**}(\bB)\) if and only if \(|b_{ij}|\ge1\) for all \(i,j\).
\end{lemma}

\subsection{W-type Regularized Deep Hashing Model}
\label{Sec:W-type Regularized Deep Hashing Model}
\blue{With the W-type regularizer \(h\) defined in \Cref{def:wtype},} the model is now trained to produce the \(i\)-th continuous network output \(\bD_i(\bx)\), and binary codes are obtained post-training via \(\operatorname{sgn}(\bD_i(\bx))\blue{\in\{-1,1\}^d}\). \blue{Consequently, we formulate the following core optimization problem for W-type regularized deep hashing methods, which serves as a continuous surrogate of Problem~\eqref{Eq:direct_deep_hashing_problem}:}
\begin{align}
\label{Eq: the primal deep hashing problem}
\blue{\min_{\bx \in \RR^{q}} \quad J_0(\bx):=\frac{1}{n}\sum_{i=1}^n f_i(\bx) + \sum_{i=1}^n h(\bD_i(\bx)),}
\tag{P0}
\end{align}\blue{where
\(f_i(\bx):=\ell_{s,i}\!\left(\mathcal{D}(\bx)\right)+\frac{\mu}{2}\|\bx\|^2\) with \(\mu>0\).} \blue{Following \cite{bai2019proxquant}, we define the quantization error in deep hashing problems as
\begin{equation}
\label{eq:quantization-error}
Q_1(\bx):=\bigl\|\mathcal{D}(\bx)-\operatorname{sgn}(\mathcal{D}(\bx))\bigr\|_1
=\sum_{i=1}^{n}\sum_{j=1}^{d}\bigl||\mathcal{D}_{ij}(\bx)|-1\bigr|,
\end{equation}\red{where the equality follows componentwise from $\min_{b\in\{-1,1\}}|z-b|=\bigl||z|-1\bigr|$. We relate each W-type regularizer to the quantization error $Q_1$ in the last column of \Cref{tab:wtype}. Specifically, for $p=1$, which is also used in our experiments, $h(\mathcal{D}(\bx))=\lambda Q_1(\bx)$, so the regularizer is exactly the scaled quantization error. For the other regularizers, $h(\mathcal{D}(\bx))\le c_\phi Q_1(\bx)$ provides only an upper bound, which may be loose.}}

\blue{Using variable splitting, we introduce continuous auxiliary variables \(\bb_i\in\RR^d\) and \(\bB=[\bb_1^\top;\ldots;\bb_n^\top]\in\RR^{n\times d}\), subject to \(\bb_i=\bD_i(\bx)\) for all \(i=1,\dots,n\),} to obtain the following equivalent nonlinearly constrained problem:
\begin{align}
    \blue{\min_{\bx \in \RR^{q},\ \bB \in \RR^{n\times d}}\quad \frac{1}{n}\sum_{i=1}^{n} f_i(\bx)+\sum_{i=1}^{n}h(\bb_i)\quad \text{s.t.}\quad \bb_i=\bD_i(\bx),\ \forall i=1,\dots,n.} \notag
\end{align}
\blue{Let \(\gamma>0\) be a penalty parameter and define}
\begin{align}
    \blue{F(\bx,\bB):= \frac{1}{n}\sum_{i=1}^{n}F_i(\bx,\bb_i),\qquad F_i(\bx,\bb_i):= f_i(\bx)+\frac{\gamma}{2}\|\bD_i(\bx)-\bb_i\|^2.} \notag
\end{align}
\blue{Averaging over \(i\) gives the decomposition \(F=f+P\), where}
\begin{align}
    \blue{f(\bx):= \frac{1}{n}\sum_{i=1}^{n}f_i(\bx),\qquad P(\bx,\bB):= \frac{\gamma}{2n}\sum_{i=1}^{n}\|\bD_i(\bx)-\bb_i\|^2.} \notag
\end{align}
We then \blue{relax these constraints by} the quadratic penalty method \cite[Sec.~17.1]{nocedalWright2006} to obtain the following two-block finite-sum optimization problem:
\begin{align}
    \label{Prop: block-wise regularized problem}
    \blue{\min_{\bx \in \RR^{q},\,\bB \in \RR^{n\times d}}\ J(\bx,\bB):=F(\bx,\bB)+h(\bB).} \tag{P}
\end{align}\red{This reformulation decomposes the original problem \eqref{Eq: the primal deep hashing problem} into simpler structural blocks and yields a tractable two-block formulation.} Building upon Problem~\eqref{Prop: block-wise regularized problem}, we develop a stochastic primal-dual algorithm {that addresses the lack of tractable proximal mappings in the nonsmooth composite form and the absence of convergence guarantees in existing deep hashing methods.}
\subsection{\blue{Basic Assumptions}}
\blue{We impose the following basic assumptions on Problem \eqref{Eq: the primal deep hashing problem}.}
\begin{assumption}
\label{Ass: basic regularity}
\leavevmode
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2em,font=\color{blue}]
    \item \blue{The average loss is bounded below: $\underline{\ell}:=\inf_{\bx}\bar{\ell}_{s}\!\left(\mathcal{D}(\bx)\right)>-\infty$ where \(\bar{\ell}_s:=n^{-1}\sum_{i=1}^{n}\ell_{s,i}\). }
    \item \blue{The function $f$ is $L_f$-smooth on $\RR^q$, and $\nabla\mathcal{D}$ is $L_{\nabla\mathcal{D}}$-Lipschitz on $\RR^q$, with $L_f,L_{\nabla\mathcal{D}}>0$. There exist constants $C_{\mathcal{D}},G_{\mathcal{D}}>0$ such that
    \[
        \|\mathcal{D}(\bx)\|\le C_{\mathcal{D}},\qquad
        \|\nabla\mathcal{D}(\bx)\|\le G_{\mathcal{D}},\qquad
        \forall\,\bx\in\RR^q.
    \]}
\end{enumerate}
\end{assumption}
\begin{remark}[Coercivity of $J(\bx, \bB)$]\label{rem:coercivity}
\blue{Assumption~\ref{Ass: basic regularity}(i) yields the coercivity of $J$. Indeed, since $P\ge0$,
\[
J(\bx,\bB)\ge\underline{\ell}+\frac{\mu}{2}\|\bx\|^2+h(\bB),
\]
and $h$ grows at least linearly in each entry because $\phi^*$ is even and $\operatorname{dom}\phi^*$ has nonempty interior (Lemma~\ref{lem:wtype-conj}).}
\end{remark}

\label{Eq: the proximal mapping of conjugate function}

\end{equation}
