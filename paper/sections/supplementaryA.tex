\section{Experimental Details}
\label{app:experiments}

We consider deep supervised pairwise hashing in our experiments, which aims to learn binary codes that minimize Hamming distances for similar pairs and maximize them for dissimilar pairs. Let $\bA = \{\ba_i \in \RR^{d_a} : i = 1, \ldots, n\}$ be the training set with label vectors $\by_i \in \{0,1\}^C$, where $d_a$ is the input dimension and $C$ is the number of classes. The label vectors are one-hot for single-label datasets and multi-hot for multi-label datasets.

\paragraph{Network Architecture}
A CNN extracts features from $\ba_i$, followed by a fully connected hash layer that produces the continuous code $\bD_i(\bx)\in\RR^d$.

\paragraph{Loss Function}
For post-training binary codes $\operatorname{sgn}(\bD_i(\bx))\in\{-1,1\}^d$, the Hamming distance is
\begin{equation*}
 d_{ij}^h = \tfrac{1}{2}\left(d - \left\langle \operatorname{sgn}(\bD_i(\bx)), \operatorname{sgn}(\bD_j(\bx)) \right\rangle\right).
\end{equation*}
Following~\cite{xiaSupervisedHashingImage2014,laiSimultaneousFeatureLearning2015}, similarity $s^o_{ij} = 1$ if samples share labels, else $0$. The likelihood is:
\[
\begin{aligned}
P\bigl(s^o_{ij} \mid \operatorname{sgn}(\bD_i(\bx)),\operatorname{sgn}(\bD_j(\bx))\bigr)
={}&\ \operatorname{sig}_{\kappa}\!\left(\left\langle \operatorname{sgn}(\bD_i(\bx)),\operatorname{sgn}(\bD_j(\bx)) \right\rangle\right)^{s^o_{ij}}\\
&\times\left(1 - \operatorname{sig}_{\kappa}\!\left(\left\langle \operatorname{sgn}(\bD_i(\bx)),\operatorname{sgn}(\bD_j(\bx)) \right\rangle\right)\right)^{1 - s^o_{ij}}.
\end{aligned}
\]
where $\operatorname{sig}_{\kappa}(t) = (1 + \exp(-\kappa t))^{-1}$, $\kappa < 1$.

Using continuous codes $\bD_i(\bx)$ and $s^h_{ij} = \tfrac{1}{2} \langle \bD_i(\bx), \bD_j(\bx) \rangle$, the loss becomes:
\begin{align}
\label{Eq: the pairwise CE loss}
\min_{\bx} \; \ell_{\calS}(\bx) = \frac{1}{|\calS|}\sum_{s^o_{ij}\in\calS}\Bigl[\log\bigl(1+\exp(\kappa s^h_{ij})\bigr)-\kappa\,s^o_{ij}s^h_{ij}\Bigr].
\end{align}
Substituting \eqref{Eq: the pairwise CE loss} into Problem~\eqref{Prop: block-wise regularized problem} gives the objective used in our experiments:
\begin{align}
% \label{Eq: the pairwise quantization-based hashing}
\min_{\bx \in \calX, \bB \in \RR^{n\times d}} \;
&\frac{1}{|\calS|}\sum_{s^o_{ij}\in\calS}\Bigl[\log\bigl(1+\exp(\kappa s^h_{ij})\bigr)-\kappa\,s^o_{ij}s^h_{ij}\Bigr] \notag \\
&+ \frac{\gamma}{2n}\sum_{i=1}^n \| \bb_i - \bD_i(\bx)\|_2^2
+ \sum_{i=1}^n \lambda \|\,|\bb_i| - \bm{1}\,\|_1. \notag
\end{align}

\paragraph{Out-of-Sample Extension}
For an unseen query $\ba^{\mathrm{qry}} \notin \bA$, its binary code is predicted as $\operatorname{sgn}\!\bigl(\bD(\bx; \ba^{\mathrm{qry}})\bigr)$.

\subsection{Parameter Sensitivity Analysis}
\label{app:sensitivity}
\begin{figure}[!htbp]
    % \vspace{-1mm}
    \centering
    \includegraphics[width=0.95\linewidth]{results_images/sensitivity_2datasets.pdf}
    \caption{The mAP results of DualHash-StoM with different settings of \(\eta\) and \(\gamma\) for 64-bit and 16-bit codes on CIFAR-10 and NUS-WIDE.}
    \label{fig:sensitivity_analysis_DualHash}
\end{figure}
To explore the sensitivity of key hyperparameters $\eta$ and $\gamma$ in 
DualHash, we conduct a grid search with $\eta \in \{10^{-5}, 10^{-3}, 
10^{-2}, 1\}$ and $\gamma \in \{0.3, 3, 30, 100\}$. Unlike many deep hashing 
methods that exhibit significant hyperparameter sensitivity, DualHash-StoM 
demonstrates robust performance across a wide range of hyperparameter 
combinations (see \Cref{fig:sensitivity_analysis_DualHash}). 
Specifically, 64-bit codes achieve excellent stability with mAP variations 
below 3\% on CIFAR-10 and 8\% on NUS-WIDE. While 16-bit codes show greater 
sensitivity (up to 18\% on CIFAR-10), this behavior is consistent with the 
intrinsic optimization challenges of extremely compact hash codes. Overall, 
the method demonstrates strong stability for standard hash lengths ($\geq$64 bits), 
which simplifies hyperparameter tuning in practical high-precision retrieval 
applications.
