\subsubsection*{(a) Relation between the primal-dual reformulation and the original objective}

\begin{itemize}\itshape
\item The key question is whether this transformed problem provides meaningful guarantees for the original nonconvex W-regularized hashing objective. I do not find this connection convincingly established. In particular, the paper should clarify whether there is strong duality, zero duality gap, or at least a rigorous stationarity relationship between the transformed primal-dual problem and the original objective.
\par\smallskip
\normalfont\upshape\responseLabel
{
We thank the reviewer for this key question, which concerns the link between our reformulation and the original hashing objective. We agree that the submitted manuscript did not establish this link. The revised manuscript addresses it in a new Section~5.1, which first examines the duality gap between Problems~(P) and~(D) and then relates KKT points of $\calL$ directly to stationary points of Problem~(P).
\begin{enumerate}[label=\textbf{(\roman*)},leftmargin=2.2em]
\item \textbf{Zero duality gap.} Proposition~5.1 shows that every primal--dual optimal pair with zero duality gap is a KKT point of $\calL$. Since both problems attain their optimal values, zero duality gap and strong duality are equivalent here. The duality gap splits as
\[
p^*-d^*=(p^*-p^{**})+(p^{**}-d^*),
\]
where $p^{**}$ is the optimal value after replacing $h$ with its convex envelope $h^{**}$; the first term is the loss caused by this replacement, and the second stems from the nonconvexity of $\calL$ in $(\bx,\bB)$. Example~5.1 shows that the first term alone can be positive: for a one-dimensional instance with convex $F$, the second term vanishes and the duality gap equals $\tfrac12$. Hence strong duality between (P) and (D) depends on the instance, and a guarantee for Problem~(P) has to come from the KKT points of $\calL$ themselves.
\item \textbf{Stationarity relationship.} Proposition~5.2 characterizes exactly when a KKT point of $\calL$ yields a stationary point of Problem~(P): this holds if and only if the Fenchel--Young equality holds for $h$, which is equivalent to $|\tilde b_{ij}|\ge1$ for all $i,j$. This relationship requires no zero duality gap, and condition~(c) involves only $\tilde{\bB}$, so it can be checked directly at the computed point.
\end{enumerate}
Proposition~5.2 is stated in Section~5.1 and proved in Appendix~A.2 \green{(Section~5.1, Page XX, lines XX--XX; Appendix~A.2, Page XX, lines XX--XX)}. For the reviewer's convenience, we reproduce it below.
\begin{quote}\normalfont\upshape
\textbf{Proposition 5.2.} Let $(\tilde{\bx},\tilde{\bB},\tilde{\bLmbd})$ be a KKT point of $\calL$, i.e., it satisfies (5.1). Then the following statements are equivalent:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $(\tilde{\bx},\tilde{\bB})$ is a stationary point of Problem~(P), i.e., $\zero\in\nabla F(\tilde{\bx},\tilde{\bB})+\{\zero\}\times\partial h(\tilde{\bB})$;
\item the Fenchel--Young equality holds for $h$ at $(\tilde{\bB},\tilde{\bLmbd})$: $h(\tilde{\bB})+h^*(\tilde{\bLmbd})=\langle\tilde{\bLmbd},\tilde{\bB}\rangle$;
\item $|\tilde{b}_{ij}|\ge1$ for all $i,j$.
\end{enumerate}

\textit{Proof.} Since $h^{*}$ is proper, lower semicontinuous, and convex, its conjugate is $h^{**}(\bB)=\lambda\sum_{i,j}(|b_{ij}|-1)_{+}$, and the subgradient inversion rule~\cite[Theorem~23.5]{rockafellar1970convex} gives
\[
\tilde{\bB}\in\partial h^{*}(\tilde{\bLmbd})
\iff \tilde{\bLmbd}\in\partial h^{**}(\tilde{\bB})
\iff h^{**}(\tilde{\bB})+h^{*}(\tilde{\bLmbd})=\langle\tilde{\bLmbd},\tilde{\bB}\rangle.\tag{A.4}
\]
Since $(\tilde{\bx},\tilde{\bB},\tilde{\bLmbd})$ satisfies (5.1), all three statements in (A.4) hold, and $\tilde{\bLmbd}=-\nabla_{\bB}F(\tilde{\bx},\tilde{\bB})$ together with $\nabla_{\bx}F(\tilde{\bx},\tilde{\bB})=\zero$ shows that (a) holds if and only if $\tilde{\bLmbd}\in\partial h(\tilde{\bB})$.

(b)$\iff$(c). By the last equality in (A.4), (b) holds if and only if $h(\tilde{\bB})=h^{**}(\tilde{\bB})$. Since $h$ and $h^{**}$ are separable and $\lambda\big||b|-1\big|=\lambda(|b|-1)_{+}$ if and only if $|b|\ge1$, this is equivalent to $|\tilde{b}_{ij}|\ge1$ for all $i,j$.

(b)$\Rightarrow$(a). By the definition of $h^{*}$, (b) means that $\tilde{\bB}$ is a global minimizer of $h(\cdot)-\langle\tilde{\bLmbd},\cdot\rangle$. Fermat's rule~\cite[Theorem~10.1]{rockafellar1998variational} and \cite[Exercise~8.8(c)]{rockafellar1998variational} then give $\zero\in\partial h(\tilde{\bB})-\tilde{\bLmbd}$, i.e., $\tilde{\bLmbd}\in\partial h(\tilde{\bB})$, which is (a).

(a)$\Rightarrow$(c). By (A.4) and (a), $\tilde{\bLmbd}\in\partial h(\tilde{\bB})\cap\partial h^{**}(\tilde{\bB})$. Since $h$ and $h^{**}$ are separable~\cite[Proposition~10.5]{rockafellar1998variational}, it suffices to consider a scalar entry $b=\tilde{b}_{ij}$. If $|b|<1$, then $\partial h^{**}(b)=\{0\}$, whereas $\partial h(b)=\{-\lambda\operatorname{sgn}(b)\}$ for $b\neq0$ and $\partial h(0)=\{-\lambda,\lambda\}$; hence $0\notin\partial h(b)$ and the intersection is empty. Therefore $|\tilde{b}_{ij}|\ge1$ for all $i,j$. \hfill$\square$
\end{quote}
}
\end{itemize}

\item {\itshape (page 13) Give more details to justify

\normalfont\upshape\responseLabel
We thank the reviewer for this comment. The bound (5.11) \red{(now (5.8))} follows from the standard variance formula for uniform sampling. With the revised probabilistic framework and conditional-expectation notation above, its derivation reads as follows.
\begin{quote}\normalfont\upshape
\textit{Proof.} Recall that the stochastic gradient error $\be^k$ of Algorithm 4.1 is defined as
\[
\be^k
=
\frac{1}{b_k}\sum_{i\in\calJ_k}
\left(
\nabla_{\bx}F_i(\bz^k,\bb_i^k)
-
\nabla_{\bx}F(\bz^k,\bB^k)
\right).
\]
Since $(\bz^k,\bB^k)$ is $\calF_k$-measurable and
$\calJ_k$ is sampled uniformly without replacement and independently
of $\calF_k$, define
\[
\bv_i^k
:=
\nabla_{\bx}F_i(\bz^k,\bb_i^k)
-
\nabla_{\bx}F(\bz^k,\bB^k).
\]
Conditionally on $\calF_k$, Assumption~5.1(iii)
gives
\[
\frac{1}{n}\sum_{i=1}^n \bv_i^k=\zero,
\qquad
\frac{1}{n}\sum_{i=1}^n\|\bv_i^k\|^2\leq\sigma^2
\quad\text{a.s.}
\]
For $n\geq2$, the variance formula for uniform sampling without replacement
therefore yields
\[
\EE_k[\be^k]=\zero,
\]
and
\[
\EE_k\!\left[\|\be^k\|^2\right]
=
\frac{n-b_k}{n-1}\frac{1}{b_k}
\frac{1}{n}\sum_{i=1}^n\|\bv_i^k\|^2
\leq
\frac{\sigma^2}{b_k}
\quad\text{a.s.}
\]
The case $n=1$ is immediate since $\be^k=\zero$. Taking the full
expectation and using the tower property gives (5.11) \red{(now (5.8))}.
\hfill$\square$
\end{quote}
