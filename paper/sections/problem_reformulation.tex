\section{\blue{Problem Formulation and Preliminaries}}
\label{Section: Problem Reformulation}
\subsection{{Preliminaries}}
\blue{An} extended real-valued function $r:\RR^{q}\to\blue{[-\infty,+\infty]}$ is called
\textit{proper} if \blue{$r(\bx)>-\infty$ for all $\bx\in\RR^{q}$ and} its domain
$\operatorname{dom}(r):=\{\bx\in\RR^{q}\mid r(\bx)<+\infty\}$ is nonempty; \blue{it is called
\textit{lower semicontinuous} if its epigraph
$\operatorname{epi}(r):=\{(\bx,t)\in\RR^{q+1}\mid r(\bx)\le t\}$ is closed.}
We write
$\by\xrightarrow{r}\bx$ to denote $\by\to\bx$ and $r(\by)\to r(\bx)$. The \textit{(limiting) subdifferential}~\cite[Definition~8.3]{Rockafellar1998VariationalA} of a function $r:\RR^{q}\to[-\infty,+\infty]$ at a point
$\bx$ where $r(\bx)$ is finite is given by
\begin{equation}\label{Eq: limiting subdifferential}
\begin{aligned}
\partial r(\bx):={}&\Bigl\{\bv\in\RR^{q}:\ \exists\,\bx^{k}\xrightarrow{r}\bx,\ \bv^{k}\to\bv\ \text{with}\\
&\quad \liminf_{\substack{\by\to\bx^{k}\\ \by\neq\bx^{k}}}
\frac{r(\by)-r(\bx^{k})-\langle\bv^{k},\by-\bx^{k}\rangle}{\|\by-\bx^{k}\|}
\ge0\quad\text{for all } k\Bigr\}.
\end{aligned}
\end{equation}When $r$ is \blue{continuously differentiable} or convex, the above subdifferential coincides with the
\blue{classical gradient} or convex subdifferential of $r$
\cite[Exercise~8.8(b) and Proposition~8.12]{Rockafellar1998VariationalA}.
Moreover, for any continuously differentiable $g$,
$\partial(r+g)(\bx)=\partial r(\bx)+\nabla g(\bx)$
\cite[Exercise~8.8(c)]{Rockafellar1998VariationalA}; and if
$r(\bx_{1},\bx_{2})=r_{1}(\bx_{1})+r_{2}(\bx_{2})$ with $r_{1},r_{2}$ proper and lower
semicontinuous, then $\blue{\partial r(\bx_{1},\bx_{2})=\partial r_{1}(\bx_{1})\times\partial r_{2}(\bx_{2})}$
\cite[Proposition~10.5]{Rockafellar1998VariationalA}.
For a proper function $r:\RR^{l}\to(-\infty,+\infty]$, its conjugate is $r^*(\by):=\sup_{\bx}\{\langle\by,\bx\rangle-r(\bx)\}$ and \blue{its biconjugate is $r^{**}:=(r^*)^*$.} \blue{If $r$ has an affine minorant, then $r^{**}$ is the largest lower semicontinuous convex function below $r$~\cite[Theorem~11.1]{Rockafellar1998VariationalA}.}
For a proper, \blue{lower semicontinuous, and bounded-below} function $r$, the \textit{proximal mapping} of $r$, denoted by $\operatorname{prox}_{\tau r}(\by)$, is defined as
\begin{align}
    % \label{Def: the definition of proximal operator}
    \operatorname{prox}_{\tau r}(\by) := \argmin\limits_{\bx \in \operatorname{dom}(r)}\left\{ r(\bx) + \frac{1}{2\tau}\|\bx - \by\|^{2}\right\}, \notag
\end{align}
where $\tau$ is a positive scalar parameter. The proximal mapping admits the following
properties.
\begin{proposition}\label{prop:rockafellar}
\textnormal{\cite{Rockafellar1998VariationalA}}
    Let $r : \mathbb{R}^{q}\rightarrow \mathbb{R}\cup \{+\infty\}$ be proper, \blue{lower semicontinuous, and bounded below}. Then, the following {statements hold}.
    \begin{enumerate}[label=(\roman*)]
        \item The set $\operatorname{prox}_{\tau r}(\bx)$ is nonempty and compact
            for any $\bx \in \mathbb{R}^{q}$ and $\tau > 0$.

        \item If $r$ is convex, then $\operatorname{prox}_{\tau r}(\bx)$
            contains exactly one value for any $\bx \in \mathbb{R}^{q}$ and
            $\tau > 0$.
    \end{enumerate}
\end{proposition}

\blue{A differentiable} function \blue{$g$}: $\RR^q \rightarrow \RR$ is said to be \blue{$L_g$}-smooth if $\| \blue{\nabla g}(\bx) - \blue{\nabla g}(\bx') \| \leq \blue{L_g} \| \bx - \bx'\|$ for all $\bx,\, \bx' \in \RR^q$.
A mapping $\mathcal{M} : \blue{\Omega} \to \RR^l$ is said to be $C$-Lipschitz continuous if $\|\mathcal{M}(\bx) - \mathcal{M}(\bx')\| \leq C\|\bx - \bx'\|$ for all $\bx, \bx' \in \blue{\Omega}$. For an \blue{$L_g$}-smooth function, the descent lemma \blue{\cite{Bertsekas1997}} states that
\begin{align}
\label{Eq: the descent lemma}
    \blue{g}(\bx) \leq \blue{g}(\bx') + \langle \blue{\nabla g}(\bx'), \bx - \bx' \rangle + \frac{\blue{L_g}}{2} \| \bx - \bx'\|^2.
\end{align}

\subsection{\blue{W-type Regularizer: Definition and Properties}}
\label{Section: W-regularizer}
% Deep hashing aims to learn network parameters \(\bx \in \RR^{\blue{q}}\) that generate binary codes preserving semantic similarity in the Hamming space. A direct formulation is
% \begin{align}
% \label{Eq:direct_deep_hashing_problem}
% \blue{\min_{\bx \in \RR^{q}}\ \frac{1}{n}\sum_{i=1}^{n}\ell_{s,i}\!\big(\operatorname{sgn}(\mathcal{D}(\bx))\big),}
% \end{align}
% where \(n\) is the number of training samples, \blue{\(\bD_i:\RR^q\to\RR^d\) is the continuous network output mapping for sample \(i\), \(\mathcal{D}(\bx):=[\bD_1(\bx)^\top;\ldots;\bD_n(\bx)^\top]\), and \(\ell_{s,i}:\RR^{n\times d}\to\RR\) is a similarity-preserving loss} (e.g., pairwise cross-entropy~\cite{zhuDeepHashingNetwork2016}). This objective is hard to optimize by backpropagation because \(\operatorname{sgn}(\cdot)\) is nondifferentiable and has zero gradient almost everywhere.

\begingroup\color{blue}
\red{To reduce the quantization error introduced by continuous relaxation}, penalties with W-shaped graphs, vanishing exactly at
$\pm1$, are added to drive the continuous outputs toward $\{-1,1\}^d$. Such penalties
are standard in deep hashing~\cite{liuDeepSupervisedHashing,yuanCentralSimilarityQuantization2020,he2025mambahash,
yin2026crh}. Their formulas differ, but our analysis depends only on their common
shape, which we formalize next.
\begin{definition}[W-type regularizer]\label{def:wtype}
Let \(\phi:\mathbb{R}\to[0,+\infty)\). For \(\bz\in\mathbb{R}^{d}\) and \(\bZ\in\mathbb{R}^{n\times d}\), define \(h(\bz):=\sum_{j=1}^{d}\phi(z_j)\) and \(h(\bZ):=\sum_{i=1}^{n}\sum_{j=1}^{d}\phi(z_{ij})\); with a slight abuse of notation, \(h\), \(h^*\), and \(h^{**}\) refer to both the vector and the matrix cases. We call \(h\) a \emph{W-type regularizer} if \(\phi\) satisfies:
\begin{enumerate}[label=\textup{(W\arabic*)}]
  \item\label{W1} \(\phi\) is even and continuous, and \(\phi(t)=0\) if and only if \(t\in\{-1,1\}\);
  \item\label{W2} \(\phi\) is convex on \([1,+\infty)\);
  \item\label{W3} \(\phi\) decreases at a nonvanishing rate on \([0,1)\): for every \(\red{\theta}\in(0,1)\) there exists \(\red{c_{\theta}}>0\) such that \(\phi(s)-\phi(t)\ge \red{c_{\theta}}(t-s)\) for all \(0\le s<t\le\red{\theta}\).
\end{enumerate}
\end{definition}
Condition~\ref{W1} fixes the W shape and, together with \ref{W2}, makes $\phi$ nondecreasing on $[1,+\infty)$, which determines the convex envelope and the conjugate below; \ref{W3} excludes flat pieces in $(-1,1)$, so that $\phi$ has no stationary point there, which yields the stationarity characterization in \Cref{prop:transfer}. We list representative W-type regularizers from the deep hashing literature together with their conjugates in \Cref{tab:wtype}. Throughout the paper, we refer to this as the $\ell_1$ W-regularizer, which is also the regularizer used in our experiments, defined by
\begin{equation}\label{eq:l1-w-regularizer}
h(\bz)=\lambda\bigl\||\bz|-\mathbf{1}\bigr\|_1.
\end{equation}\red{\Cref{fig:w-conj-prox} illustrates this regularizer.}
\begin{remark}
{The structural results in \Cref{Section: Methodology} hold for every W-type regularizer, whereas the algorithms, the complexity analysis, and the experiments use the $\ell_1$ W-regularizer, which has a \emph{compact dual domain}, i.e., $\operatorname{dom}\phi^*=[-\lambda,\lambda]$.}
\end{remark}

For general W-type regularizers, the convex envelope and the conjugate admit the following explicit description. The proof is provided in \Cref{app:wtype-conjugate-proof}.
\begin{lemma}\label{lem:wtype-conj}
Let \(\phi:\mathbb{R}\to[0,+\infty)\) satisfy \ref{W1}--\ref{W2}. Then it holds that
\begin{enumerate}[label=\textup{(\alph*)}]
\item \(\phi^{**}(t)=\phi\bigl(\max\{|t|,1\}\bigr)\) for all \(t\in\mathbb{R}\); in particular, \(\phi^{**}(t)=\phi(t)\) if and only if \(|t|\ge1\);
\item \(\phi^*\) is proper, lower semicontinuous, convex, and even; \(\phi^*(y)\ge|y|\) for all \(y\); and \(\operatorname{dom}\phi^*\) has nonempty interior.
\end{enumerate}
Consequently, \(h^*(\bLmbd)=\sum_{i,j}\phi^*(\Lambda_{ij})\), \(h^{**}(\bB)=\sum_{i,j}\phi^{**}(b_{ij})\), and \(h(\bB)=h^{**}(\bB)\) if and only if \(|b_{ij}|\ge1\) for all \(i,j\).
\end{lemma}

\endgroup

\subsection{W-type Regularized Deep Hashing Model}
\label{Sec:W-type Regularized Deep Hashing Model}
With the W-type regularizer \(h\) \blue{defined in \Cref{def:wtype}}, the model is now trained to produce the \(i\)-th continuous network output \(\bD_i(\bx)\), and binary codes are obtained post-training via \(\operatorname{sgn}(\bD_i(\bx))\in\{-1,1\}^d.\) Consequently, we formulate the following core optimization problem for W-type regularized deep hashing methods: % [2026-10-25] removed ``which serves as a continuous surrogate of Problem (direct form)''
\begin{align}
\label{Eq: the primal deep hashing problem}
\min_{\bx \in \RR^{\blue{q}}} \quad \blue{J_0(\bx):=}\frac{1}{n}\sum_{i=1}^n f_i(\bx) + \sum_{i=1}^n h(\bD_i(\bx)),
\tag{\blue{P0}}
\end{align}where
\blue{\(f_i(\bx):=\ell_{s,i}\!\left(\mathcal{D}(\bx)\right)+\frac{\mu}{2}\|\bx\|^2\) with \(\mu>0\)\red{, and \(\ell_{s,i}:\RR^{n\times d}\to\RR\) is a similarity-preserving loss (e.g., pairwise cross-entropy~\cite{zhuDeepHashingNetwork2016})}.}

\blue{Following~\cite{bai2019proxquant}, we define the $\ell_1$ quantization error in deep hashing problems as follows:
\begin{equation}
\label{eq:quantization-error}
Q_1(\bx):=\bigl\|\mathcal{D}(\bx)-\operatorname{sgn}(\mathcal{D}(\bx))\bigr\|_1
=\sum_{i=1}^{n}\sum_{j=1}^{d}\bigl||\mathcal{D}_{ij}(\bx)|-1\bigr|,
\end{equation}{where the equality follows componentwise from $\min_{b\in\{-1,1\}}|z-b|=\bigl||z|-1\bigr|$. The last column of \Cref{tab:wtype} relates each regularizer to $Q_1$: the $\ell_1$ W-regularizer gives $h(\mathcal{D}(\bx))=\lambda Q_1(\bx)$, whereas the others give only upper bounds of the form $h(\mathcal{D}(\bx))\le c\,Q_1(\bx)$.}}

Using variable splitting, we introduce continuous auxiliary variables \(\bb_i\in\RR^d\) and \(\bB=[\bb_1^\top;\ldots;\bb_n^\top]\in\RR^{n\times d}\), subject to \(\bb_i=\bD_i(\bx)\) for all \(i=1,\dots,n\), to obtain the following equivalent nonlinearly constrained problem: 
\begin{align}
    \min_{\bx \in \RR^{q},\ \bB \in \RR^{n\times d}}\quad \frac{1}{n}\sum_{i=1}^{n} f_i(\bx)+\sum_{i=1}^{n}h(\bb_i)\quad \text{s.t.}\quad \bb_i=\bD_i(\bx),\ \forall i=1,\dots,n. \notag
\end{align}
Let \(\gamma>0\) be a penalty parameter and define
\begin{align}
    F(\bx,\bB):= \frac{1}{n}\sum_{i=1}^{n}F_i(\bx,\bb_i),\qquad F_i(\bx,\bb_i):= f_i(\bx)+\frac{\gamma}{2}\|\bD_i(\bx)-\bb_i\|^2. \notag
\end{align}
Averaging over \(i\) gives the decomposition \(F=f+P\), where
\begin{align}
    f(\bx):= \frac{1}{n}\sum_{i=1}^{n}f_i(\bx),\qquad P(\bx,\bB):= \frac{\gamma}{2n}\sum_{i=1}^{n}\|\bD_i(\bx)-\bb_i\|^2. \notag
\end{align}
\blue{With the scaled penalty parameter $m:=\gamma/n$, we have $P(\bx,\bB)=\frac{m}{2}\|\mathcal{D}(\bx)-\bB\|^2$.}
We then relax these constraints by the quadratic penalty method~\cite[Sec.~17.1]{nocedalWright2006} to obtain the following two-block finite-sum optimization problem:
\begin{align}
    \label{Prop: block-wise regularized problem}
    \min_{\bx \in \RR^{q},\,\bB \in \RR^{n\times d}}\ J(\bx,\bB):=F(\bx,\bB)+h(\bB). \tag{P}
\end{align} This reformulation decomposes the original problem \eqref{Eq: the primal deep hashing problem} into simpler structural blocks, enabling us to exploit the problem structure better. Building upon Problem~\eqref{Prop: block-wise regularized problem}, we develop a stochastic primal-dual algorithm that addresses the lack of tractable proximal mappings in the nonsmooth composite form and the absence of convergence guarantees in existing deep hashing methods.
\subsection{Basic Assumptions}
We impose the following basic assumptions on Problem \eqref{Eq: the primal deep hashing problem}.
\begin{assumption}
\label{Ass: basic regularity}
\leavevmode
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2em]
    \item The average loss is \blue{bounded below: $\underline{\ell}:=\inf_{\bx}\bar{\ell}_{s}\!\left(\mathcal{D}(\bx)\right)>-\infty$} where \(\bar{\ell}_s:=n^{-1}\sum_{i=1}^{n}\ell_{s,i}\).
    \item \blue{The function $f$ is $L_f$-smooth on $\RR^q$}, and $\nabla\mathcal{D}$ is $L_{\nabla\mathcal{D}}$-Lipschitz on $\RR^q$, with $L_f,L_{\nabla\mathcal{D}}>0$. There exist constants \blue{$C_{\mathcal{D}},G_{\mathcal{D}}>0$ such that
    \[
        \|\mathcal{D}(\bx)\|\le C_{\mathcal{D}},\qquad
        \|\nabla\mathcal{D}(\bx)\|\le G_{\mathcal{D}},\qquad
        \forall\,\bx\in\RR^q.
    \]}
\end{enumerate}
\end{assumption}

\begin{remark}\label{rem:coercivity}
\begingroup\color{blue}
Assumption~\ref{Ass: basic regularity} and $\mu>0$ make $J$ coercive, and thus Problem~\eqref{Prop: block-wise regularized problem} has an optimal solution. Specifically, since $h\ge0$ and $\|\mathcal{D}(\bx)\|\le C_{\mathcal{D}}$, we have
\[
J(\bx,\bB)\ge\underline{\ell}-\frac{\gamma}{2n}C_{\mathcal{D}}^2+\frac{\mu}{2}\|\bx\|^2+\frac{\gamma}{4n}\|\bB\|^2.
\]
\endgroup
\end{remark}
