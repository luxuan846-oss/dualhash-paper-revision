\section{Stochastic Primal-Dual Hashing Algorithms}
\label{Section: Methodology}
\subsection{Algorithmic Framework}
Let us consider the equivalent constrained formulation of Problem \eqref{Prop: block-wise regularized problem}:
\begin{align} 
\min_{\bx \in \RR^{\blue{q}}, \bB \in \RR^{n\times d}, \bY \in \RR^{n\times d}} &\quad F(\bx, \bB) + h(\bY) \notag \\
\text{s.t. } &\quad \bY = \bB \nonumber ,
\end{align} whose corresponding dual problem is 
\begin{align}
\label{Prop: the dual problem}
\max_{\bLmbd \in \RR^{n\times d}} \min_{\bx \in \RR^{\blue{q}}, \bB \in \RR^{n\times d}, \bY \in \RR^{n\times d}} &\quad F(\bx, \bB) + h(\bY) + \langle \bLmbd, \bB - \bY \rangle. \tag{D}
\end{align} 
\blue{Recalling the definition of the conjugate function $h^*$} and observing that the minimization with respect to $\bY$ in Problem \eqref{Prop: the dual problem} can be performed independently, we then simplify the Lagrangian function in the problem \eqref{Prop: the dual problem} to 
\begin{equation}\label{Eq: simplified_lagrangian}
\calL(\bx,\bB,\bLmbd):=\Phi(\bx,\bB,\bLmbd)-h^*(\bLmbd),
\qquad\text{where}\quad
\Phi(\bx,\bB,\bLmbd):=F(\bx,\bB)+\langle\bB,\bLmbd\rangle.
\end{equation}
Through this transformation, we replace $h(\bY)$ with $-h^*(\bLmbd)$ and optimize the \(h^*\) in the dual space. The convexity of $h^*$ makes $\operatorname{prox}_{h^*}(\cdot)$ often much easier to compute (potentially having a closed-form solution) than $\operatorname{prox}_{h}(\cdot)$.
For the W-type regularization on the matrix variable $\bB \in \RR^{n \times d}$,
\begin{align*}
&h(\bB)= \lambda\||\bB|-\bm{1}\|_1 = \sum_{i=1}^{n}\sum_{j=1}^{d}\lambda\big||b_{ij}|-1\big|,
\end{align*}
its conjugate is separable element-wise:
\begin{align*}
&h^*(\bLmbd) = \sum_{i=1}^{n}\sum_{j=1}^{d} h^*(\Lambda_{ij}),\,\, \text{where } h^*(\Lambda_{ij}) = \begin{cases}
    |\Lambda_{ij}|, & \text{if } \Lambda_{ij} \in [-\lambda, \lambda],\\
    +\infty, & \text{otherwise},
    \end{cases}
\end{align*} whose proximal operator in the piecewise solution (see \Cref{fig:w-conj-prox} for details) is 
\begin{align}
    \label{Eq: the proximal mapping of conjugate function}
    [\operatorname{prox}_{\tau h^*}(\bY)]_{ij} = 
    \begin{cases}
        \lambda, & Y_{ij} > \lambda +\tau,\\
        Y_{ij} -\tau, & \tau < Y_{ij} \leq \lambda +\tau, \\
        0, & -\tau \leq Y_{ij} \leq\tau, \\
        Y_{ij} +\tau, &  - \lambda -\tau \leq Y_{ij} < -\tau, \\
        -\lambda, & Y_{ij} < -\lambda -\tau.
    \end{cases}
    %  \notag
\end{align}
This motivates us to perform a proximal step to update \(\bLmbd\).

We now present the \textbf{DualHash} framework, which updates $\bx$, $\bB$, and $\bm{\Lambda}$ alternately. The resulting algorithm can be viewed as a first-order multiblock primal--dual method. First, we update $\bx$ by a stochastic first-order step, as is standard in training neural networks. \blue{At iteration $k$, the $\bx$-update takes the form
\begin{equation}
    \label{Eq: the generic update of x}
    \bx^{k+1} = \by^k - \eta_k \bg^k,
\end{equation}
where $\by^k$ is the base point of the step and $\bg^k$ is a stochastic estimator of $\nabla_{\bx}F$ at a query point.
The framework accommodates different stochastic estimators $\bg^k$, including momentum-based and variance-reduced ones, which may improve both the complexity guarantees and the practical performance; the choices of $\by^k$ and $\bg^k$ for each implementation are specified in Section~\ref{Sec: Algorithm implementations}.} \blue{Second, $\bB$ is updated by a gradient descent step on $\calL$; once $\mathcal{D}(\bx^{k+1})$ is available, this step is evaluated from the closed-form expression of $\nabla_{\bB}F$.} Finally, the dual variable $\bLmbd$ is updated by a proximal step on $h^*$ at the extrapolated point $2\bB^{k+1}-\bB^k$, following \cite{chambolleFirstOrderPrimalDualAlgorithm2011}:
\begin{align}
\bB^{k+1} &= \bB^k - \tau \nabla_{\bB}\calL(\bx^{k+1},\bB^k,\bLmbd^k) \notag\\
&= \bB^k - \tau\big(\nabla_{\bB}F(\bx^{k+1},\bB^k)+\bLmbd^k\big), \label{Eq: the update of B}\\
\bLmbd^{k+1} &= \operatorname*{arg\,min}_{\bLmbd}\Big\{h^*(\bLmbd)-\langle\bLmbd,2\bB^{k+1}-\bB^k\rangle+\frac{\tau}{2}\|\bLmbd-\bLmbd^k\|^2\Big\} \notag\\
&= \operatorname{prox}_{\tau^{-1}h^*}\big(\bLmbd^k+\tau^{-1}(2\bB^{k+1}-\bB^k)\big). \label{Eq: the update of lambda}
\end{align}

\subsection{\blue{Instantiating the Stochastic Update}}
\label{Sec: Algorithm implementations}
\blue{We next present two instances of the stochastic update~\eqref{Eq: the generic update of x}, which differ in the momentum technique and in the point at which the stochastic gradient is queried. Throughout, $i\in\{1,\ldots,n\}$ indexes training samples and $k$ indexes iterations. Since computing the full gradient $\nabla_{\bx} F(\bx,\bB)$ is expensive when the number of training samples $n$ is large, as is typical in deep learning, both instances replace it with a mini-batch estimator. At iteration $k$, we draw $\calJ_k$ uniformly from all subsets of $\{1,\ldots,n\}$ of size $b_k$, independently of the past. In both instances, the batch size is constant: $b_k\equiv b$ for DualHash-StoM, and $b_1$ followed by $b_k\equiv b$ for $k\ge2$ for DualHash-StoRM; their values are specified in \Cref{Section: Convergence analysis}. For any point $(\bx,\bB)$ determined before $\calJ_k$ is drawn, define
\begin{equation}
\label{Eq: minibatch oracle}
\tilde{\nabla}_{\bx}F(\bx,\bB;\calJ_k):=\frac{1}{b_k}\sum_{i\in\calJ_k}\nabla_{\bx}F_i(\bx,\bb_i),
\end{equation}}
\subsubsection{DualHash with Momentum Acceleration}
We first present a momentum-based instance. At iteration $k$, we form two extrapolated points
\begin{equation}
\label{Eq: extrapolated points in SGDM}
\by^k=\bx^k+\alpha_k(\bx^k-\bx^{k-1}),\qquad
\bz^k=\bx^k+\beta_k(\bx^k-\bx^{k-1}),
\end{equation}
where $\by^k$ is the base point of the step and $\bz^k$ is the query point of the oracle. Since $(\bz^k,\bB^k)$ is determined before $\calJ_k$ is drawn, \blue{calling \eqref{Eq: minibatch oracle} at this point gives an unbiased estimator of $\nabla_{\bx}F(\bz^k,\bB^k)$:}
\begin{equation}
\label{Eq: the stochastic approximate in SGDM}
\blue{\bg_m^k} := \tilde{\nabla}_{\bx}F(\bz^k,\bB^k;\calJ_k).
\end{equation}
\blue{Taking $\bg^k=\bg_m^k$ in \eqref{Eq: the generic update of x} yields the $\bx$-update}
\begin{equation}
\label{Eq: the update of x in SGDM}
\bx^{k+1} = \by^k - \eta_k \blue{\bg_m^k}.
\end{equation}
The overall algorithm, DualHash-StoM, is summarized in \Cref{alg:our_algorithm_sgdm}.

\begin{remark}
The momentum acceleration framework in \eqref{Eq: the update of x in SGDM} covers these two accelerations: Nesterov acceleration  \cite{Nesterov1983AMF} corresponds to $\alpha_k = \beta_k \neq 0$; the heavy-ball method \cite{Polyak1964} is obtained with $\beta_k = 0$; vanilla SGD is recovered when $\alpha_k = \beta_k = 0$. In this paper, we set $\alpha_k, \beta_k \in (0,1)$ for all $k \geq 1$.
\end{remark}
\begin{algorithm}[t]
\caption{DualHash-StoM}
\label{alg:our_algorithm_sgdm}
\begin{algorithmic}[1]
\STATE \textbf{Input:} \blue{$\bx^0=\bx^1\in\RR^{q}$}, $\bB^1\in\RR^{n\times d}$, \blue{$\bLmbd^1\in[-\lambda,\lambda]^{n\times d}$}, parameters {$\{\alpha_k\},\{\beta_k\}\subset[0,1)$, $\{\eta_k\}$ and $\tau$, number of iterations $T$}
\FOR{\blue{$k = 1, \ldots, T$}}
  \STATE \blue{Compute $\by^k$ and $\bz^k$ by \eqref{Eq: extrapolated points in SGDM}}
  \STATE \blue{Draw $\calJ_k$ and compute $\bg_m^k$ by \eqref{Eq: the stochastic approximate in SGDM}}
  \STATE Update $\bx^{k+1}$ by
  \[
  \bx^{k+1} = \by^k - \eta_k \blue{\bg_m^k}
  \]
  \STATE Update $\bB^{k+1}$ by
  \[
  \bB^{k+1} = \bB^k - \tau \nabla_{\bB}\calL(\bx^{k+1}, \bB^k, \bLmbd^k)
  \]
  \STATE Update $\bLmbd^{k+1}$ by
  \[
  \bLmbd^{k+1} = \operatorname*{arg\,min}_{\bLmbd} \Big\{ h^*(\bLmbd) - \langle \bLmbd, 2\bB^{k+1} - \bB^k \rangle + \frac{\tau}{2} \| \bLmbd - \bLmbd^k\|^2 \Big\}
  \]
\ENDFOR
\STATE \textbf{Output:} \blue{$\bx^R$ with $R$ drawn uniformly from $\{1,\ldots,T\}$}
\end{algorithmic}
\end{algorithm}

\subsubsection{DualHash with Recursive Momentum}
Alternatively, we consider the StoRM estimator for the $\bx$-update, which achieves variance reduction without requiring full-gradient or large-batch gradient computations. Given its practical effectiveness in deep learning \cite{levy2021storm+}, we adapt the generic StoRM estimator in \eqref{Eq: the generic StoRM estimator} to construct a mini-batch recursive momentum estimator \blue{$\bg_s^k$ of $\nabla_{\bx} F(\bx^k, \bB^k)$, calling the oracle \eqref{Eq: minibatch oracle} with the same mini-batch $\calJ_k$ at two query points:}
\begin{equation}
\label{Eq: the stochastic approximate in STORM}
\blue{\bg_s^k} :=
\begin{cases} 
\tilde{\nabla}_{\bx} F(\bx^1, \bB^1; \calJ_1), &\text{if }  k = 1, \\
(1 - \rho_{k-1}) \big( \blue{\bg_s^{k-1}} - \tilde{\nabla}_{\bx} F(\bx^{k-1}, \bB^{k-1};\blue{\calJ_k})\big)+\tilde{\nabla}_{\bx} F(\bx^k, \bB^k; \calJ_k), & \text{if } k > 1.
\end{cases}
\end{equation}
\blue{Taking $\by^k=\bx^k$ and $\bg^k=\bg_s^k$ in \eqref{Eq: the generic update of x} yields the $\bx$-update}
\begin{equation}
\label{Eq: the update of x in STORM}
    \bx^{k+1} = \bx^k - \eta_k \blue{\bg_s^k}.
\end{equation}
The overall algorithm, DualHash-StoRM, is summarized in \Cref{alg:our_algorithm_storm}.
