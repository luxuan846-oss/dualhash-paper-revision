\section{Complexity analysis}
\label{Section: Convergence analysis}
We now analyze the complexity of the proposed algorithms in terms of the number of stochastic first-order oracle (SFO) calls required to find an $\varepsilon$-approximate stochastic critical point of the Lagrangian function \eqref{Eq: simplified_lagrangian}. Specifically, we seek a point satisfying 
\begin{equation}\label{equ:esp}
\mathbb{E} [\text{dist}^2(\bm{0}, \partial \calL (\bx^*, \bm{B}^*, \bm{\Lambda}^* ))] \leq \varepsilon^2
\end{equation}
for a given \(\varepsilon > 0\). This focus is standard in non-convex stochastic first-order optimization analysis, as finding a global or even local minimizer for non-convex optimization problems is generally NP-hard. We limit the maximum number of iterations to a fixed integer, denoted by $T$ with a little abuse of notation. \blue{Under the mini-batch sampling scheme introduced in Section~\ref{Sec: Algorithm implementations}, let $(\Omega,\mathcal F,\mathbb P)$ be the underlying probability space. For $k\geq1$, define $\mathcal F_k:=\sigma(\mathcal J_1,\ldots,\mathcal J_{k-1})$ with $\mathcal F_1$ trivial, as the filtration generated by the past mini-batches. We write $\mathbb E_k[\cdot]:=\mathbb E[\cdot\mid\mathcal F_k]$
for the conditional expectation given the past and $\mathbb E[\cdot]$ for the total expectation. All equalities and inequalities involving $\mathbb E_k[\cdot]$ are understood to hold almost surely, and we explicitly indicate ``a.s.'' whenever such conditional relations are stated.}

In the remainder of this paper, we impose the following assumptions.
\begin{assumption} 
\label{Ass: main assumptions}
\begin{itemize}[leftmargin=2em]
    \item[{(i)}] \blue{The average loss is bounded below: $\underline{\ell}:=\inf_{\bx}\bar{\ell}_{s}(\bx)>-\infty$, where $\bar{\ell}_{s}(\bx):=n^{-1}\sum_{i=1}^{n}\ell_{s,i}(\bx)$;}
    \item[{(ii)}] \blue{$F$ is $L_F$-smooth, with $L_F\in(0,+\infty)$, over $\RR^{q}\times\RR^{n\times d}$;}
    \item[{(iii)}] \blue{For any $\bB \in \RR^{n\times d}$, there exists a constant $\sigma^2 \geq 0$ such that
    \begin{align}
        \label{Eq: finite-sum sampling identities}
        &\frac{1}{n}\sum_{i=1}^{n}\nabla_{\bx} F_i(\bx, \bb_i)
        = \nabla_{\bx} F(\bx, \bB),\\
        &\frac{1}{n}\sum_{i=1}^{n}
        \left\|\nabla_{\bx} F_i(\bx, \bb_i)
        - \nabla_{\bx} F(\bx, \bB)\right\|^2
        \leq \sigma^2.
    \end{align}}
    \item[{(iv)}] \blue{The critical set}
    \[
    \blue{\mathrm{crit}\,\calL := \{(\bx^*, \bB^*, \bLmbd^*) \in \RR^{q} \times \RR^{n\times d} \times \RR^{n\times d}: \bm{0} \in \partial\calL(\bx^*, \bB^*, \bLmbd^*)\}}
    \]
    \blue{is nonempty.}
\end{itemize}
\end{assumption}
\begin{remark}
\blue{The smoothness constant $L_F$ is bounded by a constant independent of the sample size $n$,} as the penalty formulation cancels the potential
\blue{$\sqrt{n}$ scaling of the stacked quantities.} A rigorous analysis is in
Appendix~\ref{Appendix: discussion on assumptions}.
\end{remark}
\blue{We first record two basic facts: the model structure makes $J$ coercive and bounded below, and the dual iterates stay in a fixed box. The analysis below does not require the primal iterates $(\bx^k,\bB^k)$ to be bounded.}
{\begin{lemma}
\label{lem:basic-properties}
\blue{Under \Cref{Ass: main assumptions}(i), the following hold.
\begin{enumerate}[label=(\alph*),nosep]
\item For all $(\bx,\bB)$, $J(\bx,\bB)\ge \underline{\ell}+\frac{\mu}{2}\|\bx\|^{2}+\lambda\|\bB\|_{1}-\lambda nd$; in particular, $J$ is bounded below by $\underline{\ell}-\lambda nd$ and coercive.
\item For the sequence generated by \Cref{alg:our_algorithm_sgdm} or \Cref{alg:our_algorithm_storm}, $\bm{\Lambda}^{k}\in[-\lambda,\lambda]^{n\times d}$ for all $k\ge 2$; hence $\{\bm{\Lambda}^{k}\}$ is bounded.
\end{enumerate}}
\end{lemma}
\setcounter{proof}{0}
\begin{proof}
\blue{(a) Since $f_i=\ell_{s,i}+\frac{\mu}{2}\|\cdot\|^{2}$ with $\mu>0$, we have $\frac1n\sum_{i}f_i(\bx)\ge\underline{\ell}+\frac{\mu}{2}\|\bx\|^{2}$. The penalty term $P(\bx,\bB)$ is nonnegative, and $h(\bB)=\lambda\sum_{i,j}\bigl||B_{ij}|-1\bigr|\ge\lambda\|\bB\|_{1}-\lambda nd$. Adding the three bounds gives the inequality, whose right-hand side tends to $+\infty$ as $\|(\bx,\bB)\|\to\infty$.}

(b) By the closed-form proximal mapping in \eqref{Eq: the proximal mapping of conjugate function}, \(\bm{\Lambda}^{k+1}\in[-\lambda,\lambda]^{n\times d}\) for every $k\ge1$.
\end{proof}}
We then prove that any primal-dual optimal solution $(\bx^*, \bB^*, \bLmbd^*)$ satisfies the first-order necessary optimality condition of the Lagrangian~\eqref{Eq: simplified_lagrangian}, i.e., it is a critical point of $\calL$.
\begin{lemma}
\label{Lemma: Optimality condition}
    Under \Cref{Ass: main assumptions}(i), suppose that\((\bx^*, \bB^*, \bLmbd^*)\) are the optimal solutions of Problem \eqref{Prop: block-wise regularized problem} and Problem \eqref{Prop: the dual problem}, respectively. Then, it holds that \((\bx^*, \bB^*, \bLmbd^*) \in \text{crit} \calL\), i.e., \(\bm{0} \in \partial \mathcal{L}(\bx^*, \bB^*, \bLmbd^*)\).
\end{lemma} 
\begin{proof}
    First, due to the coercivity of $\calL(\bx, \bB, \bLmbd)$ , we have \(\min\limits_{\bx, \bB} \calL(\bx, \bB, \bLmbd) > -\infty\). According to \cite[Theorem 2.158]{temporary-citekey-11039},
    $(\bx^{*}, \bm{B}^{*}, \bm{\Lambda}^{*})$ are optimal solutions to the
    primal problem \eqref{Prop: block-wise regularized problem} and the dual problem
    \eqref{Prop: the dual problem} if and only if:
    \begin{align}
        \label{Eq: the sufficient and necessary condition}
        \begin{cases}
        (\bx^{*}, \bm{B}^{*}) \in \argmin\limits_{\bx \in \RR^{\blue{q}}, \bm{B} \in \RR^{n\times d}}\mathcal{L}(\bx^{*}, \bm{B}^{*}, \bm{\Lambda}^{*}), \\
        0 = h(\bm{B}^{*}) + h^{*}(\bm{\Lambda}^{*}) - \langle \bm{\Lambda}^{*}, \bm{B}^{*}\rangle.
        \end{cases}
    \end{align}

    The first condition in \eqref{Eq: the sufficient and necessary condition} implies
    that $\bm{0}$ belongs to the subdifferential of $\mathcal{L}$ with respect to
    $(\bx, \bm{B})$ at $(\bx^{*}, \bm{B}^{*}, \bm{\Lambda}^{*})$. Since $\calL$
    is continuously differentiable in $(\bx, \bm{B})$ \blue{and $F=f+P$ by \eqref{Prop: block-wise regularized problem}}, we have
    \begin{align}
        \label{Eq: the sufficient and necessary condition for primal}
        \begin{cases}
        \blue{\bm{0} = \nabla_{\bx}F(\bx^{*}, \bm{B}^{*}),}\\ 
        \blue{\bm{0} = \nabla_{\bB}F(\bx^{*}, \bm{B}^{*}) + \bm{\Lambda}^{*}.}
        \end{cases}
    \end{align}
    
    For the second condition in \eqref{Eq: the sufficient and necessary condition},
    by the definition of the conjugate function, if this equality holds, then $\bm{\Lambda}^{*}$ is the optimal solution of $\sup_{\bLmbd} \{ \langle \bB^{*}, \bLmbd \rangle - h^{*}(\bLmbd) \}$, i.e.,
    \begin{align}
        \label{Eq: the sufficient and necessary condition for dual}\bm{B}^{*}\in \partial h^{*}(\bm{\Lambda}^{*}).
    \end{align}
    Combining \eqref{Eq: the sufficient and necessary condition for primal} and \eqref{Eq: the sufficient and necessary condition for dual}, we obtain $\bm{0}\in \partial \mathcal{L}(\bx^{*}, \bm{B}^{*}, \bm{\bLmbd}^{*})$, completing the proof. 
    % 
\end{proof}

Our analysis aims to bound $\EE[\|\partial \calL\|^2]$. The presence of dual variables $\bLmbd$ \blue{makes it difficult to establish a descent property for the Lagrangian $\calL$ in \eqref{Eq: simplified_lagrangian}: the dual update can increase $\calL$, so $\calL$ need not decrease along the iterates, which precludes} the direct use of standard methods \cite{xu2023momentum,zhang2022stochastic}. 
To this end, we construct suitable Lyapunov functions (see \eqref{Eq: the Lyapunov function in SGDM} and \eqref{Eq: the Lyapunov function in STORM}) to obtain the approximate descent property.  We first analyze the dual dynamics by establishing bounds on $\|\bLmbd^{k+1}-\bLmbd^k\|^2$ in terms of the primal residuals, enabling the Lyapunov construction.
\begin{lemma}
\label{Lemma: dual boundedness}
Under \Cref{Ass: main assumptions}, for any $k \geq 1$, it holds that
    \begin{align}
        &\|\bm{\Lambda}^{k+1}- \bm{\Lambda}^k \|^2 \notag \\
        \leq&\,3 \tau^{-2} \|\bB^{k+2} - \bB^{k+1}\|^2 + 3(\tau^{-1} + L_F)^2  \|\bB^{k+1} - \bB^{k}\|^2 + 3L_F^2  \|\bx^{k+2} - \bx^{k+1}\|^2.
         \label{Eq: dual boundedness}
    \end{align}
\end{lemma}
\begin{proof}
    Using the update of $\bm{B}^{k}$ in \eqref{Eq: the update of B} twice yields that
\begin{align*}
    &\|\bm{\Lambda}^{k+1}- \bm{\Lambda}^{k}\| \notag \\
    =&\, \left\| \tau^{-1}(\bm{B}^{k+1}- \bm{B}^{k+2}) + \tau^{-1}(\bm{B}^{k+1}- \bm{B}^{k}) + \nabla_{\bB}F(\bx^{k+1}, \bm{B}^{k}) - \nabla_{\bB}F(\bx^{k+2}, \bm{B}^{k+1}) \right\| \\
    \leq&\, \tau^{-1}\|(\bm{B}^{k+1}- \bm{B}^{k+2})\| + (\tau^{-1}+ L_{F}) \|\bm{B}^{k+1}- \bm{B}^{k}\| + L_{F}\|\bx^{k+2}- \bx^{k+1}\|,
\end{align*}
    where the second inequality uses the smoothness of $F(\bx, \bm{B})$ in \Cref{Ass: main assumptions}. Therefore, using the fact $(a + b + c)^{2}\leq 3(a^{2}+ b^{2}+c^{2})$, one has
    \begin{align}
         &\|\bm{\Lambda}^{k+1}- \bm{\Lambda}^{k}\|^{2} \notag \\
         \leq&\, 3\tau^{-2}\|\bm{B}^{k+2}- \bm{B}^{k+1}\|^{2}+ 3(\tau^{-1}+ L_{F})^{2}\|\bm{B}^{k+1}- \bm{B}^{k}\|^{2} + 3L_{F}^{2}\|\bx^{k+2}- \bx^{k+1}\|^{2},\label{Eq: dual boundedness in proof}
    \end{align}
    and completes the proof. 
\end{proof}

Next, we analyze how the Lagrangian function \eqref{Eq: simplified_lagrangian} changes
during the updates of primal and dual variables in one iteration. The total change can be decomposed as:
\begin{align}
    \label{Eq: One-iteration progress decomposition} & \calL(\bx^{k+1}, \bB^{k+1}, \bLmbd^{k+1}) - \calL(\bx^{k}, \bB^{k}, \bLmbd^{k}) \notag                                                                                                                                                                                             \\
    =                                                & \,\underbrace{\calL(\bx^{k+1}, \bB^{k+1}, \bLmbd^{k+1}) - \calL(\bx^{k+1}, \bB^k, \bLmbd^k)}_{\text{one-iteration progress on $\bB$ and $\bLmbd$}}+ \underbrace{\calL(\bx^{k+1}, \bB^{k}, \bLmbd^{k}) - \calL(\bx^{k}, \bB^k, \bLmbd^k)}_{\text{one-iteration progress on $\bx$}}.
\end{align}
Since the updates of $\bB$ and $\bLmbd$ are common to Algorithms~\ref{alg:our_algorithm_sgdm} and~\ref{alg:our_algorithm_storm}, the following lemma bounds their one-step change. 
% We then analyze separately how the $\bx$-update affects the Lagrangian function. 
For notational simplicity, we define
\[
    \Delta_{\bx}^{k}= \frac{\|\bx^{k}- \bx^{k-1}\|^{2}}{2},\quad \Delta_{\bm{B}}^{k}
    = \frac{\|\bm{B}^{k}- \bm{B}^{k-1}\|^{2}}{2}.
\] 
\begin{lemma}[One-iteration progress on $\bB$ and $\bLmbd$]
    \label{Lemma: One-iteration progress of B and Lambda}
    Consider the sequence $\{
    \bx^{k}, \bB^{k}, \bm{\Lambda}^{k}\}_{k \in \NN}$ generated by \Cref{alg:our_algorithm_sgdm}
    or \Cref{alg:our_algorithm_storm}. Under \Cref{Ass: main assumptions}(ii), for any $k \geq 1$, it holds that
    \begin{align}
\label{item: B and Lambda}
&\calL(\bx^{k+1}, \bB^{k+1}, \bLmbd^{k+1}) - \calL(\bx^{k+1}, \bB^{k}, \bLmbd^{k}) \notag \\
\le& K_1 \Delta_{\bB}^{k+2} - K_2 \Delta_{\bB}^{k+1} + K_3 \Delta_{\bB}^{k} 
+ K_4 \Delta_{\bx}^{k+2} + K_3 \Delta_{\bx}^{k+1}, 
\end{align} where \(K_1 = 3\delta\tau^{-1}\), \(K_2 = 2\tau^{-1} - L_F - 3\delta\tau(\tau^{-1} + L_F)^2\), \(K_3 = \frac{\tau L_{F}^{2}}{\delta}\), and \(K_4 = 3\delta\tau L_F^2\).
\end{lemma}
The proof is provided in the theoretical appendix.

We now analyze the stochastic $\bx$-updates, which introduces algorithm-specific challenges:
as \eqref{Eq: dual boundedness} and \eqref{Eq: the inner product in the recursive relation of L on B and Lambda} show,
randomness in the $\bx$-updates propagates to the $\bB$- and $\bLmbd$-updates, necessitating careful parameter tuning and the construction of Lyapunov functions. We present the key lemmas and complexity results of both algorithms below\footnote{Detailed proofs of all lemmas and theorems, along with complete parameter derivations,
are given in Appendices~\ref{Appendix: the complexity analysis of DualHash-StoM} and~\ref{Appendix: the complexity analysis of DualHash-StoRM}.}
\subsection{Oracle complexity of DualHash-StoM}
Let us denote the gradient error in the $k$-th iteration as
\[
\be^{k} = \blue{\bg_m^{k}} - \nabla_{\bx}F(\bz^{k}, \bB^{k}),
\]where $\blue{\bg_m^k}$ is defined in \eqref{Eq: the stochastic approximate in SGDM}. Under Assumption~\ref{Ass: main assumptions}(iii), one has \blue{
\begin{align}
    \label{eq: the gradient error in SGDM}
    \mathbb{E}\!\left[\|\be^k\|^2\right]
    \leq \frac{\sigma^2}{b_k}.
\end{align} with the detailed derivation provided in \Cref{Subsec: proof gradient error SGDM}.}

The one-iteration progress on $\bx$ is estimated as follows, with the proof provided in \Cref{Subsubsec: proof x progress SGDM}.
\begin{lemma}[One-iteration progress on $\bx$]
    \label{Lemma: One-iteration progress of x in SGDM}
    Consider the sequence $\{\bx^k, \bB^k, \bm{\Lambda}^k\}_{k \in \NN}$ generated by \Cref{alg:our_algorithm_sgdm}. Under \Cref{Ass: main assumptions}(ii) and the parameters in \eqref{Eq: the parameters conditions of dualhash-sgdm}, for any $k \geq 1$, it holds that
    \begin{align}
        \label{Eq: One-iteration progress of x in SGDM}
        \calL (\bx^{k+1}, \bB^{k}, \bLmbd^{k}) - \calL (\bx^{k}, \bB^{k}, \bLmbd^{k})
        \leq -K_5\Delta_{\bx}^{k+1}+K_6\Delta_{\bx}^{k}+ \frac{1}{L_{F}}\|\be^{k}\|^{2},
    \end{align}
    where \(K_5 = \frac{1 - \alpha_{k}}{\eta_{k}}- 2L_F\) and \(K_6 = \frac{\alpha_{k}}{\eta_{k}}+ 2L_F\beta_k^2\).
\end{lemma}

To handle the \blue{oscillations of the Lagrangian function $\calL$ in} \eqref{Eq: simplified_lagrangian}, we introduce the following Lyapunov function and sequence to establish an in-expectation approximate sufficient descent property for \Cref{alg:our_algorithm_sgdm}:
\begin{align}
    \Psi_m(\mathcal{Q}^{k}) &= \mathcal{L}(\bx^{k}, \bm{B}^{k}, \bLmbd^{k}) 
        - C_{1}\Delta_{\bB}^{k+1} + C_{2}\Delta_{\bB}^{k} 
        - C_{3}\Delta_{\bm{x}}^{k+1} + C_{4}\Delta_{\bm{x}}^{k}, 
        \label{Eq: the Lyapunov function in SGDM} \\
    \mathcal{Q}^{k} &:= \{\bx^{k}, \bB^{k}, \bm{\Lambda}^{k}, 
        \bB^{k+1}, \bB^{k-1}, \bx^{k+1}, \bx^{k-1}\}, 
        \quad \{\mathcal{Q}^{k}\}_{k \in \mathbb{N}}, \notag
        % \label{Eq: the Lyapunov sequence in SGDM}
\end{align}where the positive coefficients $C_1, C_2, C_3, C_4$ \blue{(independent of $k$, since $\alpha_k\equiv\alpha$, $\beta_k\equiv\beta$, and $\tau$ is a constant; see \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl} below)} are chosen as:
\begin{align}
    \label{Eq: the coefficients of the Lyapunov function in SGDM}
    \begin{cases}
        C_{1} &= K_{1},\\
        C_{2} &= \frac{1}{2}(K_{2} - K_{1} + K_{3}),\\
        C_{3} &= K_{4},\\
        C_{4} &= \frac{\bar{L}_F + \frac{2L_F \beta^2}{\alpha}}{1 - 2\alpha - \nu}, 
            \quad \text{where } \bar{L}_{F} = 2L_{F} + K_{3} + K_{4}.
    \end{cases}
\end{align}
We also assume the parameters $\{\eta_{k}\}$, $\{\alpha_{k}, \beta_{k}\}$, $\tau$
in \Cref{alg:our_algorithm_sgdm} are set as follows:
\begin{align}
    \label{Eq: the parameters conditions of dualhash-sgdm in suppl}\eta_{k}= \frac{\eta}{L_{F}},\quad \alpha_{k}= \alpha, \quad \beta_{k}= \beta,\quad \tau \leq \frac{\sqrt{\tilde{\delta}}}{L_{F}},
\end{align}
where
$\eta =\frac{(1 - 2\alpha - \nu)/2}{\frac{\bar{L}_{F}}{L_{F}} +
\frac{(1 + \nu)\beta^{2}}{(1 - \alpha)\alpha}}$, $\alpha \in ( 0, \frac{1 - \nu}{2}
)$, $\beta \in (0,1)$, $\tilde{\delta}= c_{\delta}\delta^{2}$ are given constants
independent of $T$ with some $\nu$, $\delta$, $c_{\delta}> 0$.
The construction yields the following in-expectation approximate sufficient
descent property of \blue{$\{\Psi_m(\mathcal{Q}^{k})\}$}. The proof is provided in
\Cref{Subsubsec: proof descent SGDM}, and the positivity of the coefficients in
\eqref{Eq: the coefficients of the Lyapunov function in SGDM} and the descent
constants in \eqref{Eq: the descent coefficients of potential function in SGDM}
\blue{are} established in \Cref{lem:lyapunov-coef-positivity}.
\begin{lemma}
    \label{Lemma: approximate sufficient descent property in SGDM} Under \Cref{Ass: main assumptions} and the parameter conditions \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl}, for any $k \geq 1$, it holds that
    \begin{align}
        \label{Eq: the descent property of Lyapunov func in SGDM}\EE [\Psi_m(\mathcal{Q}^{k+1})] - \EE[\Psi_m(\mathcal{Q}^{k})] \leq -C_{\bB}\EE[\Delta_{\bB}^{k+1}+ \Delta_{\bB}^{k}] -C_{\bx}\EE [\Delta_{\bm{x}}^{k+1}+ \Delta_{\bm{x}}^{k}] + \frac{\sigma^{2}}{b_{k}L_{F}},
    \end{align}
    where the descent coefficients are given by
    \begin{align}
        \label{Eq: the descent coefficients of potential function in SGDM}\begin{cases}C_{\bB}&= C_{2}- K_{3}= \frac{K_2 - K_3 -K_1}{2}>0, \\ C_{\bx}&= \nu C_{4}>0.\end{cases}
    \end{align}
\end{lemma}

We now present the convergence rate of \Cref{alg:our_algorithm_sgdm} under the parameter conditions \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl}, which guides the selection of the batch size $b_k$. 

\begin{lemma}[Convergence rate of \Cref{alg:our_algorithm_sgdm}]
    \label{lemma: convergence rate sgdm} Under the conditions of \Cref{Lemma:
    approximate sufficient descent property in SGDM}, for any $T \geq 1$, there exists
    \blue{$R$ uniformly selected from $\{1,\ldots,T\}$} such that $(\bx^{R}, \bB
    ^{R}, \bLmbd^{R})$ satisfies \eqref{equ:esp} with
    \begin{align}
        \label{Eq: the convergence rate of sgdm}\mathbb{E}[\mathrm{dist}^{2}(0, \partial \mathcal{L}(\bx^{R}, \bB^{R}, \bLmbd^{R}))] \leq \frac{L_{\Delta}\Delta_{1}}{T}+ \blue{\frac{C_{\sigma}\sigma^{2}}{T}\sum_{k=1}^{T}\frac{1}{b_{k}}},
    \end{align}where \blue{$L_{\Delta} := \frac{\max\{S_{1}, S_{2}\} + 
    \max\{S_{3}, S_{4}\}}{\min\{C_{\bB}, C_{\bx}\}}$ with $S_1, S_2, S_3, S_4$ defined in 
    \eqref{Eq: the coefficients of stationarity error bound in SGDM}}, 
    $C_{\sigma} := \frac{L_{\Delta}}{L_{F}} + 3$, and $\Delta_{1} := \Psi_m^1 - \Psi_m^{*}$ denotes the initial objective gap with $\Psi_m^{1}= \Psi_m(\calQ^{1})$ and the lower bound $\Psi_m^{*}$. Here $L_{\Delta}$ and 
    $C_{\sigma}$ are constants independent of $T$. \blue{The same bound holds for the averaged quantity $\frac{1}{T}\sum_{k=1}^{T}\mathbb{E}[\mathrm{dist}^{2}(0, \partial \mathcal{L}(\bx^{k}, \bB^{k}, \bLmbd^{k}))]$.}
%  where 
% \(
% L_{\Delta} := \frac{\max\{S_{1}, S_{2}\} + \max\{S_{3}, S_{4}\}}{\min\{C_{\bB}, C_{\bx}\}}, \quad
% C_{\sigma} := \frac{L_{\Delta}}{L_{F}} + 3, \quad
% \Delta_{1} := \Psi_m(\calQ^{1}) - \Psi_m(\mathcal{Q}^{*}),
% \) and $S_1, S_2, S_3, S_4$ are defined in \eqref{Eq: the coefficients of stationarity error bound in SGDM}, independent of  \(T\).
% depending on the parameters \((L_F,  \, \eta, \,\alpha,\, \beta,\, \tau, \, \delta, \, \nu)\).
\end{lemma}
From \eqref{Eq: the convergence rate of sgdm}, to achieve the $\mathcal{O}(T^{-1/2})$ 
convergence rate, we need to set $b_k = \Theta(T)$. Therefore, combining 
\eqref{Eq: the parameters conditions of dualhash-sgdm in suppl} with $b_k = b = c_b T$, the complete parameter setting in \Cref{alg:our_algorithm_sgdm} is 
\begin{align}
\label{Eq: the parameters conditions of dualhash-sgdm}
    \eta_k = \frac{\eta}{L_F}, \quad \alpha_k = \alpha, \quad \beta_k = \beta, \quad 
    \tau \leq \frac{\sqrt{\tilde{\delta}}}{L_F}, \quad b_k = b = c_b T,
\end{align}
where $\eta$, $\alpha$, $\beta$, $\tau$, $\tilde{\delta}$, $c_b$ are given constants 
independent of $T$, and $\nu$, $\delta$, $c_{\delta} > 0$ are positive constants. The complexity result is established in \Cref{Theorem: the oracle complexity in SGDM}, with the proof in  \Cref{Subsec: proof theorem SGDM}.
\begin{theorem}
% [Oracle complexity of \Cref{alg:our_algorithm_sgdm}]
\label{Theorem: the oracle complexity in SGDM} 
Under \Cref{Ass: main assumptions}, let \(\{\bx^k, \bB^k, \bLmbd^k\}_{k \in \NN}\) be the iterate sequence from \Cref{alg:our_algorithm_sgdm} with the parameters satisfying \eqref{Eq: the parameters conditions of dualhash-sgdm}. Then there exists \blue{$R$ uniformly selected from $\{1,\dots,T\}$} such that \( (\bx^R, \bB^R, \bLmbd^R )\) satisfies \eqref{equ:esp} provided that \(T\) satisfies
\begin{align}
    \label{Eq: the iterate number upper bound of algorithm SGDM}
        T = \left\lceil \frac{L_{\Delta} \Delta_1+ C_{\sigma}\frac{\sigma^2}{c_b}}{\varepsilon^2} \right\rceil.
\end{align}
Moreover, the total number of SFO calls is of order \(\calO(\varepsilon^{-4})\).
\end{theorem}
\subsection{Oracle complexity of DualHash-StoRM} 
Before proceeding, we introduce an additional assumption, also known as the mean-squared smoothness
assumption, widely used in variance reduction methods \cite{arjevani2023lower,fang2018spider,tran2020hybridvrsgd,driggs2021stochastic}.
\blue{
\begin{assumption}
\label{Ass: the mean squared smoothness of F}
    There exists $L_{F} > 0$ such that for all $\bx, \bx' \in \RR^{\blue{q}}$ and $\bB, \bB' \in \RR^{n\times d}$, it holds that
\begin{align}
    \label{eq:mss}
    \frac{1}{n}\sum_{i=1}^{n}
    \left\| \nabla_{\bx} F_{i}(\bx,\bb_i) - \nabla_{\bx} F_{i}(\bx',\bb_i')\right\|^{2}
    \leq L^{2}_F \left[ \|\bx - \bx'\|^{2} + \|\bB - \bB'\|^{2} \right]. 
\end{align}
\end{assumption}}The condition \eqref{eq:mss} is slightly stronger than the \(L_F\)-smoothness of the finite-sum function \(F\) in \Cref{Ass: main assumptions}(ii). In particular, this assumption implies \Cref{Ass: main assumptions}(ii) by Jensen's inequality, but not conversely.

Let us denote the gradient error \[\be^{k} = \blue{\bg_s^{k}} - \nabla_{\bx}F(\bx^{k}, \bB^{k})
,\]where $\blue{\bg_s^k}$ is defined in \eqref{Eq: the stochastic approximate in STORM}. Similarly, the following lemma estimates the one-iteration progress
of the Lagrangian function \eqref{Eq: simplified_lagrangian} via the $\bx$-update \eqref{Eq: the update of x in STORM}.
\begin{lemma}[One-iteration progress on $\bx$ \cite{xu2023momentum}]
    \label{Lemma: One-iteration progress of x in STORM} Consider the sequence $\{\bx^k, \bB^k, \bm{\Lambda}^k\}_{k \in \NN}$ generated by \Cref{alg:our_algorithm_storm}. Under \Cref{Ass: main assumptions}(i)-(ii), for any $k \geq 1$, it holds that
    \begin{align}
        \label{Eq: the recursive relation of L on x}\calL (\bx^{k+1}, \bB^{k}, \bLmbd^{k}) - \calL (\bx^{k}, \bB^{k}, \bLmbd^{k}) \leq \frac{\eta_{k}}{2}\| \be^{k}\|^{2}- (\eta_{k}^{-1}- L_{F})\Delta_{\bx}^{k+1}.
    \end{align}
\end{lemma}

    - C_{1}\Delta_{\bB}^{k+1}+ C_{2}\Delta_{\bB}^{k}- C_{3}\Delta_{\bm{x}}^{k+1},

    % \\

\eqref{Eq: the coefficients of the Lyapunov function in SGDM}.
In the next lemma, we establish an approximate sufficient descent property 
of $\Psi_r$, with the proof provided in \Cref{Subsubsec: proof descent STORM}.
\begin{lemma}
\label{Lemma: approximate sufficient descent property storm}
    Under \Cref{Ass: main assumptions}(i)-(ii) and \Cref{Ass: the mean squared smoothness of F}, with $\tau$ satisfying \eqref{Eq: the parameters conditions of dualhash-sgdm in suppl}, for any $k \geq 1$, it holds that
    \begin{align}
        \label{Eq: the descent property of Lyapunov func in storm}\Psi_r(\mathcal{Q}^{k+1}) - \Psi_r(\mathcal{Q}^{k}) \leq -C_{\bB}\left( \Delta_{\bB}^{k+1}+ \Delta_{\bB}^{k}\right)- C_{\bx}^{k}\Delta_{\bm{x}}^{k+1}+ \frac{\eta_{k}}{2}\| \be^{k}\|^{2},
    \end{align}
    where the descent coefficients are given by:
    \begin{align}
        % \label{Eq: the descent coefficients of potential function}
        \begin{cases}C_{\bB}&= C_{2}- K_{3}= \frac{K_2 - K_3 -K_1}{2}>0, \\ C_{\bx}^{k}&=\eta_{k}^{-1}- \tilde{L}_{F}, \text{ where }\tilde{L}_{F}= L_{F}+ K_{3}+ K_{4}.\end{cases} \notag 
    \end{align}
    Furthermore, the following inequalities hold:
    \begin{align}
        \label{Eq: bound B and x in storm}
        \left\{ \begin{array}{ll}C_{\bB} \left( \Delta_{\bB}^{k+1} + \Delta_{\bB}^k \right) \leq \Psi_r(\mathcal{Q}^k) - \Psi_r(\mathcal{Q}^{k+1}) + \frac{\eta_{k}}{2} \| \be^k\|^2, \\ C_{\bx}^k \Delta_{\bm{x}}^{k+1} \leq \Psi_r(\mathcal{Q}^k) - \Psi_r(\mathcal{Q}^{k+1}) +\frac{\eta_{k}}{2} \| \be^k\|^2.\end{array} \right.
    \end{align}
\end{lemma}

\begin{lemma}[Convergence rate of \Cref{alg:our_algorithm_storm}]
    \label{Lemma: convergence rate of storm} Under \Cref{Ass: main assumptions}(i)--(iii), \Cref{Ass: the mean squared smoothness of F}, and the parameter conditions \eqref{Eq: the parameters conditions in dualhash-storm}--\eqref{Eq: the constants for storm}, for any $T \geq 1$, there exists \blue{$R$ uniformly selected from $\{1,\ldots,T\}$} such that $(\bx^{R}, \bB^{R}, \bLmbd^{R})$ satisfies \eqref{equ:esp} with

\begin{theorem}
\label{Theorem: the oracle complexity in STORM}
Under \Cref{Ass: main assumptions}(i), (iii) and \Cref{Ass: the mean squared smoothness of F}, let \(\{\bx^k, \bB^k, \bLmbd^k\}_{k \in \NN}\) be the iterate sequence from \Cref{alg:our_algorithm_storm} with the parameters satisfying \eqref{Eq: the parameters conditions in dualhash-storm}--\eqref{Eq: the constants for storm}. Then there exists \blue{$R$ uniformly selected from $\{1,\dots,T\}$} such that \( (\bx^R, \bB^R, \bLmbd^R )\) satisfies \eqref{equ:esp} whenever
\begin{align}
    \label{Eq: the iterate number upper bound of algorithm VR}
        T = \left\lceil \frac{\left(\bar{N}\,\Theta\right)^{\frac{3}{2}}}{\varepsilon^3} \right\rceil,
\end{align}
with $\bar{N}$ and $\Theta$ from \Cref{Lemma: convergence rate of storm}. Moreover, the total number of SFO calls is of order \(\calO(\varepsilon^{-3})\).
\end{theorem} 

\begin{remark}
\begin{enumerate}[label=(\roman*), leftmargin=2em]
\item The results in Theorems \ref{Theorem: the oracle complexity in SGDM} and \ref{Theorem: the oracle complexity in STORM} are novel for stochastic, non-convex, multi-block primal-dual optimization, a challenging area where existing work has largely focused on one-block \cite{xu2023momentum,shi2025momentum,tran2022hybrid} or deterministic settings \cite{ADMMNolinear2024}.  
    \item  \Cref{alg:our_algorithm_sgdm} with constant parameters from \eqref{Eq: the parameters conditions of dualhash-sgdm} attains both oracle and sample \(\mathcal{O}(\varepsilon^{-4})\) complexities, which matches that of stochastic first-order methods (SFOMs) for one-block stochastic optimization \cite{Ghadimi2013MinibatchSA}. Meanwhile, \Cref{alg:our_algorithm_storm} with the refined constant parameters from \eqref{Eq: the parameters conditions in dualhash-storm} attains both oracle and sample $\mathcal{O}(\varepsilon^{-3})$ complexities. This matches the best-known lower bound \cite{arjevani2023lower} and the complexity of variance-reduced SFOMs for single-block problems \cite{shi2025momentum,xu2023momentum}. Notably, despite tackling a multi-block problem, our primal-dual framework preserves this efficiency without degrading the dependence on $\varepsilon$.
%    \item  While DualHash-StoRM with parameters in \eqref{Eq: the parameters conditions in dualhash-storm} achieves better theoretical complexity than DualHash-StoM under \eqref{Eq: the parameters conditions of dualhash-sgdm}, the latter relies on a weaker assumption (\Cref{Ass: main assumptions} [ii]). In our experiments, while DualHash-StoRM enjoys faster training convergence, DualHash-StoM enerally delivers better final retrieval performance.
\end{enumerate}
\end{remark}
