\section{Stochastic Primal-Dual Hashing Algorithms}
\label{Section: Methodology}
\subsection{Algorithmic Framework}
% Inspired by PDHG \cite{chambolleFirstOrderPrimalDualAlgorithm2011},
% and its nonconvex extension PPDG \cite{guoPreconditionedPrimalDualGradient2023}, 
Let us consider the equivalent constrained formulation of Problem \eqref{Prop: block-wise regularized problem}:
\begin{align} 
\min_{\bx \in \RR^{d_{\bx}}, \bB \in \RR^{n\times d}, \bY \in \RR^{n\times d}} &\quad F(\bx, \bB) + h(\bY) \notag \\
\text{s.t. } &\quad \bY = \bB \nonumber ,
\end{align} whose corresponding dual problem is 
\begin{align}
\label{Prop: the dual problem}
\max_{\bLmbd \in \RR^{n\times d}} \min_{\bx \in \RR^{d_{\bx}}, \bB \in \RR^{n\times d}, \bY \in \RR^{n\times d}} &\quad F(\bx, \bB) + h(\bY) + \langle \bLmbd, \bB - \bY \rangle. \tag{D}
\end{align} 
Recalling the definition of the conjugate function \cite{Rockafellar1998VariationalA}:
\begin{equation}
% \label{eq:fenchel_conjugate}
h^*(\bx) := \sup_{\by \in \RR^{n}} \{ \langle \bx, \by \rangle - h(\by) \}, \quad \bx \in \RR^{n}
\end{equation} and observing that the minimization with respect to $\bY$ in Problem \eqref{Prop: the dual problem} can be performed independently, we then simplify the Lagrangian function in the problem \eqref{Prop: the dual problem} to 
\begin{equation}
\label{Eq: simplified_lagrangian}
\calL(\bx, \bB, \bLmbd) := F(\bx, \bB) + \langle \bB, \bLmbd \rangle - h^*(\bLmbd).
\end{equation}
Through this transformation, we replace $h(\bY)$ with $-h^*(\bLmbd)$ and optimize the \(h^*\) in the dual space. The convexity of $h^*$ makes $\operatorname{prox}_{h^*}(\cdot)$ often much easier to compute (potentially having a closed-form solution) than $\operatorname{prox}_{h}(\cdot)$.
For the W-type regularization on the matrix variable $\bB \in \RR^{n \times d}$,
\begin{align*}
&h(\bB)= \lambda\||\bB|-\bm{1}\|_1 = \sum_{i=1}^{n}\sum_{j=1}^{d}\lambda\big||b_{ij}|-1\big|,
\end{align*}
its conjugate is separable element-wise:
\begin{align*}
&h^*(\bLmbd) = \sum_{i=1}^{n}\sum_{j=1}^{d} h^*(\Lambda_{ij}),\,\, \text{where } h^*(\Lambda_{ij}) = \begin{cases}
    |\Lambda_{ij}|, & \text{if } \Lambda_{ij} \in [-\lambda, \lambda],\\
    +\infty, & \text{otherwise},
    \end{cases}
\end{align*} whose proximal operator in the piecewise solution (see \Cref{fig:w-conj-prox} for details) is 
\begin{align}
    \label{Eq: the proximal mapping of conjugate function}
    [\operatorname{prox}_{\tau h^*}(\bY)]_{ij} = 
    \begin{cases}
        \lambda, & Y_{ij} > \lambda +\tau,\\
        Y_{ij} -\tau, & \tau < Y_{ij} \leq \lambda +\tau, \\
        0, & -\tau \leq Y_{ij} \leq\tau, \\
        Y_{ij} +\tau, &  - \lambda -\tau \leq Y_{ij} < -\tau, \\
        -\lambda, & Y_{ij} < -\lambda -\tau.
    \end{cases}
    %  \notag
\end{align}
This motivates us to perform a proximal step to update \(\bLmbd\).

We now outline the \textbf{DualHash} framework, which employs an alternating update scheme for $\bx, \bB, \bm{\Lambda}$. 

First, we adopt a stochastic first-order method for the $\bx$-update since computing the full gradient \( \nabla_{\bx} F(\bx, \bB) \) is computationally expensive when the number of samples \(n\) is large, as is typical in deep learning. Specifically, at each iteration $k$, we update $\bx$ by utilizing a stochastic gradient descent step
\[
\bx^{k+1} = \bx^k - \eta_k \tilde{\nabla}_{\bx} F(\bx^k, \bB^k),
\]where $\tilde{\nabla}_{\bx} F(\bx^k, \bB^k)$ is a stochastic estimator of $\nabla_{\bx} F(\bx^k, \bB^k)$.

Second, the auxiliary variable $\bB$ is updated via a vanilla gradient descent step \eqref{Eq: the update of B}; this suffices because the objective \eqref{Eq: simplified_lagrangian} with respect to $\bB$ is smooth with an inexpensive gradient, 
\begin{align}
   \bB^{k+1} = \bB^k - \tau \nabla_{\bB}\calL(\bx^{k+1}, \bB^k, \bLmbd^k) = \bB^k - \tau \left( \nabla_{\bB} F(\bx^{k+1}, \bB^k) + \bLmbd^k \right). \label{Eq: the update of B}
\end{align}

Finally, the dual variable $\bLmbd$ is updated via proximal gradient ascent on $h^*$, using an extrapolation step introduced in \cite{chambolleFirstOrderPrimalDualAlgorithm2011}. The update solves the following optimization problem:
\begin{align}
    \bLmbd^{k+1} &= \text{prox}_{(1/\tau h^*)} \left( \bLmbd^k + \left(2\bB^{k+1} - \bB^k  \right) \right) \notag \\
    &=  \arg\min_{\bLmbd \in \RR^{n\times d} } \left\{ h^*(\bLmbd) - \langle \bLmbd, 2\bB^{k+1} - \bB^k \rangle  + \frac{\tau}{2} \| \bLmbd - \bLmbd^k\|^2 \right\}. 
    \label{Eq: the update of lambda}
\end{align}

Our framework accommodates flexible choices of $\tilde{\nabla}_{\bx} F(\bx^k, \bB^k)$, enabling the incorporation of advanced momentum techniques to enhance both theoretical complexity and practical performance.

\subsubsection{DualHash with Momentum Acceleration}
We first propose a stochastic first-order algorithm with momentum acceleration for solving Problem \eqref{Prop: block-wise regularized problem}. At each iteration $k$, we uniformly sample a mini-batch $\calJ_k \subseteq \{1,\cdots,n\}$ with $b_k = |\calJ_k| \leq n$, and compute $\tilde{\nabla}_{\bx} F(\bx^k, \bB^k) := \nabla_{\bx} F(\bx^k, \bB^k; \calJ_k)$ evaluated at the extrapolated point $\bz^k$:
\begin{align}
\label{Eq: the stochastic approximate in SGDM}
    \bG^k = \nabla_{\bx} F(\bz^k, \bB^k; \calJ_k),\quad \text{where } \bz^k = \bx^k + \beta_k (\bx^k - \bx^{k-1}).
\end{align}The update of \( \bx^k\) is then:
\begin{align}
\label{Eq: the update of x in SGDM}
    \bx^{k+1} = \by^k - \eta_k \bG^k,\,\text{where } \by^k = \bx^k + \alpha_k (\bx^k - \bx^{k-1}). 
\end{align}The overall algorithm, DualHash-StoM, is detailed in \Cref{alg:our_algorithm_sgdm}.
\begin{remark}
The momentum acceleration framework in \eqref{Eq: the update of x in SGDM} covers these two accelerations: Nesterov acceleration  \cite{Nesterov1983AMF} corresponds to $\alpha_k = \beta_k \neq 0$; the heavy-ball method \cite{Polyak1964} is obtained with $\beta_k = 0$; vanilla SGD is recovered when $\alpha_k = \beta_k = 0$. In this paper, we set $\alpha_k, \beta_k \in (0,1)$ for all $k \geq 1$.
\end{remark}
\begin{algorithm}[t]
\caption{DualHash-StoM}
\label{alg:our_algorithm_sgdm}
\begin{algorithmic}[1]
\STATE \textbf{Initialize:} $\bx^1 = \bz^1 = \by^1 \in \RR^{d_{\bx}} $, \( \bB^1 \in \RR^{n\times d},\, \bLmbd^1\in \RR^{n\times d}\), parameters $\{\eta_k\} $, $\tau$, $b_k$, $\{\alpha_k, \beta_k \in (0,1)\}$
\FOR{$k = 1, 2, \ldots$}               
  \STATE Compute \(\bG^k\) by \eqref{Eq: the stochastic approximate in SGDM}
  \STATE Update $\bx^{k+1}$ by \eqref{Eq: the update of x in SGDM}
  \STATE Update $\bB^{k+1}$ by \eqref{Eq: the update of B}
  \STATE Update $\bm{\Lambda}^{k+1}$ by \eqref{Eq: the update of lambda}
\ENDFOR
\STATE \textbf{Return:} $\bx_R$ where $R$ is uniformly sampled from $\{2, \ldots, T+1\}$
\end{algorithmic}
\end{algorithm}
\subsubsection{DualHash with Recursive Momentum}
 % StoRM utilizes a recursive momentum mechanism to reduce gradient variance, which is motivated by SGDM. Unlike some other variance reduction techniques (e.g., SVRG \cite{johnsonAcceleratingStochasticGradient2013}, SARAH \cite{SARAH2017}) that require full-gradient or large-batch gradient computations, 
% StoRM achieves variance reduction without require full-gradient or large-batch gradient computations. 
Alternatively, we consider the StoRM estimator \cite{cutkoskyMomentumBasedVarianceReduction2019} for the $\bx$-update, which achieves variance reduction without requiring full-gradient or large-batch gradient computations. Given its practical effectiveness in deep learning \cite{levy2021storm+}, we construct a recursive momentum estimator $\bD^k$ for $\tilde{\nabla}_{\bx} F(\bx^k, \bB^k)$ as follows:
{
\begin{align}
\label{Eq: the stochastic approximate in STORM}
\bD^k =
\begin{cases} 
\nabla_{\bx} F(\bx^1, \bB^1; \mathcal{J}_1), &\text{if }  k = 1, \\
(1 - \rho_{k-1}) \big( \bD^{k-1} - \nabla_{\bx} F(\bx^{k-1}, \bB^{k-1};\mathcal{J}_{k-1})\big)+\nabla_{\bx} F(\bx^k, \bB^k; \mathcal{J}_k), & \text{if } k > 1.
\end{cases}
\end{align}}The update of $\bx$ is then 
\begin{align}
\label{Eq: the update of x in STORM}
    \bx^{k+1} = \bx^k - \eta_k \bD^k.
\end{align} The overall algorithm, DualHash-StoRM, is detailed in \Cref{alg:our_algorithm_storm}.
\begin{algorithm}[t]
\caption{DualHash-StoRM}
\label{alg:our_algorithm_storm}
\begin{algorithmic}[1]
\STATE  \textbf{Initialize:} $\bx^1  \in \RR^{d_{\bx}}$, \( \bB^1 \in \RR^{n\times d},\, \bLmbd^1\in \RR^{n\times d}\), parameters $\{\eta_k\} $, $\tau $, $b_k$, $\{\rho_k\in (0,1)\}$
\FOR{$k = 1, 2, \ldots$}
\STATE Compute $\bD^k$ by \eqref{Eq: the stochastic approximate in STORM}
\STATE Update $\bx^{k+1}$ by \eqref{Eq: the update of x in STORM}.
\STATE Update $\bB^{k+1}$ by \eqref{Eq: the update of B}.
\STATE Update $\bm{\Lambda}^{k+1}$ by \eqref{Eq: the update of lambda}.
\ENDFOR
\STATE \textbf{Return:} $\bx_R$ where $R$ is uniformly sampled from $\{2, \ldots, T+1\}$
\end{algorithmic}
\end{algorithm}
